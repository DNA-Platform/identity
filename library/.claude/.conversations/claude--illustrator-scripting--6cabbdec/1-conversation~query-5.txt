/// <reference path="./utils.ts" />

interface Item {
    [key: string]: any;
    name?: string;
    readonly typename: string;
    readonly parent: any;
    readonly length?: number;
    selected?: any;
    remove?(): void;
    removeAll?(): void;
    getIndex?(): number;
    getByName?(name: string): Item;
    add?(...args: any[]): Item;
    insert?(...args: any[]): void;
    duplicate?(relativeObject: object, insertionLocation: ElementPlacement): Item;
    move?(relativeObject: object, insertionLocation: ElementPlacement): Item;
    q: Queryable<this, any>;
}

class EmptyItem implements Item {
    name: string = "";
    typename: string = "EMPTY";
    parent: Application = app;
    length: number = 0;
    selected: any = false;
    remove(): void { }
    removeAll(): void { }
    getIndex(): number { return -1; }
    getByName(name: string): Item { return this; }
    add(...args: any[]): Item { return this; }
    insert(...args: any[]): void { }
    duplicate(relativeObject: object, insertionLocation: ElementPlacement): Item { return this; }
    move?(relativeObject: object, insertionLocation: ElementPlacement): Item { return this; }
    q: Queryable<this, any> = undefined as any;

    private constructor() { }

    private static _empty: EmptyItem;
    static empty(): EmptyItem {
        if (EmptyItem._empty) return EmptyItem._empty;
        EmptyItem._empty = new EmptyItem();
        return EmptyItem._empty;
    }
}

class QueryItem {
    _item: any;
    code: number;
    enumerable: boolean = false;
    app: Application = app;
    typename() { return this._item?.typename ?? ""; }
    parent(): QueryItem { return this._item.parent ? q(this._item.parent) : Queryable.empty(); }
    empty(): boolean { return this._item instanceof EmptyItem; }
    activate() { if (this._item) this._item.selected = true; }
    activated() { return !!this._item?.selected; }
    inactivate() { if (this._item) this._item.selected = false; }

    name(value?: string): string {
        if (value !== undefined) {
            this._item.name = value;
            return this._item.name;
        }
        return this._item?.name ? this._item.name.toString() : "";
    }

    constructor(item: any = EmptyItem.empty()) {
        this.code = QueryItem.nextCode();
        this._item = item;
    }

    remove() { this._item?.remove?.(); }

    insert(...args: QueryItem[]) {
        f(args).do(i => i.move(this, ElementPlacement.PLACEATEND));
    }

    duplicate(relativeObject: QueryItem, insertionLocation: ElementPlacement) {
        var duplicate = this._item?.duplicate?.(relativeObject._item, insertionLocation);
        return q(duplicate);
    }

    move(relativeObject: QueryItem, insertionLocation: ElementPlacement): QueryItem {
        var moved = this._item?.move?.(relativeObject._item, insertionLocation);
        return q(moved);
    }

    cast<T extends QueryItem>() { return this as any as T }

    private static _nextCode: number = 1;
    private static nextCode(): number {
        var code = QueryItem._nextCode;
        QueryItem._nextCode += 1;
        return code;
    }

    private static _empty: Queryable<any, any>;
    static empty(): Queryable<any, any> {
        if (QueryItem._empty) return QueryItem._empty;
        QueryItem._empty = new Queryable<any, any>(EmptyItem.empty(), undefined);
        QueryItem._empty._item.q = QueryItem._empty;
        return QueryItem._empty;
    }
}

class Queryable<I extends Item, T extends QueryItem> extends QueryItem {
    item(): I { return this._item; }
    collectionProperty: CollectionProperty | undefined;
    active() { return !!this._item.selected; }
    activate() { this._item.selected = true; }
    inactivate() { this._item.selected = false; }

    constructor(item: I, collectionProperty?: CollectionProperty) {
        super(item);
        this._item = item;
        this.collectionProperty = collectionProperty;
        this.enumerable = this.collectionProperty !== undefined;
    }

    collection(): any[] {
        if (!this._item || !this.enumerable || !this.collectionProperty) return [];
        var collection = this._item[this.collectionProperty] as any[];
        if (!collection) return [];
        return collection;
    }

    query(selector: string = ":"): A<T> {
        var collection = this.collection();
        if (collection.length == 0) return f();
        var indexes = evaluateRangeSelector(selector, collection.length)
        var len = indexes.length;
        var result: A<T> = f();
        for (var i = 0; i < len; i++) {
            var item = collection[indexes[i]];
            if (item)
                result.push(q(item));
        }

        return result;
    }

    index(): number { 
        return (this.parent() as Queryable<any, any>).firstIndex(i => i._item == this._item); 
    }

    at(index: number): T {
        return this.query(`${index}`).at(0);
    }

    get(selector: string): A<T> {
        return this.query(selector);
    }

    getByName(name: string): Queryable<any, any> { 
        var collection = this.collection() as any;
        return q(collection?.getByName?.(name)); 
    }

    length(): number { return this.collection().length; }

    create(name?: string): T {
        var collection = this.collection() as any;
        var newItem = q(collection.add()) as T;
        if (name !== undefined)
            newItem.name(name);

        if (this.length() > 1)
            newItem.move(this.last() as T, ElementPlacement.PLACEAFTER);

        return newItem;
    }

    replace(index: number, item: T): void {
        var itemAtIndex = this.at(index);
        item.move(itemAtIndex, ElementPlacement.PLACEBEFORE);
        itemAtIndex.remove?.();
    }

    insertAt(index: number, item: T): void {
        var itemBeforeIndex = this.at(index - 1);
        item.move(itemBeforeIndex, ElementPlacement.PLACEAFTER);
    }

    insertManyAt(index: number, items: A<T>) {
        var len = items.length();
        index = evaluateNegativeInsertIndex(index, this.collection.length);
        for (var i = 0; i < len; i++) {
            this.insertAt(index + i, items.at(i));
        }
    }

    delete(item: T) {
        item.remove();
    }

    deleteMany(items: A<T>) {
        items.do(item => item.remove());
    }

    deleteAt(index: number, count: number = 1) {
        for (var i = 0; i < count; i++)
            this.delete(this.at(index)._item);
    }

    removeAll() {
        this.query().do(item => { item.remove() });
    }

    forEach(callback: (item: T, index?: number) => any | void): any | void {
        return this.query().forEach(callback);
    }

    do(callback: (item: T, index?: number) => any | void): any | void {
        return this.query().forEach(callback);
    }

    doOnce(callback: (item: T, index?: number) => boolean): boolean {
        var collection = this.collection();
        for (let i = 0; i < collection.length; i++) {
            if (callback(q(collection[i]), i)) {
                return true;
            }
        }
        return false;
    }

    select<U>(selector: (item: T) => U): A<U> {
        return this.query().select(selector);
    }

    where(predicate: (item: T) => boolean): A<T> {
        return this.query().where(predicate);
    }

    first(predicate?: (item: T) => boolean): T | undefined {
        var len = this.length();
        if (!predicate) return len > 0 ? this.at(0) : undefined;
        for (let i = 0; i < len; i++) {
            var item = this.at(i);
            if (predicate(item)) return item;
        }
        return undefined;
    }

    firstIndex(predicate: (item: T) => boolean): number {
        var len = this.length();
        for (let i = 0; i < len; i++) {
            var item = this.at(i);
            if (predicate(item)) return i;
        }
        return -1;
    }

    last(predicate?: (item: T) => boolean): T | undefined {
        var len = this.length();
        if (!predicate) return len > 0 ? this.at(-1) : undefined;
        for (let i = len - 1; i >= 0; i--) {
            var item = this.at(i);
            if (predicate(item)) return item;
        }
        return undefined;
    }

    lastIndex(predicate: (item: T) => boolean): number {
        var len = this.length();
        for (let i = len - 1; i >= 0; i--) {
            var item = this.at(i);
            if (predicate(item)) return i;
        }
        return -1;
    }

    skip(count: number): A<T> {
        return this.query(`${count}:`);
    }

    take(count: number): A<T> {
        if (count == 0) return f() as A<T>;
        count = Math.min(count, this.length());
        return this.query(`:${count - 1}`);
    }

    any(predicate?: (item: T) => boolean): boolean {
        var item = this.first(predicate);
        return !!item;
    }

    includes(item: T): boolean {
        return this.any(i => i._item == item._item);
    }
}

class ApplicationQuery extends Queryable<Application, DocumentQuery<LayerQuery<PageItemQuery>>> {

    constructor(application: Application) {
        super(application, "documents");
    }

    documents(): ApplicationQuery { return this; }
    activeDocument(): DocumentQuery<LayerQuery<PageItemQuery>> {
        return q(this.app.activeDocument);
    }

    exists(fileName: string): boolean {
        // Check if the document exists in the currently open documents
        var exists = false;
        this.documents().forEach(doc => {
            if (doc.name() === fileName) {
                exists = true;
            }
        });

        if (exists) {
            return true;
        }

        // Check if the document exists in the file system
        const file = new File(combinePaths(DOCUMENTS_PATH, fileName));
        return file.exists;
    }

    createDocument(fileName: string, activate: boolean = true): DocumentQuery {
        var lastActiveDoc = this.app.activeDocument;
        var doc: DocumentQuery = q(this._item.documents.add(DocumentColorSpace.RGB));
        doc.save(fileName);

        if (activate) {
            doc.activate();
        } else {
            this.app.activeDocument = lastActiveDoc;
        }

        return doc;
    }

    openDocument(fileName: string, create: boolean = false, activate: boolean = true): DocumentQuery {
        const file = new File(combinePaths(DOCUMENTS_PATH, fileName));

        var openDoc: DocumentQuery | undefined;
        if (this.exists(fileName)) {
            this.documents().doOnce(doc => {
                if (doc.name() === fileName) {
                    openDoc = doc;
                    return true;
                }
                return false;
            });

            if (openDoc) {
                if (activate) openDoc.activate();
                return openDoc;
            } else {
                openDoc = q(this.app.open(file));
                if (activate) openDoc?.activate();
                return openDoc as DocumentQuery;
            }
        }

        if (create) {
            return this.createDocument(fileName, activate);
        }

        throw `Failed to open document: ${fileName}`;
    }

    openResource(fileName: string, activate = true): DocumentQuery {
        return this.openDocument(combinePaths("resources", fileName), activate);
    }

    deleteDocument(fileName: string) {
        if (api.exists(fileName)) {
            // Try to delete through Illustrator first if it's open
            api.documents().doOnce(doc => {
                if (doc.name() === fileName) {
                    doc.remove();
                    return true;
                }
                return false;
            });

            // Also try to delete the file directly
            const file = new File(combinePaths(DOCUMENTS_PATH, fileName));
            if (file.exists) {
                file.remove();
            }
        }
    }

    redraw(milliseconds: number = 1000): void {
        // Force Illustrator to redraw the canvas
        app.redraw();

        // Introduce a delay using a blocking loop
        const start = new Date().getTime();
        while (new Date().getTime() - start < milliseconds) {
            // Keep the loop running until the specified time has passed
        }
    }

    reset(milliseconds: number = 1000) {
        this.app.selection = [];
        this.redraw(milliseconds);
    };
}

class DocumentQuery<T extends QueryItem = LayerQuery<PageItemQuery>> extends Queryable<Document, T> {
    private _layers: DocumentQuery<LayerQuery<PageItemQuery>> | undefined;
    private _artboards: DocumentQuery<ArtboardQuery> | undefined;
    private _stories: DocumentQuery<StoryQuery<TextRangeQuery>> | undefined;
    private _variables: DocumentQuery<VariableQuery> | undefined;
    private _swatches: DocumentQuery<SwatchQuery> | undefined;
    private _brushes: DocumentQuery<BrushQuery> | undefined;
    private _symbols: DocumentQuery<SymbolQuery> | undefined;
    private _patterns: DocumentQuery<PatternQuery> | undefined;
    private _spots: DocumentQuery<SpotQuery> | undefined;
    private _gradients: DocumentQuery<GradientQuery> | undefined;
    private _tags: DocumentQuery<TagQuery> | undefined;

    parentApp(): ApplicationQuery { return this.app.q; }
    active(): boolean { return this.app.activeDocument == this._item; }
    activate() { this.app.activeDocument = this._item; }
    inactivate() { this.app.activeDocument = f(this.app.documents).at(-1); }
    remove() { this._item?.close?.(); }
    activeArtboard(): ArtboardQuery {
        var artboards = this._item.artboards as Artboards;
        var index = artboards.getActiveArtboardIndex();
        return this.artboards().at(index);
    }

    constructor(doc: Document, collectionProperty: CollectionProperty = "layers") {
        super(doc, collectionProperty);
    }

    create(name?: string): T {
        if (this.collectionProperty !== "artboards") 
            return super.create(name);

        var collection: Artboards = this.collection() as any;
        var artboard = q(collection.add(ArtboardQuery.defaultRect()));
        artboard.name(name);

        if (this.length() > 1)
            artboard.move(this.last() as T, ElementPlacement.PLACEAFTER);

        return artboard;
    }

    removeAll(): void {
        if (this.collectionProperty == "artboards") {
            var artboard = this.create("Artboard 1");
            this.take(this.length()).select(a => a._item).do(a => a?.remove());
            artboard.activate();
        } else if (this.collectionProperty == "layers") {
            super.removeAll();
            this.at(0).name("Layer 1");
        } else {
            super.removeAll();
        }
    }

    layers() {
        if (this.collectionProperty === "layers") return this;
        if (this._layers) return this._layers;
        this._layers = new DocumentQuery<LayerQuery<PageItemQuery>>(this._item, "layers");
        return this._layers;
    }

    artboards() {
        if (this.collectionProperty === "artboards") return this;
        if (this._artboards) return this._artboards;
        this._artboards = new DocumentQuery<ArtboardQuery>(this._item, "artboards");
        return this._artboards;
    }

    stories() {
        if (this.collectionProperty === "stories") return this;
        if (this._stories) return this._stories;
        this._stories = new DocumentQuery<StoryQuery<TextRangeQuery>>(this._item, "stories");
        return this._stories;
    }

    variables() {
        if (this.collectionProperty === "variables") return this;
        if (this._variables) return this._variables;
        this._variables = new DocumentQuery<VariableQuery>(this._item, "variables");
        return this._variables;
    }

    swatches() {
        if (this.collectionProperty === "swatches") return this;
        if (this._swatches) return this._swatches;
        this._swatches = new DocumentQuery<SwatchQuery>(this._item, "swatches");
        return this._swatches;
    }

    brushes() {
        if (this.collectionProperty === "brushes") return this;
        if (this._brushes) return this._brushes;
        this._brushes = new DocumentQuery<BrushQuery>(this._item, "brushes");
        return this._brushes;
    }

    symbols() {
        if (this.collectionProperty === "symbols") return this;
        if (this._symbols) return this._symbols;
        this._symbols = new DocumentQuery<SymbolQuery>(this._item, "symbols");
        return this._symbols;
    }

    patterns() {
        if (this.collectionProperty === "patterns") return this;
        if (this._patterns) return this._patterns;
        this._patterns = new DocumentQuery<PatternQuery>(this._item, "patterns");
        return this._patterns;
    }

    spots() {
        if (this.collectionProperty === "spots") return this;
        if (this._spots) return this._spots;
        this._spots = new DocumentQuery<SpotQuery>(this._item, "spots");
        return this._spots;
    }

    gradients() {
        if (this.collectionProperty === "gradients") return this;
        if (this._gradients) return this._gradients;
        this._gradients = new DocumentQuery<GradientQuery>(this._item, "gradients");
        return this._gradients;
    }

    tags() {
        if (this.collectionProperty === "tags") return this;
        if (this._tags) return this._tags;
        this._tags = new DocumentQuery<TagQuery>(this._item, "tags");
        return this._tags;
    }

    save(fileName?: string): void {
        try {
            if (fileName === undefined) {
                this._item.save();
                return;
            }
            const file = new File(combinePaths(DOCUMENTS_PATH, fileName));
            this._item.saveAs(file);
        } catch (error: any) {
            throw new Error(`Failed to save document '${fileName}': ${error.message}`);
        }
    }
}

class ArtboardQuery extends Queryable<Artboard, QueryItem> {
    document() { return (this.parent() as DocumentQuery).artboards(); }
    index(): number { return this.document().firstIndex(i => i._item == this._item); }
    activate() { (this.parent()._item.artboards as Artboards).setActiveArtboardIndex(this.index()); }
    activated() { return (this.parent()._item.artboards as Artboards).getActiveArtboardIndex() === this.index(); }
    inactivate() { this.parent()._item.artboards.setActiveArtboardIndex(0); }

    constructor(artboard: Artboard) {
        super(artboard);
    }

    copy(target: DocumentQuery): ArtboardQuery {
        // Activate the source
        var source = this.document();

        // Copy these while the source is active
        var artboardRect = clone(this.item().artboardRect);
        var artboardName = this.name();
        var artboardRulerOrigin = clone(this.item().rulerOrigin); 

        // Get the source view properties
        var sourceZoom = source.item().activeView.zoom;
        var sourceCenterPoint = source.item().activeView.centerPoint;

        // Remove the target artboards
        target.artboards().removeAll();
        var targetBoard = target.activeArtboard();

        target.item().activeView.zoom = sourceZoom;
        target.item().activeView.centerPoint = sourceCenterPoint;

        // Copy properties from the source artboard to the target artboard
        targetBoard.name(artboardName);
        targetBoard.item().artboardRect = artboardRect;
        targetBoard.item().rulerOrigin = artboardRulerOrigin;

        // Set the zoom again
        target.item().views[0].zoom = sourceZoom
        target.item().views[0].centerPoint = sourceCenterPoint;

        return targetBoard;
    }

    static defaultRect(): Rect { return [0, 1000, 1000, 0]; }
}

function q(obj: any): any {
    if (obj === null || obj === undefined) return undefined;
    if (obj.q) return obj.q;
    if (obj.typename === undefined) return undefined;

    // Switch based on the typename and create the appropriate query instance
    switch (obj.typename) {
        case "Document":
            obj.q = new DocumentQuery(obj as Document);
            break;
        case "Layer":
            obj.q = new LayerQuery(obj as Layer);
            break;
        case "PathItem":
            obj.q = new PathItemQuery(obj as PathItem);
            break;
        case "TextFrame":
            obj.q = new TextFrameQuery(obj as TextFrame);
            break;
        case "Story":
            obj.q = new StoryQuery(obj as Story);
            break;
        case "Gradient":
            obj.q = new GradientQuery(obj as Gradient);
            break;
        case "Artboard":
            obj.q = new ArtboardQuery(obj as Artboard);
            break;
        case "PageItem":
            obj.q = new PageItemQuery(obj as PageItem);
            break;
        case "CompoundPathItem":
            obj.q = new CompoundPathItemQuery(obj as CompoundPathItem);
            break;
        case "RasterItem":
            obj.q = new RasterItemQuery(obj as RasterItem);
            break;
        case "PlacedItem":
            obj.q = new PlacedItemQuery(obj as PlacedItem);
            break;
        case "GroupItem":
            obj.q = new GroupItemQuery(obj as GroupItem);
            break;
        case "SymbolItem":
            obj.q = new SymbolItemQuery(obj as SymbolItem);
            break;
        case "MeshItem":
            obj.q = new MeshItemQuery(obj as MeshItem);
            break;
        case "PluginItem":
            obj.q = new PluginItemQuery(obj as PluginItem);
            break;
        case "GraphItem":
            obj.q = new GraphItemQuery(obj as GraphItem);
            break;
        case "Swatch":
            obj.q = new SwatchQuery(obj as Swatch);
            break;
        case "Brush":
            obj.q = new BrushQuery(obj as Brush);
            break;
        case "Tag":
            obj.q = new TagQuery(obj as Tag);
            break;
        case "Symbol":
            obj.q = new SymbolQuery(obj as Symbol);
            break;
        case "TextRange":
            obj.q = new TextRangeQuery(obj as TextRange);
            break;
        case "InsertionPoint":
            obj.q = new InsertionPointQuery(obj as InsertionPoint);
            break;
        case "CharacterStyle":
            obj.q = new CharacterStyleQuery(obj as CharacterStyle);
            break;
        case "ParagraphStyle":
            obj.q = new ParagraphStyleQuery(obj as ParagraphStyle);
            break;
        case "Pattern":

    return obj.q;

class DocumentQueryTests extends AppTestClass {
    constructor() {
        super();
        this.decorateExceptions = true;
    }

    _testDocumentDelete() {
        // Test deleting non-existent document
        const nonExistentFile = "NonExistent.ai";
        api.deleteDocument(nonExistentFile); // Should not throw

        // Test deleting existing file
        const testFile = "TestDelete.ai";
        api.createDocument(testFile);

        const expectedExistsBefore = true;
        const actualExistsBefore = api.exists(testFile);
        if (actualExistsBefore !== expectedExistsBefore)
            throw `Before delete: expected exists=${expectedExistsBefore}, got ${actualExistsBefore}`;

        api.deleteDocument(testFile);

        const expectedExistsAfter = false;
        const actualExistsAfter = api.exists(testFile);
        if (actualExistsAfter !== expectedExistsAfter)
            throw `After delete: expected exists=${expectedExistsAfter}, got ${actualExistsAfter}`;

        // Test deleting open document
        const openFile = "TestDeleteOpen.ai";
        const openDoc = api.createDocument(openFile);
        openDoc.save();

        api.deleteDocument(openFile);

        const expectedExistsAfterOpen = false;
        const actualExistsAfterOpen = api.exists(openFile);
        if (actualExistsAfterOpen !== expectedExistsAfterOpen)
            throw `After delete open: expected exists=${expectedExistsAfterOpen}, got ${actualExistsAfterOpen}`;
    }

    _testDocumentCreate() {
        const testFile = "TestCreate.ai";
        api.deleteDocument(testFile);  // Cleanup any existing file

        // Test basic creation
        const doc = api.createDocument(testFile);
        const expectedName = testFile;
        const actualName = doc.name();
        if (actualName !== expectedName)
            throw `Document name mismatch: expected '${expectedName}', got '${actualName}'`;

        // Test creation sets active document
        const expectedActiveDoc = doc._item;
        const actualActiveDoc = app.activeDocument;
        if (actualActiveDoc !== expectedActiveDoc)
            throw `Active document mismatch: expected ${expectedActiveDoc.q.name()}, got ${actualActiveDoc.q.name()}`;

        // Test creation with activation false
        const inactiveFile = "TestCreateInactive.ai";
        api.deleteDocument(inactiveFile);  // Cleanup

        api.createDocument(inactiveFile, false);
        const expectedActive = doc._item;
        const actualActive = app.activeDocument;
        if (actualActive !== expectedActive)
            throw `Active document after inactive creation: expected ${expectedActive.q.name()}, got ${actualActive.q.name()}`;

        // Cleanup
        api.deleteDocument(testFile);
        api.deleteDocument(inactiveFile);
    }

    _testDocumentOpen() {
        const testFile = "TestOpen.ai";
        api.deleteDocument(testFile);  // Cleanup

        // Create and save a test document
        const originalDoc = api.createDocument(testFile);
        originalDoc.save();

        // Test basic opening
        const openedDoc = api.openDocument(testFile);
        const expectedName = testFile;
        const actualName = openedDoc.name();
        if (actualName !== expectedName)
            throw `Opened document name: expected '${expectedName}', got '${actualName}'`;

        // Test opening with creation disabled
        const nonExistentFile = "NonExistent.ai";
        api.deleteDocument(nonExistentFile);  // Cleanup

        this.assertThrows(
            () => api.openDocument(nonExistentFile, false),
            `Opening non-existent file '${nonExistentFile}' should throw`
        );

        // Test opening with creation enabled
        const newFile = "CreateNew.ai";
        api.deleteDocument(newFile);  // Cleanup

        const createdDoc = api.openDocument(newFile, true);
        const expectedNewName = newFile;
        const actualNewName = createdDoc.name();
        if (actualNewName !== expectedNewName)
            throw `Created document name: expected '${expectedNewName}', got '${actualNewName}'`;

        // Cleanup
        api.deleteDocument(testFile);
        api.deleteDocument(newFile);
        api.deleteDocument(nonExistentFile);
    }
}

class ArtboardQueryTests extends AppTestClass {
    testDocument() {
        const doc = api.activeDocument();
        const artboard = doc.artboards().at(0);
        
        // Test document access
        const artboardDoc = artboard.document();
        const expectedDocName = doc.name();
        const actualDocName = artboardDoc.name();
        if (actualDocName !== expectedDocName)
            throw `Document name: expected ${expectedDocName}, got ${actualDocName}`;
        
        // Verify it's configured for artboards
        const expectedProperty = "artboards";
        const actualProperty = artboardDoc.collectionProperty;
        if (actualProperty !== expectedProperty)
            throw `Collection property: expected ${expectedProperty}, got ${actualProperty}`;
    }
 
    testIndex() {
        const doc = api.activeDocument();
        const firstArtboard = doc.artboards().at(0);
        
        // Test index of first artboard
        const expectedFirstIndex = 0;
        const actualFirstIndex = firstArtboard.index();
        if (actualFirstIndex !== expectedFirstIndex)
            throw `First artboard index: expected ${expectedFirstIndex}, got ${actualFirstIndex}`;
 
        // Create second artboard and test its index
        const secondArtboard = doc.artboards().create("Second");
        const expectedSecondIndex = 1;
        const actualSecondIndex = secondArtboard.index();
        if (actualSecondIndex !== expectedSecondIndex)
            throw `Second artboard index: expected ${expectedSecondIndex}, got ${actualSecondIndex}`;
    }
 
    testActivation() {
        const doc = api.activeDocument();
        const firstArtboard = doc.artboards().at(0);
        const secondArtboard = doc.artboards().create("Second");
        
        // Test activation
        secondArtboard.activate();
        if (!secondArtboard.activated())
            throw `Second artboard should be activated`;
        if (firstArtboard.activated())
            throw `First artboard should not be activated`;
 
        // Test actual active artboard index
        const expectedIndex = 1;
        const actualIndex = doc._item.artboards.getActiveArtboardIndex();
        if (actualIndex !== expectedIndex)
            throw `Active artboard index: expected ${expectedIndex}, got ${actualIndex}`;
 
        // Test inactivation (should activate first artboard)
        secondArtboard.inactivate();
        if (secondArtboard.activated())
            throw `Second artboard should not be activated`;
        if (!firstArtboard.activated())
            throw `First artboard should be activated`;
 
        // Verify first artboard is active
        const expectedFinalIndex = 0;
        const actualFinalIndex = doc._item.artboards.getActiveArtboardIndex();
        if (actualFinalIndex !== expectedFinalIndex)
            throw `Final active index: expected ${expectedFinalIndex}, got ${actualFinalIndex}`;
    }
 
    testCreate() {
        const doc = api.activeDocument();
        
        // Test creation with default rect
        const newArtboard = doc.artboards().create();
        const expectedRect = ArtboardQuery.defaultRect();
        const actualRect = newArtboard._item.artboardRect;
        for (let i = 0; i < 4; i++) {
            if (actualRect[i] !== expectedRect[i])
                throw `Artboard rect[${i}]: expected ${expectedRect[i]}, got ${actualRect[i]}`;
        }
 
        // Test creation with name
        const namedArtboard = doc.artboards().create("Named");
        const expectedName = "Named";
        const actualName = namedArtboard.name();
        if (actualName !== expectedName)
            throw `Created artboard name: expected ${expectedName}, got ${actualName}`;
 
        // Verify artboard count
        const expectedCount = 3; // Original + 2 new ones
        const actualCount = doc.artboards().length();
        if (actualCount !== expectedCount)
            throw `Final artboard count: expected ${expectedCount}, got ${actualCount}`;
    }
