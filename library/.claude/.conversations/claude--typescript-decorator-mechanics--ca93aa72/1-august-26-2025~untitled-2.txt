"use client"
import React, { ReactNode, ReactElement, useState, useRef, useEffect } from 'react';
import stringify from 'fast-safe-stringify';

// Core framework types
type Constructor<T = {}> = new (...args: any[]) => T;
type Type = Constructor;

type Properties<T> = {
    [K in keyof T as K extends `$${infer Rest}` ? Rest : never]: T[K]
} & {
    children?: ReactNode;
};

// Symbols on $Chemical
const inertProperties = Symbol('inertProperties');
const dynamicProperties = Symbol('comparisons');
const updateFunctions = Symbol('updateFunctions');

export class $Chemical {
    private _parent?: $Chemical;
    get parent(): $Chemical | undefined { return this._parent; }

    @inert()
    get Component(): React.FC<Properties<this>> { return this._component!; }
    private _component?: React.FC<Properties<this>>;

    @inert()
    get reactive(): boolean { return this._reactive; } 
    private _reactive: boolean = true;

    @inert()
    public children?: ReactNode;

    constructor() {
        const type = this.getType();
        if (!type._formula) {
            type._formula = this;
            this._component = this.createComponent();
        }
    }

    // Override this to control rendering
    view(): ReactNode {
        // Default implementation shows the existing children from the props
        return this.children;
    }

    reactivate() {
        this._reactive = true;
        // Make the properties reactive
    }

    deactivate() {
        this._reactive = false;
    }

    update(): void {
        if (!this._reactive) return;
        const updates = (this as any)[updateFunctions];
        if (Array.isArray(updates))
            updates.map(update => update());
    }

    getType<T extends Type = typeof $Chemical>(): T {
        return this.constructor as any;
    }

    private createComponent(): React.FC<Properties<this>> {
        const Component: React.FC<Properties<this>> = (props: any) => {
            const instance = Object.assign(
                Object.create(Object.getPrototypeOf(this.getType()._formula)), 
                this.getType()._formula
            );

            instance[updateFunctions] = [];
            if (props.children) 
                instance.children = props.children;
            for (const key in props)
                if (key !== 'children' && key !== 'key' && key !== 'ref') 
                    instance['$' + key] = props[key];
            
            // Set up this component's update function
            const [, update] = useState({});
            instance[updateFunctions].push(() => update({}));

            // Clean up on unmount
            useEffect(() => {
                return () => {
                    instance[updateFunctions] = [];
                };
            }, []);

            return instance.view();
        }
        return Component;
    }

    static get formula() { return this._formula; }
    static _formula: $Chemical;
}

// Decorator to mark a field as inert (non-reactive)
export function inert() {
    return function (target: any, property: string) {
        if (!target[inertProperties]) 
            target[inertProperties] = new Set<string>();
        target[inertProperties].add(property);
    };
}

// Decorator for properties that should use deep equality comparison
export function dynamic(comparer?: (a: any, b: any) => boolean) {
   return function (target: any, propertyKey: string) {
       if (!target[dynamicProperties])
           target[dynamicProperties] = new Map<string, (a: any, b: any) => boolean>();

       const compareFn = comparer || ((a: any, b: any) => represent(a) === represent(b));
       target[dynamicProperties].set(propertyKey, compareFn);
   };
}

// A decorator for expressing configurable binding
export function where(type: Type) {
    type = resolveType(type);
    return function (target: $Chemical, property: string) {
    };
}

// A decorator for reassigning the parent property
export function parent(type: Type) {
    type = resolveType(type);
    return function (target: $Chemical, property: string) {
        Object.defineProperty(target, property, {
            get: function(this: $Chemical) {
                const p = this.parent;
                // Optional: runtime type check
                if (p && !(p instanceof type)) {
                    throw new Error(`Expected parent to be ${type.name}, got ${p.constructor.name}`);
                }
                return p;
            },
            enumerable: true,
            configurable: true
        });
        inert()(target, property);
    };
}


function resolveType(type: Type | (() => Type)): Type {
    // Check if it's a function
    if (typeof type === 'function') {
        // Try to determine if it's a constructor or a regular function
        // Constructors typically have a prototype property with a constructor
        if (type.prototype && type.prototype.constructor === type) {
            // It's a constructor
            return type as Type;
        } else {
            // It's a function that returns a constructor
            return (type as () => Type) ();
        }
    }
    
    throw new Error('Input must be a constructor or a function returning a constructor');
}

function represent(value: any): string {
    return stringify(value, function (this: any, property: string, value: any): any {
        if (property === '') return value;
        if (this instanceof $Chemical && !isReactiveProperty(this, property, value))
            return undefined;
        
        // Only keep the Proxy check if Next.js proxies cause issues
        // Otherwise, fast-safe-stringify handles most edge cases
        if (value?.constructor?.name === 'Proxy') return '[NextProxy]';
        return value;
    });
}

function isReactiveProperty(instance: any, key: string, value: any): boolean {
   // Skip symbols, functions, underscore properties
   if (typeof key === 'symbol' || 
       typeof value === 'function' || 
       key.startsWith('_')) 
       return false;
   
   // Skip if marked as @inert
   const proto = Object.getPrototypeOf(instance);
   if (proto?.[inertProperties]?.has(key)) return false;
   
   return true;
}
