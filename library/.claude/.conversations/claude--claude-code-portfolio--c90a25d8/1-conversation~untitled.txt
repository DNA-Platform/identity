// Pop Art (vintage halftone) proof-of-concept — Argentina.
// Uses the app's silhouette standard: countries-110m + geoMercator fit to the
// hero's main cluster. Halftone = SVG <pattern> per zone (fast + crisp export).
const fs = require('fs');
const puppeteer = require('puppeteer');
const { geoMercator, geoPath, geoArea, geoBounds } = require('d3-geo');
const { feature, merge } = require('topojson-client');
const ROOT = 'c:/Source/dna-platform/dna-atlas';
const OUT = `${ROOT}/.claude/perspective`;
const W = 980, H = 500, PAD = 36;
const HERO_ID = '32'; // Argentina

const topo = JSON.parse(fs.readFileSync(`${ROOT}/public/countries-110m.json`, 'utf8'));
const geoms = topo.objects.countries.geometries.filter((g) => String(g.id) !== '010'); // drop Antarctica

function keepMainCluster(geo) {
  if (geo.type !== 'MultiPolygon') return geo;
  const polys = geo.coordinates.map((c) => ({ type: 'Polygon', coordinates: c }));
  const areas = polys.map((p) => geoArea(p));
  let mi = 0; for (let i = 1; i < areas.length; i++) if (areas[i] > areas[mi]) mi = i;
  const [[w, s], [e, n]] = geoBounds(polys[mi]);
  const padX = (e - w) * 0.4 + 2, padY = (n - s) * 0.4 + 2;
  const box = [w - padX, s - padY, e + padX, n + padY];
  const kept = geo.coordinates.filter((_, i) => { const b = geoBounds(polys[i]); const cx = (b[0][0] + b[1][0]) / 2, cy = (b[0][1] + b[1][1]) / 2; return cx >= box[0] && cx <= box[2] && cy >= box[1] && cy <= box[3]; });
  return { type: 'MultiPolygon', coordinates: kept.length ? kept : geo.coordinates };
}

const heroGeom = geoms.find((g) => String(g.id) === HERO_ID);
const heroMain = keepMainCluster(feature(topo, heroGeom).geometry);
const proj = geoMercator().fitExtent([[PAD, PAD], [W - PAD, H - PAD]], heroMain);
const path = geoPath(proj);
const allLand = merge(topo, geoms);
const allLandD = path(allLand);
const heroD = path(heroMain);

function mul(a) { return function () { a |= 0; a = (a + 0x6d2b79f5) | 0; let t = Math.imul(a ^ (a >>> 15), 1 | a); t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t; return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; }
// hand-drawn wobble: project each vertex + seeded jitter, build path manually
function jitterD(geom, amp, rng) {
  const polys = geom.type === 'Polygon' ? [geom.coordinates] : geom.coordinates;
  let d = '';
  for (const poly of polys) for (const ring of poly) {
    let started = false;
    for (const c of ring) { const p = proj(c); if (!p) continue; const x = p[0] + (rng() - 0.5) * amp * 2, y = p[1] + (rng() - 0.5) * amp * 2; d += (started ? 'L' : 'M') + x.toFixed(1) + ' ' + y.toFixed(1) + ' '; started = true; }
    d += 'Z ';
  }
  return d;
}

const seed = 7;
const heroOutline = jitterD(heroMain, 0.8, mul(seed));
const coastOutline = jitterD(allLand, 0.6, mul(seed + 99));

let borders = '';
for (const g of geoms) { if (String(g.id) === HERO_ID) continue; const d = path(feature(topo, g)); if (!d) continue; borders += `<path d="${d}" fill="none" stroke="#20140f" stroke-width="0.9" stroke-linejoin="round"/>`; }

// ── palette (from the described reference) ──
const paper = '#f4ecd6', water = '#3f7fc4', neigh = '#ec5f97', hero = '#cdd93a', ink = '#20140f';
const S = 7, R = 2.35; // dot spacing + radius
const pat = (id, color, angle, dx, dy) => `<pattern id="${id}" width="${S}" height="${S}" patternUnits="userSpaceOnUse" patternTransform="translate(${dx} ${dy}) rotate(${angle})"><circle cx="${S / 2}" cy="${S / 2}" r="${R}" fill="${color}"/></pattern>`;

// vintage grain: sparse tiny specks slightly off-paper (crisp, no opacity/blur)
const grng = mul(seed + 200); let grain = ''; const N = Math.round(W * H * 0.0006);
for (let i = 0; i < N; i++) { const x = grng() * W, y = grng() * H; grain += `<circle cx="${x.toFixed(1)}" cy="${y.toFixed(1)}" r="${(0.4 + grng() * 0.5).toFixed(2)}" fill="${grng() < 0.5 ? '#d9ceb2' : '#fffبf0'.replace('ب', 'a')}"/>`; }

const svg = `<svg xmlns="http://www.w3.org/2000/svg" width="${W}" height="${H}" viewBox="0 0 ${W} ${H}">
<defs>${pat('htw', water, 15, 0, 0)}${pat('htn', neigh, 75, 1.4, -0.8)}${pat('hth', hero, 0, -0.9, 1.1)}</defs>
<rect width="${W}" height="${H}" fill="${paper}"/>
<rect width="${W}" height="${H}" fill="url(#htw)"/>
<path d="${allLandD}" fill="${paper}"/>
<path d="${allLandD}" fill="url(#htn)"/>
<path d="${heroD}" fill="${paper}"/>
<path d="${heroD}" fill="url(#hth)"/>
<g>${grain}</g>
${borders}
<path d="${coastOutline}" fill="none" stroke="${ink}" stroke-width="1.5" stroke-linejoin="round" stroke-linecap="round"/>
<path d="${heroOutline}" fill="none" stroke="${ink}" stroke-width="2.8" stroke-linejoin="round" stroke-linecap="round"/>
</svg>`;

fs.writeFileSync(`${OUT}/popart-argentina.svg`, svg);

(async () => {
  const br = await puppeteer.launch({ headless: 'new', args: ['--no-sandbox'] });
  const p = await br.newPage(); await p.setViewport({ width: W, height: H, deviceScaleFactor: 2 });
  await p.setContent(`<!doctype html><body style="margin:0">${svg}</body>`, { waitUntil: 'networkidle0' });
  await p.screenshot({ path: `${OUT}/popart-argentina.png` });
  // zoom crop to show dot crispness
  await p.setViewport({ width: 420, height: 300, deviceScaleFactor: 3 });
  await p.evaluate(() => { document.querySelector('svg').setAttribute('viewBox', '380 120 220 160'); document.querySelector('svg').setAttribute('width', 420); document.querySelector('svg').setAttribute('height', 300); });
  await p.screenshot({ path: `${OUT}/popart-argentina-zoom.png` });
  await br.close(); console.log('DONE');
})().catch((e) => { console.error(e); process.exit(1); });
