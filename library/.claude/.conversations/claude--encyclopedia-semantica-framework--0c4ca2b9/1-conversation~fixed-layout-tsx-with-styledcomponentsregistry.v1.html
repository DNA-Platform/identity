app/layout.tsx

import './globals.css';
import StyledComponentsRegistry from './registry';

export default function RootLayout({
  children,
}: {
  children: React.ReactNode;
}) {
  return (
    <html lang="en">
      <body>
        <StyledComponentsRegistry>
          {children}
        </StyledComponentsRegistry>
      </body>
    </html>
  );
}

File Structure Required:

app/
  ├── layout.tsx
  ├── registry.tsx   <-- This file must be here
  ├── page.tsx
  └── globals.css

Note:

Make sure the registry.tsx file is in the same app directory as layout.tsx. The import path './registry' expects it to be a sibling file.
