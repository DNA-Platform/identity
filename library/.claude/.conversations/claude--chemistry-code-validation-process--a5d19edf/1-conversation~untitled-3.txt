export class $ValidationTestDisplay extends $Chemical {
    results: { test: TestCase; passed: boolean; error?: string }[] = [];
    loading = true;
    initialized = false;
    
    view() {
        // Initialize on first render
        if (!this.initialized) {
            this.initialized = true;
            setTimeout(() => {
                this.results = validationTestCases.map(testCase => ({
                    test: testCase,
                    ...TestRunner.runTest(testCase)
                }));
                this.loading = false;
            }, 100);
        }
        
        if (this.loading) {
            return (
                <div style={{ padding: '40px', textAlign: 'center' }}>
                    <div>Running validation tests...</div>
                </div>
            );
        }
        
        return (
            <div style={{ padding: '40px', fontFamily: 'system-ui' }}>
                <h1>Validation Error Message Tests</h1>
                <p style={{ color: '#666', marginBottom: '30px' }}>
                    Testing $check validation with intentionally wrong arguments to verify error messages
                </p>
                
                {/* Expected signature display */}
                <div style={{ 
                    marginBottom: '30px', 
                    padding: '20px', 
                    background: '#e8f5e9',
                    borderRadius: '8px',
                    border: '1px solid #4caf50'
                }}>
                    <h3>Expected $ComplexValidator Signature:</h3>
                    <pre style={{ fontFamily: 'monospace', fontSize: '13px' }}>
{`$ComplexValidator(
    header: $Label,
    count: Number,
    items: $Chapter[],
    config: Object | undefined,
    callback: Function | undefined,
    matrix: Number[][]
)`}
                    </pre>
                </div>
                
                {/* Test results */}
                {this.results.map((result, i) => (
                    <div key={i} style={{ 
                        marginBottom: '25px', 
                        border: result.passed ? '2px solid #4caf50' : '2px solid #f44336',
                        borderRadius: '8px',
                        overflow: 'hidden'
                    }}>
                        {/* Test header */}
                        <div style={{ 
                            padding: '15px',
                            background: result.passed ? '#e8f5e9' : '#ffebee',
                            borderBottom: '1px solid #ddd'
                        }}>
                            <div style={{ 
                                display: 'flex', 
                                alignItems: 'center', 
                                gap: '10px',
                                marginBottom: '5px'
                            }}>
                                <span style={{ 
                                    fontSize: '20px',
                                    color: result.passed ? '#4caf50' : '#f44336'
                                }}>
                                    {result.passed ? '✓' : '✗'}
                                </span>
                                <strong>{result.test.name}</strong>
                            </div>
                            <div style={{ color: '#666', fontSize: '14px', marginLeft: '30px' }}>
                                {result.test.description}
                            </div>
                        </div>
                        
                        {/* Error message */}
                        <div style={{ padding: '15px' }}>
                            <div style={{ marginBottom: '5px', fontWeight: 'bold', color: '#333' }}>
                                Error Message:
                            </div>
                            <pre style={{ 
                                background: '#f5f5f5',
                                padding: '15px',
                                borderRadius: '4px',
                                fontSize: '12px',
                                fontFamily: 'monospace',
                                whiteSpace: 'pre-wrap',
                                wordWrap: 'break-word',
                                margin: 0,
                                border: '1px solid #e0e0e0'
                            }}>
                                {result.error || 'No error message'}
                            </pre>
                        </div>
                    </div>
                ))}
                
                {/* Summary */}
                <div style={{ 
                    marginTop: '40px',
                    padding: '20px',
                    background: '#f0f8ff',
                    borderRadius: '8px',
                    border: '1px solid #2196f3'
                }}>
                    <h3>Summary</h3>
                    <p>
                        All {this.results.length} tests should show validation errors with clear messages 
                        indicating what was expected vs what was received.
                    </p>
                    <p>
                        Tests passed: {this.results.filter(r => r.passed).length}/{this.results.length}
                        {' '}(passed = error was correctly thrown)
                    </p>
                </div>
            </div>
        );
    }
}
