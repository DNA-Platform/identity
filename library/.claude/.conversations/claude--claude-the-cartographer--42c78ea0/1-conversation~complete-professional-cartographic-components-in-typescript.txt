// 6. Topographic Contour Lines Generator
// Creates professional contour lines based on elevation data
// ============================================================================
interface TopographicContoursProps {
  geoData: GeoJSON.FeatureCollection;
  elevationAccessor: (feature: GeoJSON.Feature) => number;
  contourLevels: number[];
  contourColors?: string[];
  strokeWidth?: number;
  projection: (coordinates: [number, number]) => [number, number];
}

export const TopographicContours: React.FC<TopographicContoursProps> = ({ 
  geoData, 
  elevationAccessor, 
  contourLevels, 
  contourColors = ['#94765A', '#A68564', '#B8946E', '#CAA278', '#DCB082'], 
  strokeWidth = 0.5,
  projection
}) => {
  // For a real implementation, you would use d3-contour to generate contour paths
  // This is a simplified version that draws contour lines at specific elevations
  
  const createContourLines = (): React.ReactNode[] => {
    const contours: React.ReactNode[] = [];
    
    contourLevels.forEach((level, index) => {
      const color = contourColors[index % contourColors.length];
      
      // Filter features at this elevation level
      const featuresAtLevel = geoData.features.filter(feature => {
        const elevation = elevationAccessor(feature);
        return Math.abs(elevation - level) < 10; // Tolerance for matching elevation
      });
      
      if (featuresAtLevel.length > 0) {
        // Create a path for this contour level
        const contourPath = featuresAtLevel.map(feature => {
          // This is a simplification - in reality you would use d3.geoPath()
          // with the projection to generate the SVG path string
          if (feature.geometry.type === 'LineString') {
            const coordinates = (feature.geometry as GeoJSON.LineString).coordinates;
            const pathData = coordinates
              .map((coord, i) => {
                const [x, y] = projection(coord as [number, number]);
                return `${i === 0 ? 'M' : 'L'} ${x},${y}`;
              })
              .join(' ');
            return pathData;
          }
          return '';
        }).join(' ');
        
        contours.push(
          <path
            key={`contour-${level}`}
            d={contourPath}
            fill="none"
            stroke={color}
            strokeWidth={strokeWidth}
            strokeLinejoin="round"
            strokeLinecap="round"
          />
        );
      }
    });
    
    return contours;
  };
  
  return <g className="topographic-contours">{createContourLines()}</g>;
};
