class Vector extends A<Position> {
    private _type: VType;
    type(): VType { return this._type; }
    x(): Vector { return new Vector(this.xa(), "x"); }
    xa(): A<number> { return (this.select(x => x[0])); }
    y(): Vector { return new Vector(this.ya(), "x"); }
    ya(): A<number> { return this.select(x => x[1]); }
    v(): Vector {
        if (this.dim() == 2)
            throw "The numeric values of a Vector are only defined for 1D vectors";
        return this.get(this.axis()).x();
    }
    va(): A<number> { return this.v().xa(); }
    p(): Vector { return this; }
    pa(): A<Position> { return f(this.a); }
    dim(): number { return (this.type() == "xy" || this.type() == "c") ? 2 : 1; }
    axis(): VType {
        if (this.dim() == 2) return "x";
        return this.type() == "y" ? "y" : "x";
    }
    axisIndex(): number { return this.axis() == "x" ? 0 : 1; }

    constructor(x: A<number> | A<Position>, type: VType) {
        let arg: Position[] = [];
        if (!VectorTypes.isArrayExtension(x)) throw "The Vector type only accepts array extensions A<T> as input";
        if (VectorTypes.isPositionArrayExtension(x)) arg = x.a;
        else if (VectorTypes.isNumberArrayExtension(x)) {
            var z: A<number> = f([]);
            for (var i = 0; i < x.a.length; i++)
                z.push(0);
            if (type == "x" || type == "b") arg = x.zip(z).a;
            else arg = z.zip(x).a;
        }
        super(arg);
        this._type = type;
    }

    att(index: number, type: "x" | "y" = "x"): number {
        index = evaluateNegativeIndex(index, this.a.length);
        var axis = this.dim() == 1 ? this.axisIndex() : (type == "x" ? 0 : 1);
        return this.a[index][axis];
    }

    get(selector: string | Vector | VType | undefined): Vector {
        if (selector === undefined) return this;
        if (VectorTypes.isVType(selector)) {
            var axis = this.axis();
            if (axis === undefined) return this;
            if (axis === "y") return this.y();
            return this.x();
        }
        if (typeof selector === "string")
            return v(super.get(selector), this.type());
        if (selector.length() !== this.length())
            throw "Boolean vectors must be the same length when used as a selector";
        var result: A<Position> = f([]);
        for (var i = 0; i < this.length(); i++) {
            if (selector.a[i][0] === 1)
                result.push(this.at(i));
        }
        return v(result, this.type());
    }

    clone(): Vector {
        return v(f(this.a).select(p => [p[0], p[1]]), this.type());
    }

    setX(x: Vector | A<number>, type: VType = "xy") {
        if (x instanceof A) x = new Vector(x, "x");
        return x.join(this.y(), type);
    }

    setY(y: Vector | A<number>, type: VType = "xy") {
        if (y instanceof A) y = new Vector(y, "x");
        return this.x().join(y, type);
    }

    real(): Vector {
        this._assertComplex();
        return this.x();
    }

    imag(): Vector {
        this._assertComplex();
        return this.y();
    }

    toBoolean(): Vector {
        if (this.dim() !== 1) throw new Error("Only 1D vectors can be made boolean");
        return v(this.va().select(x => x > 0 ? 1 : 0), "b");
    }

    toComplex(): Vector {
        return v(this, "c");
    }

    toPosition(): Vector {
        return v(this, "xy");
    }

    normDim(): Vector {
        if (this.type() == "xy" || this.type() == "c") return new Vector(this, "xy");
        if (this.type() == "y") return new Vector(this.y(), "x");
        return new Vector(this.x(), "x");
    }

    flipDim(): Vector {
        var newType = this.type();
        if (newType == "x" || newType == "b") newType = "y";
        else if (newType == "y") newType = "x";
        return v(f(this.a).select(p => [p[1], p[0]]), newType);
    }

    aggregate(op: (acc: Position, value: Position) => Position, seed?: Position): Position {
        if (seed === undefined) seed = [0, 0];
        let result = seed;
        for (let i = 0; i < this.length(); i++) {
            result = op(result, this.at(i));
        }
        return result;
    }

    aggregate1(op: (acc: number, value: number) => number, seed?: number): number {
        if (this.dim() !== 1) throw new Error("The aggregate1 function only works on 1D vectors")
        if (seed === undefined) seed = 0;
        let result = seed;
        var values = this.va();
        for (let i = 0; i < this.length(); i++) {
            result = op(result, values.at(i));
        }
        return result;
    }

    coerce(other: Vector): Vector {
        // Try to make other match this.type()
        if (other.type() === this.type()) return other;
        if (this.dim() == 1 && other.dim() == 2)
            throw "Cannot coerce a 2D vector into a 1D vector"
        if (this.type() == "x" && other.type() == "y")
            return other.flipDim();
        if (this.type() == "y" && other.type() == "x")
            return other.flipDim();
        if (this.type() == "b" && other.type() == "y")
            return other.flipDim().toBoolean();
        if (this.type() == "b" && other.type() == "x")
            return other.toBoolean();
        if (this.dim() == 1 && other.type() === "y")
            return other.flipDim();
        if (this.type() == "x" && other.type() == "b")
            return new Vector(other.xa(), "x");
        if (this.type() == "xy" && other.dim() == 2)
            return new Vector(other, "xy");
        if (this.type() == "c" && other.type() !== "c")
            return new Vector(other, "c");

        throw `Cannot coerce vector of type ${other.type()} to ${this.type()}`;
    }

    pointwise2to2(other: Vector, op: (a: Position, b: Position) => Position): Vector {
        const coerced = this.coerce(other);
        if (this.length() !== coerced.length()) throw "Vectors must be same length for pointwise operations";
        return new Vector(this.zip(coerced).select(([a, b]) => op(a, b)), this.type());
    }

    pointwise1to1(other: Vector, op: (a: number, b: number) => number): Vector {
        if (this.dim() == 2 || other.dim() == 2) throw "Cannot perform 1-to-1 operations on 2D vectors";
        if (this.length() !== other.length()) throw "Pointwise operations require vectors of the same length";
        other = this.coerce(other);

        var x1 = this.type() == "y" ? this.ya() : this.xa();
        var x2 = other.type() == "y" ? other.ya() : other.xa();
        var type: VType = this.type() == "y" ? "y" : "x";
        return new Vector(x1.zip(x2).select(([x, y]) => op(x, y)), type);
    }

    pointwise2to1(other: Vector, op: (a: Position, b: Position) => number): Vector {
        if (this.length() !== other.length()) throw "Vectors must be same length for pointwise operations";
        other = this.coerce(other);
        return new Vector(this.zip(other).select(([a, b]) => op(a, b)), this.type());
    }

    apply(f: (x: Position) => Position): Vector {
        if (this.dim() !== 2)
            throw "The apply function is meant for 2D vectors";
        return new Vector(this.select(f), this.type());
    }

    apply1(f: (x: number) => number, dim: "x" | "y" = "x"): Vector {
        if (this.type() == "x")
            return new Vector(this.xa().select(f), "x");
        if (this.type() == "y")
            return new Vector(this.ya().select(f), "y");
        if (this.type() == "b")
            return new Vector(this.xa().select(f), "x");
        if (dim === "x")
            return v(this.xa().select(f)).join(this.ya(), this.type());
        if (dim === "y")
            return this.x().join(this.ya().select(f), this.type());
        throw "Debug missing condition";
    }

    addAxis(rng: Range | number = [0, 1]): Vector {
        if (this.dim() > 1) return this;
        if (!rng) rng = [0, 1];
        if (VectorTypes.isNumber(rng)) rng = [0, rng];

        var axis = Vector.linspace(this.length(), rng);
        if (this.type() == "y") return this.setX(axis);
        return this.setY(axis);
    }

    join(y: A<number> | Vector, type: VType = "xy"): Vector {
        if (type != "xy" && type != "c")
            throw "The result of a join must be two dimensional";
        if ((this.dim() !== 1) || (y instanceof Vector && y.dim() !== 1))
            throw new Error("Only one-dimensional things can be joined");
        if (this.length() != y.length())
            throw new Error("Vectors must be the same length");
        if (y instanceof A) y = new Vector(y as A<number>, "x");
        return new Vector(this.va().zip(y.va()), type);
    }

    norm(rng: Range | number): Vector;
    norm(xrng: Range | number, yrng: Range | number): Vector;
    norm(rng: Range | number, yrng?: Range | number): Vector {
        if (typeof rng === "number") rng = [0, rng];
        if (yrng === undefined) yrng = rng;
        if (typeof yrng === "number") yrng = [0, yrng];

        if (this.dim() === 1) {
            const [low, high] = rng;
            const min = this.gmin();
            const max = this.gmax();
            const scale = (high - low) / (max - min);
            const offset = low - min * scale;
            return this.multiply(v(scale)).add(v(offset));
        }

        // Handle two-dimensional case
        const xNorm = this.x().norm(rng);
        const yNorm = this.y().norm(yrng ?? rng);
        return xNorm.join(yNorm.va(), this.type());
    }

    /// Scle

    fit(region: Region): Vector {
        var xrng = [0, region.width()];
        var yrng = [0, region.height()];
        return this.norm(xrng, yrng);
    }

    gmax(type: "x" | "y" = "x"): number {
        if (this.type() == "x" || this.type() == "y") type = this.type() as any;
        if (this.type() == "b") type = "x";
        const values = type === "x" ? this.xa().a : this.ya().a;
        return Math.max.apply($.global, values);
    }

    gmin(type: "x" | "y" = "x"): number {
        if (this.type() == "x" || this.type() == "y") type = this.type() as any;
        if (this.type() == "b") type = "x";
        const values = type === "x" ? this.xa().a : this.ya().a;
        return Math.min.apply($.global, values);
    }

    sum(type?: "x" | "y"): number {
        var result = this.aggregate((acc, val) => [acc[0] + val[0], acc[1] + val[1]]);
        if (this.type() == "x" || this.type() == "b") return result[0];
        if (this.type() == "y") return result[1];
        if (type === undefined) type = "x";
        if (type == "x") return result[0];
        return result[1];
    }

    mean(type?: "x" | "y"): number {
        return this.sum(type) / this.length();
    }

    median(type?: "x" | "y"): number {
        var values = this.dim() == 1 ? this.va().a : type == "x" ? this.xa().a : this.ya().a;
        values.sort((a, b) => a - b);
        const mid = Math.floor(values.length / 2);

        return values.length % 2 === 0 ?
            (values[mid - 1] + values[mid]) / 2 :
            values[mid];
    }

    std(type?: "x" | "y"): number {
        var mn = this.mean(type);
        var vector: Vector;
        if (this.dim() == 1) vector = this.v();
        else if (type == "y") vector = this.y();
        else vector = this.x();
        var vs = vector.aggregate1((acc, v) => acc + (v - mn) ** 2);
        var variance = vs / this.length();
        return Math.sqrt(variance);
    }

    // Pointwise operations using pointwise
    add(x: Vector | Position | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        if (this.dim() == 1)
            return this.pointwise1to1(x, (x, y) => x + y);
        else
            return this.pointwise2to2(x, (x, y) => [x[0] + y[0], x[1] + y[1]]);
    }

    subtract(x: Vector | Position | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        if (this.dim() == 1)
            return this.pointwise1to1(x, (x, y) => x - y);
        else
            return this.pointwise2to2(x, (x, y) => [x[0] - y[0], x[1] - y[1]]);
    }

    multiply(x: Vector | Position | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        if (this.dim() == 1)
            return this.pointwise1to1(x, (x, y) => x * y);
        else if (this.type() == "xy")
            return this.pointwise2to2(x, (x, y) => [x[0] * y[0], x[1] * y[1]]);
        else {
            return this.pointwise2to2(x, ([re1, im1], [re2, im2]) => [
                re1 * re2 - im1 * im2, // Real part
                re1 * im2 + im1 * re2, // Imaginary part
            ]);
        }
    }

    divide(x: Vector | Position | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        if (this.dim() == 1) {
            // Scalar division (1D vectors)
            return this.pointwise1to1(x, (x, y) => x / y);
        } else if (this.type() == "xy") {
            // Element-wise division for 2D positional vectors
            return this.pointwise2to2(x, (x, y) => [x[0] / y[0], x[1] / y[1]]);
        } else {
            // Complex division
            return this.pointwise2to2(x, ([re1, im1], [re2, im2]) => {
                const denom = re2 * re2 + im2 * im2; // Magnitude squared of the divisor
                return [
                    (re1 * re2 + im1 * im2) / denom, // Real part
                    (im1 * re2 - re1 * im2) / denom, // Imaginary part
                ];
            });
        }
    }

    mod(base: number, dim?: "x" | "y"): Vector {
        // Override the type for single-dimensional cases
        if (this.dim() === 1) {
            if (this.type() === "b") dim = "x"; // Treat "b" like "x"
            else dim = this.type() as any;
        }

        // Handle specified dimension
        if (dim) {
            return this.apply1((v) => ((v % base) + base) % base, dim);
        }

        // Handle multi-dimensional cases without a specified dimension
        if (this.type() === "xy") {
            return this.x().mod(base).join(this.y().mod(base));
        }

        // this.type === "c"
        return this.apply(([re, im]) => {
            const modRe = ((re % base) + base) % base;
            const modIm = ((im % base) + base) % base;
            return [modRe, modIm];
        });
    }

    greaterThan(x: Vector | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.dim() !== 1 || x.dim() !== 1) throw new Error("Boolean operations only defined for one dimensional vectors");
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        if (this.dim() == 1) {
            return this.pointwise1to1(x, (x, y) => (x > y ? 1 : 0)).toBoolean();
        } else {
            return this.pointwise2to2(x, (x, y) => [x[0] > y[0] ? 1 : 0, x[1] > y[1] ? 1 : 0]).toBoolean();
        }
    }

    greaterThanOrEqualTo(x: Vector | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.dim() !== 1 || x.dim() !== 1) throw new Error("Boolean operations only defined for one dimensional vectors");
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        if (this.dim() == 1) {
            return this.pointwise1to1(x, (x, y) => (x >= y ? 1 : 0)).toBoolean();
        } else {
            return this.pointwise2to2(x, (x, y) => [x[0] >= y[0] ? 1 : 0, x[1] >= y[1] ? 1 : 0]).toBoolean();
        }
    }

    lessThan(x: Vector | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.dim() !== 1 || x.dim() !== 1) throw new Error("Boolean operations only defined for one dimensional vectors");
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        if (this.dim() == 1) {
            return this.pointwise1to1(x, (x, y) => (x < y ? 1 : 0)).toBoolean();
        } else {
            return this.pointwise2to2(x, (x, y) => [x[0] < y[0] ? 1 : 0, x[1] < y[1] ? 1 : 0]).toBoolean();
        }
    }

    lessThanOrEqualTo(x: Vector | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.dim() !== 1 || x.dim() !== 1) throw new Error("Boolean operations only defined for one dimensional vectors");
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        return this.pointwise1to1(x, (x, y) => (x <= y ? 1 : 0)).toBoolean();
    }

    equalTo(x: Vector | Position | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        if (this.dim() == 1) {
            return this.pointwise1to1(x, (x, y) => (x === y ? 1 : 0)).toBoolean();
        } else {
            return this.pointwise2to2(x, (x, y) => [(x[0] === y[0] && x[1] === y[1]) ? 1 : 0, 0]).toBoolean();
        }
    }

    pow(x: Vector | Position | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        if (this.dim() == 1)
            return this.pointwise1to1(x, (x, y) => Math.pow(x, y));
        else if (this.type() == "xy")
            return this.pointwise2to2(x, (x, y) => [Math.pow(x[0], y[0]), Math.pow(x[1], y[1])]);
        else {
            return this.pointwise2to2(x, ([re1, im1], [re2, im2]) => {
                const baseMagnitude = Math.sqrt(re1 * re1 + im1 * im1);
                const baseAngle = Math.atan2(im1, re1);

                const logRe = Math.log(baseMagnitude);
                const logIm = baseAngle;

                const powerRe = logRe * re2 - logIm * im2;
                const powerIm = logRe * im2 + logIm * re2;

                const resultMagnitude = Math.exp(powerRe);
                const resultAngle = powerIm;

                return [
                    resultMagnitude * Math.cos(resultAngle), // Real part
                    resultMagnitude * Math.sin(resultAngle)  // Imaginary part
                ];
            });
        }
    }

    exp(x: Vector | Position | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        if (this.dim() == 1)
            return this.pointwise1to1(x, (x, y) => Math.pow(Math.E, y));
        else if (this.type() == "xy")
            return this.pointwise2to2(x, (x, y) => [Math.exp(x[0]), Math.exp(x[1])]);
        else {
            return this.pointwise2to2(x, ([re, im]) => {
                const magnitude = Math.exp(re);
                return [
                    magnitude * Math.cos(im), // Real part
                    magnitude * Math.sin(im)  // Imaginary part
                ];
            });
        }
    }

    max(x: Vector | Position | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        x = this.coerce(x)
        if (this.dim() == 1)
            return this.pointwise1to1(x, (x, y) => Math.max(x, y));
        else if (this.type() == "xy")
            return this.pointwise2to2(x, (x, y) => [Math.max(x[0], y[0]), Math.max(x[1], y[1])]);
        else {
            return this.pointwise2to2(x, ([re1, im1], [re2, im2]) => {
                const mag1 = Math.sqrt(re1 * re1 + im1 * im1);
                const mag2 = Math.sqrt(re2 * re2 + im2 * im2);
                return mag1 > mag2 ? [re1, im1] : [re2, im2];
            });
        }
    }

    min(x: Vector | Position | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        x = this.coerce(x)
        if (this.dim() == 1)
            return this.pointwise1to1(x, (x, y) => Math.min(x, y));
        else if (this.type() == "xy")
            return this.pointwise2to2(x, (x, y) => [Math.min(x[0], y[0]), Math.min(x[1], y[1])]);
        else {
            return this.pointwise2to2(x, ([re1, im1], [re2, im2]) => {
                const mag1 = Math.sqrt(re1 * re1 + im1 * im1);
                const mag2 = Math.sqrt(re2 * re2 + im2 * im2);
                return mag1 < mag2 ? [re1, im1] : [re2, im2];
            });
        }
    }

    atan2(x: Vector | Position | number): Vector {
        if (!(x instanceof Vector)) x = v(x, this.type());
        if (this.length() > 1 && x.length() == 1) x = Vector.constant(x.at(0), this.length(), this.type());
        if (this.length() == 1 && x.length() > 1) this._expand(x.length());
        if (this.dim() == 1)
            return this.pointwise1to1(x, (x, y) => Math.atan2(x, y));
        else if (this.type() == "xy")
            return this.pointwise2to2(x, (x, y) => [
                Math.atan2(x[1], x[0]), // Angle of vector 1
                Math.atan2(y[1], y[0])  // Angle of vector 2
            ]);
        else {
            return this.pointwise2to2(x, ([re1, im1], [re2, im2]) => {
                const deltaRe = re2 - re1;
                const deltaIm = im2 - im1;
                return [Math.atan2(deltaIm, deltaRe), 0]; // Result is real; imaginary part is 0
            });
        }
    }

    sin(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.apply(([re, im]) => {
                const sinhIm = (Math.exp(im) - Math.exp(-im)) / 2;
                const coshIm = (Math.exp(im) + Math.exp(-im)) / 2;
                return [
                    Math.sin(re) * coshIm,
                    Math.cos(re) * sinhIm
                ];
            });
        }
        return this.apply1(Math.sin, dim);
    }

    cos(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.apply(([re, im]) => {
                const sinhIm = (Math.exp(im) - Math.exp(-im)) / 2;
                const coshIm = (Math.exp(im) + Math.exp(-im)) / 2;
                return [
                    Math.cos(re) * coshIm,
                    -Math.sin(re) * sinhIm
                ];
            });
        }
        return this.apply1(Math.cos, dim);
    }

    tan(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            const sin = this.sin();
            const cos = this.cos();
            return sin.divide(cos);
        }
        return this.apply1(Math.tan, dim);
    }

    abs(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.apply(([re, im]) => [Math.sqrt(re * re + im * im), 0]);
        }
        return this.apply1(Math.abs, dim);
    }

    sqrt(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.apply(([re, im]) => {
                const magnitude = Math.sqrt(Math.sqrt(re * re + im * im));
                const angle = Math.atan2(im, re) / 2;
                return [
                    magnitude * Math.cos(angle),
                    magnitude * Math.sin(angle)
                ];
            });
        }
        return this.apply1(Math.sqrt, dim);
    }

    log(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.apply(([re, im]) => {
                const magnitude = Math.sqrt(re * re + im * im);
                const angle = Math.atan2(im, re);
                return [Math.log(magnitude), angle];
            });
        }
        return this.apply1(Math.log, dim);
    }

    round(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.apply(([re, im]) => [Math.round(re), Math.round(im)]);
        }
        return this.apply1(Math.round, dim);
    }

    atan(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.apply(([re, im]) => {
                const re2 = Math.atan2(im, 1 + re);
                const im2 = Math.atan2(-im, 1 - re);
                return [0.5 * (re2 + im2), 0.5 * (re2 - im2)];
            });
        }
        return this.apply1(Math.atan, dim);
    }

    acos(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.apply(([re, im]) => {
                const z = v([re, im], "c").multiply(v([re, im], "c"));
                const oneMinusZ = v(1, "c").subtract(z);
                const sqrtOneMinusZ = oneMinusZ.sqrt();
                const logPart = Vector.i().multiply(this.add(sqrtOneMinusZ)).log();
                return logPart.multiply(v([-1, 0], "c")).at(0);
            });
        }
        return this.apply1(Math.acos, dim);
    }

    asin(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.apply(([re, im]) => {
                const sqrtPart = this.multiply(this).subtract(v(1)).sqrt();
                const logPart = Vector.i().multiply(this.add(sqrtPart)).log();
                return logPart.multiply(v([-1, 0], "c")).at(0);
            });
        }
        return this.apply1(Math.asin, dim);
    }

    log10(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.log().divide(v([Math.LN10, 0], "c"));
        }
        return this.apply1((v) => Math.log(v) / Math.LN10, dim);
    }

    log2(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.log().divide(v([Math.LN2, 0], "c"));
        }
        return this.apply1((v) => Math.log(v) / Math.LN2, dim);
    }

    sinh(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.apply(([re, im]) => {
                const sinhRe = (Math.exp(re) - Math.exp(-re)) / 2;
                const coshRe = (Math.exp(re) + Math.exp(-re)) / 2;
                return [
                    sinhRe * Math.cos(im),
                    coshRe * Math.sin(im)
                ];
            });
        }
        return this.apply1((v) => (Math.exp(v) - Math.exp(-v)) / 2, dim);
    }

    cosh(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.apply(([re, im]) => {
                const sinhRe = (Math.exp(re) - Math.exp(-re)) / 2;
                const coshRe = (Math.exp(re) + Math.exp(-re)) / 2;
                return [
                    coshRe * Math.cos(im),
                    sinhRe * Math.sin(im)
                ];
            });
        }
        return this.apply1((v) => (Math.exp(v) + Math.exp(-v)) / 2, dim);
    }

    tanh(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            const sinh = this.sinh();
            const cosh = this.cosh();
            return sinh.divide(cosh);
        }
        return this.apply1((v) => (Math.exp(v) - Math.exp(-v)) / (Math.exp(v) + Math.exp(-v)), dim);
    }

    sign(dim: "x" | "y" = "x"): Vector {
        if (this.type() === "c") {
            return this.apply(([re, im]) => {
                const magnitude = Math.sqrt(re * re + im * im);
                return [re / magnitude, im / magnitude];
            });
        }
        return this.apply1((v) => (v > 0 ? 1 : v < 0 ? -1 : 0), dim);
    }


    magnitude(): Vector {
        this._assertComplex();
        return new Vector(this.select(([re, im]) => Math.sqrt(re * re + im * im)), "x");
    }

    phase(): Vector {
        this._assertComplex();
        return new Vector(this.select(([re, im]) => Math.atan2(im, re)), "x");
    }

    angle(): Vector {
        var phase = this.phase();
        return phase.apply1(x => Region.toDegrees(x));
    }

    conjugate(): Vector {
        this._assertComplex();
        return new Vector(this.select(([re, im]) => [re, -im]), "c");
    }

    toPolar(): Vector {
        this._assertComplex();
        return this.magnitude().x().join(this.phase().xa());
    }

    fromPolar(): Vector {
        this._assertComplex();
        return this.pointwise2to2(this, ([mag, angle]) => [
            mag * Math.cos(angle),
            mag * Math.sin(angle)
        ]);
    }

    toString(): string {
        try {
            if (this.dim() == 1) return `Vector<${this.type()}>[${this.va().select(x => `${x}`).a}]`;
            else return `Vector<${this.type()}>[${stringJoin(this.select(x => `[${x}]`).a, ", ")}]`;
        } catch (e) {
            return `Vector<${this.type()}>[${this.a}]`;
        }
    }

    private _expand(n: number) {
        if (this.length() !== 1)
            throw "Expand is for singleton vectors";

        var expanded = Vector.constant(this.at(0), n, this.type());
        this.a = expanded.a;
    }

    private _assertComplex(): void {
        if (this.type() !== "c")
            throw "The specified vector must be complex valued";
    }

    static i(): Vector { return v([0, 1], "c"); }

    static zeros(n: number, type: VType = "x"): Vector {
        var x = v(0, type);
        x._expand(n);
        return x;
    }

    static constant(c: number | Position, n: number, type: VType = "x"): Vector {
        return new Vector(f(r(`:${n - 1}`).select(() => c)) as any, type);
    }

    static linspace(n: number, xrng?: Range): Vector
    static linspace(n: number, xmin?: number, xmax?: number): Vector
    static linspace(n: number, xmin: number | Range = 0, xmax: number = 1): Vector {
        if (VectorTypes.isRange(xmin)) {
            xmax = xmin[1];
            xmin = xmin[0];
        }
        const step = (xmax - xmin) / (n - 1);
        const values = f(r(`:${n - 1}`).select(i => xmin + i * step));
        return new Vector(values, "x");
    }

    static recursive(n: number, func: (seq: number[], i: number) => number, initial: number[]): Vector {
        const sequence = initial.slice();
        for (let i = initial.length; i < n; i++) {
            sequence.push(func(sequence, i));
        }
        return new Vector(f(sequence), "x");
    }

    // Primes
    static primes(n: number): Vector {
        const isPrime = (num: number): boolean => {
            if (num < 2) return false;
            for (let i = 2; i <= Math.sqrt(num); i++) {
                if (num % i === 0) return false;
            }
            return true;
        };

        const primes = [];
        let current = 2;
        while (primes.length < n) {
            if (isPrime(current)) primes.push(current);
            current++;
        }
        return new Vector(f(primes), "x");
    }

    // Fibonacci
    static fib(n: number, first: number = 0, second: number = 1): Vector {
        const fibSeq = [first, second];
        for (let i = 2; i < n; i++) {
            fibSeq.push(fibSeq[i - 1] + fibSeq[i - 2]);
        }
        return new Vector(f(fibSeq.slice(0, n)), "x");
    }

    // Collatz Sequence
    static collatz(start: number): Vector {
        const sequence = [start];
        while (start > 1) {
            start = start % 2 === 0 ? start / 2 : 3 * start + 1;
            sequence.push(start);
        }
        return new Vector(f(sequence), "x");
    }

    // Triangular Numbers
    static triangular(n: number): Vector {
        const triangularSeq = [];
        for (let i = 1; i <= n; i++) {
            triangularSeq.push((i * (i + 1)) / 2);
        }
        return new Vector(f(triangularSeq), "x");
    }

    // Factorials
    static factorial(n: number): Vector {
        const factorials = [1];
        for (let i = 1; i <= n; i++) {
            factorials.push(factorials[i - 1] * i);
        }
        return new Vector(f(factorials.slice(1)), "x");
    }

    // Pascal's Triangle Row
    static pascalRow(n: number): Vector {
        const row = [1];
        for (let i = 1; i <= n; i++) {
            row.push((row[i - 1] * (n - i + 1)) / i);
        }
        return new Vector(f(row), "x");
    }

    // Golden Ratio Modulus
    static goldenMod(n: number): Vector {
        const phi = (1 + Math.sqrt(5)) / 2;
        const sequence = [];
        for (let i = 0; i < n; i++) {
            sequence.push((i * phi) % 1); // Fractional part of i * phi
        }
        return new Vector(f(sequence), "x");
    }

    // Logistic Map
    static logisticMap(n: number, r: number = 3.9, x0: number = 0.5): Vector {
        const sequence = [x0];
        for (let i = 1; i < n; i++) {
            sequence.push(r * sequence[i - 1] * (1 - sequence[i - 1]));
        }
        return new Vector(f(sequence), "x");
    }

    // Cantor Dust
    static cantorDust(n: number): Vector {
        const sequence = [0.5];
        for (let i = 0; i < n; i++) {
            sequence.push(sequence[sequence.length - 1] / 3);
        }
        return new Vector(f(sequence), "x");
    }

    // Halton Sequence
    static haltonSequence(n: number, base: number = 2): Vector {
        const sequence = [];
        for (let i = 1; i <= n; i++) {
            let f = 1;
            let x = 0;
            let k = i;
            while (k > 0) {
                f /= base;
                x += f * (k % base);
                k = Math.floor(k / base);
            }
            sequence.push(x);
        }
        return new Vector(f(sequence), "x");
    }

    static sierpinski(n: number): Vector {
        const sequence = [];
        for (let i = 0; i < Math.pow(2, n); i++) {
            sequence.push((i & (i - 1)) === 0 ? 1 : 0); // Powers of 2
        }
        return new Vector(f(sequence), "x");
    }
