interface Item {
    [key: string]: any;
    name?: string;
    hidden?: boolean;
    readonly typename: string;
    readonly parent: any;
    readonly length?: number;
    readonly geometricBounds?: Rect;
    selected?: any;
    remove?(): void;
    removeAll?(): void;
    getIndex?(): number;
    getByName?(name: string): Item;
    add?(...args: any[]): Item;
    insert?(...args: any[]): void;
    duplicate?(relativeObject: object, insertionLocation: ElementPlacement): Item;
    move?(relativeObject: object, insertionLocation: ElementPlacement): Item;
    q: Queryable<this, any>;
    p: PositionableItem | undefined;
    c: ContainerItem | undefined;
}

class EmptyItem implements Item {
    name: string = "";
    typename: string = "EMPTY";
    parent: Application = app;
    length: number = 0;
    geometricBounds: Rect = [0, 0, 0, 0];
    selected: any = false;
    remove(): void { }
    removeAll(): void { }
    getIndex(): number { return -1; }
    getByName(name: string): Item { return this; }
    add(...args: any[]): Item { return this; }
    insert(...args: any[]): void { }
    duplicate(relativeObject: object, insertionLocation: ElementPlacement): Item { return this; }
    move?(relativeObject: object, insertionLocation: ElementPlacement): Item { return this; }
    q: Queryable<this, any> = undefined as any;
    p: PositionableItem | undefined;
    c: ContainerItem | undefined;

    private constructor() { }

    private static _empty: EmptyItem;
    static empty(): EmptyItem {
        if (EmptyItem._empty) return EmptyItem._empty;
        EmptyItem._empty = new EmptyItem();
        return EmptyItem._empty;
    }
}

interface ItemCollection extends Array<any> {
    readonly length: number
    readonly parent: object
    readonly typename: string
    getByName(name: string): any
}

class QueryItem {
    _item: any;
    id: number;
    itemType: string;
    itemDefaults: any;
    queryItemType: string;
    queryItemDefaults: any;
    queryType: string;
    queryDefaults: any;
    collectionType: CollectionType | undefined;
    
    p(): PositionableItem {
        if (this._item.p === undefined)
            this._item.p = new PositionableItem(this as any);
        return this._item.p;
    }

    set(name?: string): SetProperties { return new SetProperties(this, name); }

    name(value?: string): string {
        if (value !== undefined) {
            this._item.name = value;
            return this._item.name;
        }
        return this._item?.name ? this._item.name.toString() : "";
    }

    api(): ApplicationQuery { return api; }
    app(): Application { return _app(); }
    empty(): boolean { return this._item.typename === "EMPTY"; }
    activated(): boolean { return !!this._item?.selected; }
    activate(): QueryItem { if (this._item) { this._item.selected = true; } return this; }
    deactivate(): QueryItem { if (this._item) { this._item.selected = false; } return this; }
    hide(): QueryItem { if (this._item.hidden !== undefined) { this._item.hidden = true; } return this; }
    hidden(): boolean { return this._item.hidden === true; }
    show(): QueryItem { if (this._item.hidden !== undefined) { this._item.hidden = false; } return this; }
    visible(): boolean { return this._item.hidden !== true; }
    length(): number { return 0; }
    enumerable(): boolean { return this.collectionType !== undefined; }
    positionable(): boolean { return (this._item as PageItem)?.position !== undefined; }
    collection(): any | any[] { return this.collectionType !== undefined ? (this._item as any)[this.collectionType] : []; }

    parent(): QueryItem { 
        if (!this._item.parent) QueryItem.empty() as any; 
        var parent = q(this._item.parent) as any;
        var collection = this.itemType.charAt(0).toLowerCase() + this.itemType.slice(1) + "s";
        if (parent[collection])
            parent = parent[collection]();
        return parent;
    }

    document(): DocumentQuery {
        if (this.empty()) return this as any;
        let currentItem = this._item;
        while (currentItem) {
            if (currentItem.typename === "Document")
                return q(currentItem);
            currentItem = currentItem.parent;
        }
        return QueryItem.empty() as DocumentQuery;
    }

    artboard(): ArtboardQuery {
        if (this.empty()) return this as any;
        return this.document().artboard();
    }

    layer(): LayerQuery {
        if (this.empty()) return this as any;
        return this.document().layer();
    }

    region(newRegion? : Region): Region {
        if (newRegion !== undefined) {
            if (this.positionable()) 
                this.p().region(newRegion);
            else 
                throw new Error("Only positionalble objects can have custom regions");
        }
        return this.positionable() ? 
            this.p().region() : 
            this.artboard().region();
    }

    constructor(item: any = EmptyItem.empty(), collectionType?: CollectionType) {
        this._item = item;
        this._item.q = this;
        this.id = QueryItem.nextId();
        this.itemType = this._item?.typename ?? "EMPTY";
        this.queryItemType = this.constructor.name;
        this.queryType = `${this.queryItemType}<${this.itemType}>`;
        this.collectionType = collectionType;

        if (QueryItem._itemDefaults[this.itemType] === undefined)
            QueryItem._itemDefaults[this.itemType] = {}
        if (QueryItem._queryItemDefaults[this.queryItemType] === undefined)
            QueryItem._queryItemDefaults[this.queryItemType] = {}
        if (QueryItem._queryDefaults[this.queryType] === undefined)
            QueryItem._queryDefaults[this.queryType] = {}

        this.itemDefaults = QueryItem._itemDefaults[this.itemType];
        this.queryItemDefaults = QueryItem._queryItemDefaults[this.queryItemType];
        this.queryDefaults = QueryItem._queryDefaults[this.queryType];
        this._configureDefaults();
    }

    cast<T extends QueryItem>() {
        return this as any as T
    }

    equals(item: QueryItem): boolean {
        if (item === undefined) return false;
        return this._item == item._item;
    }

    remove() {
        this._item?.remove?.();
        this._item = QueryItem.empty();
    }

    removeIfEmpty(): boolean {
        var empty = this.empty();
        if (empty)
            this.remove();
        return empty;
    }

    duplicate(name: string, relativeObject?: QueryItem, insertionLocation: ElementPlacement = ElementPlacement.PLACEAFTER): Queryable<any, any> {
        if (relativeObject === undefined) relativeObject = this;
        var duplicate = this._item?.duplicate?.(relativeObject._item, insertionLocation);
        duplicate.name = name;
        return q(duplicate);
    }

    move(relativeObject: QueryItem, insertionLocation: ElementPlacement): void {
        this._item?.move?.(relativeObject._item, insertionLocation);
    }

    moveTo(index: number): this { 
        return this; 
    }

    locked(): boolean { return this._item.locked; }
    lock(): this { this._item.locked = true; return this; }
    unlock(): this { this._item.locked = false; return this; }

    protected _create<TCollection extends ItemCollection, TQuery extends QueryItem>(
        select: (item: this) => TCollection, 
        create: (colletion: TCollection) => TQuery,
        name?: string
    ): TQuery {
        var collection = select(this);
        var queryItem = create(collection);
        if (name !== undefined)
            queryItem.set().name(name);
        var item: Item = queryItem._item;
        item?.move?.(queryItem.parent()._item, ElementPlacement.PLACEATEND);
        return queryItem;
    }

    protected _configureDefaults() { }

    static createAnonymousName(item: QueryItem): string {
        if (!item) return `.Item[${item}]`;
        return `.${item.itemType ? item.itemType : "Item"}[${item.id}]`;
    }

    private static _empty: QueryItem;
    static empty(): QueryItem {
        if (QueryItem._empty) return QueryItem._empty;
        QueryItem._empty = new Queryable<any, any>(EmptyItem.empty(), undefined);
        QueryItem._empty._item.q = QueryItem._empty;
        return QueryItem._empty;
    }

    private static _nextId: number = 1;
    private static nextId(): number {
        var nextId = QueryItem._nextId;
        QueryItem._nextId += 1;
        return nextId;
    }

    private static _itemDefaults = dictionary();
    private static _queryItemDefaults = dictionary();
    private static _queryDefaults = dictionary();
}

class Queryable<I extends Item, T extends QueryItem> extends QueryItem {
    item(): I { return this._item; }
    active(): boolean { return !!this._item.selected; }
    inactive(): boolean { return !this.active(); }
    activate() { this._item.selected = true; return this; }
    deactivate() { this._item.selected = false; return this; }

    constructor(item: I, collectionType?: CollectionType) {
        super(item, collectionType);
        this._item = item;
    }

    query(selector: string = ":"): A<T> {
        var collection = this.collection();
        if (collection.length == 0) return f();
        var indexes = evaluateRangeSelector(selector, collection.length)
        var len = indexes.length;
        var result: A<T> = f();
        for (var i = 0; i < len; i++) {
            var item = collection[indexes[i]];
            if (item)
                result.push(q(item));
        }

        return result;
    }

    index(newIndex?: number): number {
        if (newIndex !== undefined) {
            var index = this.index();
            var parent = this.parent() as Queryable<any, any>;

            newIndex = evaluateNegativeIndex(newIndex, parent.length());
            if (index == newIndex) 
                return index;
            if (index < newIndex)
                this.move(parent.at(newIndex), ElementPlacement.PLACEAFTER);
            if (index > newIndex)
                this.move(parent.at(newIndex), ElementPlacement.PLACEBEFORE);
        }
        return (this.parent() as Queryable<any, any>).firstIndex(i => i._item == this._item);
    }

    create(name?: string): T {
        var collection = this.collection() as any;
        var newItem = q(collection.add()) as T;
        if (name !== undefined)
            newItem.name(name);

        if (this.length() > 1)
            newItem.move(this.last() as T, ElementPlacement.PLACEAFTER);

        return newItem;
    }

    moveTo(index: number): this {
        if (index == undefined) throw "Index not specified";
        this.index(index);
        return this;
    }

    reverse(selector: string = ":"): A<T> {
        return this.query(selector).reverse();
    }

    at(index: number): T {
        return this.query(`${index}`).at(0);
    }

    get(selector: string): A<T> {
        return this.query(selector);
    }

    getByName(name: string): Queryable<any, any> {
        var collection = this.collection() as any;
        return q(collection?.getByName?.(name));
    }

    length(): number { return this.collection().length; }

    replace(index: number, item: T): void {
        var itemAtIndex = this.at(index);
        item.move(itemAtIndex, ElementPlacement.PLACEBEFORE);
        itemAtIndex.remove?.();
    }

    insert(item: T) {
        this.insertAt(-1, item);
    }

    insertAt(index: number, item: T): void {
        var len = this.length();
        index = evaluateNegativeInsertIndex(index, len);
        if (index == 0) {
            item._item.move(this.item(), ElementPlacement.PLACEATBEGINNING);
            return;
        }
        if (index == len) {
            item._item.move(this.item(), ElementPlacement.PLACEATEND);
            return;
        }
        var itemBeforeIndex = this.at(index - 1);
        item.move(itemBeforeIndex, ElementPlacement.PLACEAFTER);
    }

    insertMany(items: T[] | A<T>) {
        f(items).do(item => this.insert(item));
    }

    insertManyAt(index: number, items: A<T>) {
        var len = items.length();
        index = evaluateNegativeInsertIndex(index, this.collection.length);
        for (var i = 0; i < len; i++) {
            this.insertAt(index + i, items.at(i));
        }
    }

    delete(item: T) {
        item.remove();
    }

    deleteMany(items: A<T>) {
        items.do(item => item.remove());
    }

    deleteAt(index: number, count: number = 1) {
        for (var i = 0; i < count; i++)
            this.delete(this.at(index)._item);
    }

    removeAll() {
        this.query().do(item => { item.remove() });
    }

    forEach(callback: (item: T, index?: number) => any | void): any | void {
        return this.query().forEach(callback);
    }

    do(callback: (item: T, index?: number) => any | void): any | void {
        return this.query().forEach(callback);
    }

    doOnce(callback: (item: T, index?: number) => boolean): boolean {
        var collection = this.collection();
        for (let i = 0; i < collection.length; i++) {
            if (callback(q(collection[i]), i)) {
                return true;
            }
        }
        return false;
    }

    select<U>(selector: (item: T) => U): A<U> {
        return this.query().select(selector);
    }

    where(predicate: (item: T) => boolean): A<T> {
        return this.query().where(predicate);
    }

    whereNameMatches(pattern: RegExp): A<T> {
        return this.where(item => pattern.test(item.name()));
    }

    // Simple string pattern matchers that use whereNameMatches
    whereNameStartsWith(prefix: string): A<T> {
        return this.whereNameMatches(new RegExp(`^${this._escapeRegExp(prefix)}`));
    }

    whereNameEndsWith(suffix: string): A<T> {
        return this.whereNameMatches(new RegExp(`${this._escapeRegExp(suffix)}$`));
    }

    whereNameContains(substring: string): A<T> {
        return this.whereNameMatches(new RegExp(this._escapeRegExp(substring)));
    }

    // Helper to escape special regex characters in string literals
    private _escapeRegExp(string: string): string {
        return string.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
    }

    first(predicate?: (item: T) => boolean): T {
        var len = this.length();
        if (!predicate) return len > 0 ? this.at(0) : undefined as any as T;
        for (let i = 0; i < len; i++) {
            var item = this.at(i);
            if (predicate(item)) return item;
        }
        return undefined as any as T;
    }

    firstIndex(predicate: (item: T) => boolean): number {
        var len = this.length();
        for (let i = 0; i < len; i++) {
            var item = this.at(i);
            if (predicate(item)) return i;
        }
        return -1;
    }

    last(predicate?: (item: T) => boolean): T {
        var len = this.length();
        if (!predicate) return len > 0 ? this.at(-1) : undefined as any as T;
        for (let i = len - 1; i >= 0; i--) {
            var item = this.at(i);
            if (predicate(item)) return item;
        }
        return undefined as any as T;
    }

    lastIndex(predicate: (item: T) => boolean): number {
        var len = this.length();
        for (let i = len - 1; i >= 0; i--) {
            var item = this.at(i);
            if (predicate(item)) return i;
        }
        return -1;
    }

    skip(count: number): A<T> {
        var len = this.length();
        if (count >= len) return f() as A<T>;
        count = Math.min(count, len);
        return this.query(`${count}:`);
    }

    take(count: number): A<T> {
        if (count == 0) return f() as A<T>;
        count = Math.min(count, this.length());
        return this.query(`:${count - 1}`);
    }

    any(predicate?: (item: T) => boolean): boolean {
        if (predicate === undefined) return this.length() > 0;
        return this.where(predicate).length() > 0;
    }

    all(predicate: (item: T) => boolean): boolean {
        return this.where(predicate).length() == this.length();
    }

    array(): A<T> {
        var result: A<T> = f([]);
        this.do(item => result.push(item));
        return result;
    }

    includes(item: T): boolean {
        return this.any(i => i._item == item._item);
    }

    hideAll(predicate?: (item: T) => boolean): this {
        if (predicate === undefined) {
            this.do(item => item.hide());
        } else {
            this.where(predicate).do(item => item.hide());
        }
        return this;
    }

    showAll(predicate?: (item: T) => boolean): this {
        if (predicate === undefined) {
            this.do(item => item.show());
        } else {
            this.where(predicate).do(item => item.show());
        }
        return this;
    }

    activateAll(predicate?: (item: T) => boolean): this {
        if (predicate === undefined) {
            this.do(item => item.activate());
        } else {
            this.where(predicate).do(item => item.activate());
        }
        return this;
    }

    inactivateAll(predicate?: (item: T) => boolean): this {
        if (predicate === undefined) {
            this.do(item => item.deactivate());
        } else {
            this.where(predicate).do(item => item.deactivate());
        }
        return this;
    }
}

class ApplicationQuery extends Queryable<Application, DocumentQuery<LayerQuery<PageItemQuery>>> {
    set(name?: string): SetApplicationProperties { return new SetApplicationProperties(this, name); }
    documents(): ApplicationQuery { return this; }
    document(): DocumentQuery { return this.activeDocument(); }
    defaults() { return clone<ApplicationDefaults>(this.queryDefaults); }
    activeDocument(): DocumentQuery<LayerQuery<PageItemQuery>> { return q(this.item().activeDocument); }
    dpi(): number { return RasterEffectOptions.resolution; }

    constructor() {
        if (api !== undefined)
            throw "Application cannot be defined twice";

        var app = _app();
        if (!app)
            throw "App is undefined";

        super(_app(), "documents");
        this.init();
    }

    init() {
        // Make sure all coordinates in all new documents are in points
        DocumentPreset.units = RulerUnits.Points;

        // Set the ruler origin to bottom-left
        this.app().preferences.setBooleanPreference("isRulerOriginTopLeft", false);

        // Set the Y-axis to increase upward (1st quadrant behavior)
        this.app().preferences.setBooleanPreference("isRulerIn4thQuad", false);

        // Sets ruleUnits to points for all new documents
        this.app().preferences.setIntegerPreference("rulerUnits", 0); // 0 = Points
    }

    isOpen(fileName: string): boolean {
        // Check if the document exists in the currently open documents
        var isOpen = false;
        this.documents().forEach(doc => {
            if (doc.name() === fileName) {
                isOpen = true;
            }
        });
        return isOpen;
    }

    exists(fileName: string): boolean {
        // Check if the document exists in the currently open documents
        if (this.isOpen(fileName))
            return true;

        // Check if the document exists in the file system
        const file = new File(combinePaths(DOCUMENTS_PATH, fileName));
        return file.exists;
    }

    createDocument(fileName: string, width: number, height: number, activate: boolean = true): DocumentQuery {
        var doc: DocumentQuery = q(app.documents.add(DocumentColorSpace.RGB, width, height, 1));
        doc.save(fileName);

        if (activate) doc.activate();
        return doc;
    }

    openDocument(fileName: string, create: boolean = false): DocumentQuery {
        const file = new File(combinePaths(DOCUMENTS_PATH, fileName));
        var openDoc: DocumentQuery | undefined;
        if (this.exists(fileName)) {
            this.documents().doOnce(doc => {
                if (doc.name() === fileName) {
                    openDoc = doc;
                    openDoc.activate();
                    return true;
                }
                return false;
            });

            if (openDoc) {
                openDoc.activate();
                openDoc.activeArtboard().fullScreen();
                return openDoc;
            } else {
                return q<DocumentQuery>(this.api().item().open(file));
            }
        }

        if (create) {
            var size = this.defaults().documentSize;
            var newDoc = this.createDocument(fileName, size, size, true);
            newDoc.activate();
            return newDoc;
        }

        throw `Failed to open document: ${fileName}`;
    }

    openResource(fileName: string): DocumentQuery {
        return this.openDocument(combinePaths("resources", fileName), false);
    }

    closeDocument(doc: DocumentQuery | string) {
        this.api().item().userInteractionLevel = UserInteractionLevel.DONTDISPLAYALERTS;
        if (typeof doc == "string") {
            api.documents().do(d => { if (d.name() == doc) d.item().close(SaveOptions.DONOTSAVECHANGES); });
            return;
        }

        doc?.item().close(SaveOptions.DONOTSAVECHANGES);
        app.userInteractionLevel = UserInteractionLevel.DISPLAYALERTS;
    }

    closeResource(doc: DocumentQuery | string) {
        this.closeDocument(doc);
    }

    deleteDocument(fileName: string) {
        if (api.exists(fileName)) {
            // Try to delete through Illustrator first if it's open
            api.documents().doOnce(doc => {
                if (doc.name() === fileName) {
                    doc.remove();
                    return true;
                }
                return false;
            });

            // Also try to delete the file directly
            const file = new File(combinePaths(DOCUMENTS_PATH, fileName));
            if (file.exists) {
                file.remove();
            }
        }
    }

    redraw(count: number = 1) {
        if (!api.documents().any())
            return;

        // Force Illustrator to redraw the canvas
        for (var i = 0; i < count; i++)
            app?.redraw();

        $.sleep(10);
    }

    reset(count: number = 1) {
        this.api().item().selection = [];
        this.redraw(count);
    };
               
    sleep(msecs: number) {
        $.sleep(msecs);
    }

    protected _configureDefaults(): void {
        const defaults = this.queryDefaults as ApplicationDefaults;
        defaults.documentSize = 5 * 12 * this.dpi();
        defaults.layerName = "Layer 1";
        defaults.artboardRect = [0, 1199, 1198, 0];
    }
}

interface ApplicationDefaults {
    documentSize: number;
    layerName: string;
    artboardRect: Rect;
}

class DocumentQuery<T extends QueryItem = LayerQuery<PageItemQuery>> extends Queryable<Document, T> {
    private _layers: DocumentQuery<LayerQuery<PageItemQuery>> | undefined;
    private _artboards: DocumentQuery<ArtboardQuery> | undefined;
    private _views: DocumentQuery<ViewQuery> | undefined;
    private _stories: DocumentQuery<StoryQuery<TextRangeQuery>> | undefined;
    private _variables: DocumentQuery<VariableQuery> | undefined;
    private _swatches: DocumentQuery<SwatchQuery> | undefined;
    private _brushes: DocumentQuery<BrushQuery> | undefined;
    private _symbols: DocumentQuery<SymbolQuery> | undefined;
    private _patterns: DocumentQuery<PatternQuery> | undefined;
    private _spots: DocumentQuery<SpotQuery> | undefined;
    private _gradients: DocumentQuery<GradientQuery> | undefined;
    private _tags: DocumentQuery<TagQuery> | undefined;

    set(name?: string): SetDocumentProperties { return new SetDocumentProperties(this as any, name); }
    active(): boolean { return this.api().item().activeDocument == this._item; }
    activate() { this.api().item().activeDocument = this._item; return this; }
    deactivate() { this.api().item().activeDocument = f(this.api().item().documents).at(-1); return this; }
    remove() { this._item?.close?.(); this._item = QueryItem.empty(); }
    activeArtboard(): ArtboardQuery { return this.artboards().at(this.getActiveArtboardIndex()); }
    activeLayer(): LayerQuery { return q(this.item().activeLayer); }
    activeView(): ViewQuery { return q(this.item().activeView); }
    defaults(): ApplicationDefaults { return this.api().defaults(); }

    rulerUnits(): RulerUnits { return this.item().rulerUnits; }
    dpi(): number { return this.api().dpi(); }

    // QueryItem overrides
    document(): DocumentQuery { return this as any; }
    artboard(): ArtboardQuery { return this.activeArtboard(); }
    layer(): LayerQuery { return this.activeLayer(); }
    region(): Region { return this.activeArtboard().region(); }

    isNew(): boolean {
        var layers = this.layers();
        var layer = layers.first();
        return layers.length() == 1 && 
            layer.name() == "Layer 1" && 
            layer.pageItems().length() == 0;  
    }

    constructor(doc: Document, collectionType: CollectionType = "layers") {
        super(doc, collectionType);

        // Ensures Cartesian coordinates
        this.item().rulerOrigin = [0, 0];
    }

    save(fileName?: string): this {
        try {
            if (fileName === undefined) {
                this._item.save();
                return this;
            }
            const file = new File(combinePaths(DOCUMENTS_PATH, fileName));
            this._item.saveAs(file);
            return api.documents().first(d => d.name() == fileName) as any;
        } catch (error: any) {
            throw new Error(`Failed to save document '${fileName}': ${error.message}`);
        }
    }

    close() {
        this.remove();
    }

    reset() {
        if (this.artboards().any()) {
            this.layers().removeAll();
            this.artboards().removeAll();
        }
    }

    fullscreen() {
        if (this.artboards().any())
            this.activeArtboard().fullScreen();
    }

    create(name?: string): T {
        if (this.collectionType !== "artboards")
            return super.create(name);

        throw "Cannot create an artbaord because it is treated as a singleston";
    }

    removeAll(): void {
        if (this.collectionType == "artboards") {
            var activeArtboard = this.activeArtboard();
            this.where(a => a._item !== activeArtboard._item).do(a => a.remove());
            this.activeArtboard().reset();
        } else if (this.collectionType == "layers") {
            var defaultLayer = this.layers().length() > 0 ? this.layers().at(0) : this.layers().create();
            defaultLayer.name(this.defaults().layerName);
            defaultLayer.pageItems().removeAll();
            defaultLayer.activate();
            this.layers().skip(1).do(layer => layer.remove());
            this.layers().first().removeAll();
        } else {
            super.removeAll();
        }
        api.redraw();
    }

    copyLayer(layer: LayerQuery, index: number = -1): LayerQuery {
        var differentDocuments = this.item() !== layer.document().item();
        if (differentDocuments) {
            // Match the aspect ratio of the source artboard
            layer.document().activate();
            const sourceAspectRatio = layer.document().activeArtboard().aspectRatio();

            this.activate();
            this.activeArtboard().reset();
            this.activeArtboard().aspectRatio(sourceAspectRatio);
            this.rearrangeArtboards();
        }

        // Activate the source document and get its artboard rect
        layer.document().activate();
        const sourceRect = layer.document().activeArtboard().rect();

        // Pre-calculate relative centers for all items in the source layer
        const itemCenters = layer.pageItems().select(p => p.p().position());

        // Activate the target document and get its artboard rect
        this.activate();
        const targetRect = this.activeArtboard().rect();

        // Create the new layer in the target document
        const layers = this.layers();
        const newLayer = layers.create(layer.name());
        newLayer.replaceWith(layer);
        newLayer.moveTo(index);

        if (differentDocuments) {
            const sourceWidth = Math.abs(sourceRect[2] - sourceRect[0]);
            const sourceHeight = Math.abs(sourceRect[1] - sourceRect[3]);
            const targetWidth = Math.abs(targetRect[2] - targetRect[0]);
            const targetHeight = Math.abs(targetRect[1] - targetRect[3]);
            const scaleFactors = { scaleX: targetWidth / sourceWidth, scaleY: targetHeight / sourceHeight }

            // Apply scaling and translation using the pre-stored relative centers
            newLayer.pageItems().forEach((item, i) => {
                i = i as number;
                item.p().scale([scaleFactors.scaleX, scaleFactors.scaleY]);
                this.p().position(itemCenters.at(i));
            });
        }

        return newLayer;
    }

    copyArtboard(artboard: ArtboardQuery) {
        artboard.document().activate();
        this.activeArtboard().aspectRatio(artboard.aspectRatio());
        this.activate();
    }

    getActiveArtboardIndex(): number {
        var artboards: Artboards = this.artboards().collection() as Artboards;
        var activeIndex = artboards.getActiveArtboardIndex();
        if (activeIndex === undefined || activeIndex === null || activeIndex < 0) {
            artboards.setActiveArtboardIndex(0);
            return 0;
        }
        return activeIndex;
    }

    rearrangeArtboards(artboardLayout?: DocumentArtboardLayout, artboardRowsOrCols?: number, artboardSpacing?: number, artboardMoveArtwork?: boolean): boolean {
        return this.item().rearrangeArtboards(artboardLayout, artboardRowsOrCols, artboardSpacing, artboardMoveArtwork);

class ArtboardQuery extends Queryable<Artboard, QueryItem> {
    set(name?: string): SetArtboardProperties { return new SetArtboardProperties(this as any, name); }
    document(): DocumentQuery { return (this.parent() as DocumentQuery).artboards() as any; }
    index(): number { return this.document().artboards().firstIndex(i => i._item == this._item); }
    activate() { this.document().item().artboards.setActiveArtboardIndex(this.index()); return this; }
    activated(): boolean { return this.document().activeArtboard().item() === this.item(); }
    deactivate() { ((this.parent() as DocumentQuery).item()).artboards.setActiveArtboardIndex(0); return this; }

    // How to change the absoluie coordinate of the Artboard
    region(region?: Region): Region {
        if (region !== undefined) {
            var newArtboardRect = region.base().rect();
            this.item().artboardRect = newArtboardRect;
        }
        return Region.fromRect(this.item().artboardRect);
    }

    // Absolute bounds in document coordinates
    rect(): Rect { return this.item().artboardRect; }

    center(newCenter?: Position): Position {
        if (newCenter) {
            var region = this.region();
            region.center(newCenter);
            this.region(region);
        }
        return this.region().center();
    }

    width(newWidth?: number, proportional?: boolean): number {
        if (newWidth !== undefined) {
            var region = this.region();
            region.width(newWidth, proportional);
            this.region(region);
        }
        return this.region().width();
    }

    height(newHeight?: number, proportional?: boolean): number {
        if (newHeight !== undefined) {
            var region = this.region();
            region.height(newHeight, proportional);
            this.region(region);
        }
        return this.region().height();
    }

    size(newSize?: Size): Size {
        if (newSize) {
            var region = this.region();
            region.size(newSize);
            this.region(region);
        }
        return this.region().size();
    }

    aspectRatio(widthToHeight?: number): number {
        if (widthToHeight !== undefined) {
            const width = this.width(); // Current width in relative units
            const height = this.height(); // Current height in relative units

            // Calculate the new dimensions while maintaining the specified aspect ratio
            const newWidth = Math.min(width, height * widthToHeight);
            const newHeight = Math.min(height, width / widthToHeight);

            // Set the new size centered on the current position
            this.size([newWidth, newHeight]);
        }
        return this.width() / this.height();
    }

    constructor(artboard: Artboard) {
        super(artboard);
    }

    reset() {
        var rect = this.document().defaults().artboardRect;
        if (f(rect).doesntEqual(f(this.item().artboardRect))) {
            var region = Region.fromRect(rect);
            this.region(region);
        }
        this.fullScreen();
    }

    fullScreen(padding: number = 0.05): ArtboardQuery {
        if (this.isFullScreen(padding)) return this;

        const view = this.document().item().activeView;

        // Step 1: Reset to a known zoom level (100%)
        view.zoom = 1;

        // Step 2: Calculate viewable area at this zoom
        const viewBounds = view.bounds; // [left, top, right, bottom]
        const viewWidth = viewBounds[2] - viewBounds[0];
        const viewHeight = viewBounds[1] - viewBounds[3];

        // Step 3: Adjust artboard dimensions with padding
        const region = this.region();
        const paddedWidth = region.widthAbs() * (1 + padding);
        const paddedHeight = region.heightAbs() * (1 + padding);

        // Step 4: Determine the scale to fit the padded artboard into the view
        const scale = Math.min(viewWidth / paddedWidth, viewHeight / paddedHeight);
        const centerPoint = region.centerAbs() as [number, number];

        // Step 5: Apply the calculated zoom and center the view
        view.centerPoint = centerPoint;
        view.zoom = scale;

        return this;
    }

    isFullScreen(padding: number = 0.05): boolean {
        const view = this.document().item().activeView;

        // Get current view bounds [left, top, right, bottom]
        const viewBounds = view.bounds;
        const viewWidth = viewBounds[2] - viewBounds[0];
        const viewHeight = viewBounds[1] - viewBounds[3];

        // Get artboard region and padded dimensions
        const region = this.region();
        const paddedWidth = region.widthAbs() * (1 + padding);
        const paddedHeight = region.heightAbs() * (1 + padding);

        // Define a tolerance for comparison
        const tolerance = 2; // Allow a difference of points

        // Check if view dimensions approximately match the padded artboard dimensions
        const widthMatch = Math.abs(viewWidth - paddedWidth) <= tolerance;
        const heightMatch = Math.abs(viewHeight - paddedHeight) <= tolerance;

        // Check if center points approximately align
        const viewCenter = view.centerPoint;
        const regionCenter = region.centerAbs();
        const centerXMatch = Math.abs(viewCenter[0] - regionCenter[0]) <= tolerance;
        const centerYMatch = Math.abs(viewCenter[1] - regionCenter[1]) <= tolerance;

        // Only one dimension will be full-sized because the screen and the board have different shapes
        return (widthMatch || heightMatch) && centerXMatch && centerYMatch;
    }

    square(): ArtboardQuery {
        this.aspectRatio(1);
        return this;
    }
}

    private _paragraphs: TextFrameQuery<TextRangeQuery> | undefined;
    set(name?: string): SetTextFrameProperties { return new SetTextFrameProperties(this as any, name); }

    constructor(textFrame: TextFrame, collectionType: CollectionType = "characters") {
        super(textFrame, collectionType);
    }

    characters() {
        if (this.collectionType === "characters") return this;
        if (this._characters) return this._characters;
        this._characters = new TextFrameQuery<TextRangeQuery>(this._item, "characters");
        return this._characters;
    }

    words() {
        if (this.collectionType === "words") return this;
        if (this._words) return this._words;
        this._words = new TextFrameQuery<TextRangeQuery>(this._item, "words");
        return this._words;
    }

    lines() {
        if (this.collectionType === "lines") return this;
        if (this._lines) return this._lines;
        this._lines = new TextFrameQuery<TextRangeQuery>(this._item, "lines");
        return this._lines;
    }

    paragraphs() {
        if (this.collectionType === "paragraphs") return this;
        if (this._paragraphs) return this._paragraphs;
        this._paragraphs = new TextFrameQuery<TextRangeQuery>(this._item, "paragraphs");
        return this._paragraphs;
    }

    createCharacter(contents: string, name?: string): TextRangeQuery {
        return this._create(i => i.item().characters, items => q(items.add(contents)), name);
    }
    
    createWord(contents: string, name?: string): TextRangeQuery {
        return this._create(i => i.item().words, items => q(items.add(contents)), name);
    }
    
    createParagraph(contents: string, name?: string): TextRangeQuery {
        return this._create(i => i.item().paragraphs, items => q(items.add(contents)), name);
    }
}

function _app(): Application {
    if ("app" in $.global)
        return app;
    throw "The app is not defined. Modify your script to open a document first";
}
if (!_app().q) {
    _app().q = new ApplicationQuery();
    var api = _app().q;
}

function q<TQuery extends QueryItem = QueryItem>(obj: any): TQuery {
    if (obj && obj.q) return obj.q;
    if (!obj || obj.typename === undefined) return QueryItem.empty() as any;

    // Switch based on the typename and create the appropriate query instance
    var q: any = undefined;;
    switch (obj.typename) {
        case "Application":
            return api as any;
        case "Document":
            q = new DocumentQuery(obj as Document);
            break;
        case "Layer":
            q = new LayerQuery(obj as Layer);
            break;
        case "PathItem":
            q = new PathItemQuery(obj as PathItem);
            break;
        case "TextFrame":
            q = new TextFrameQuery(obj as TextFrame);
            break;
        case "Story":
            q = new StoryQuery(obj as Story);
            break;
        case "Gradient":
            q = new GradientQuery(obj as Gradient);
            break;
        case "Artboard":
            q = new ArtboardQuery(obj as Artboard);
            break;
        case "PageItem":
            q = new PageItemQuery(obj as PageItem);
            break;
        case "CompoundPathItem":
            q = new CompoundPathItemQuery(obj as CompoundPathItem);
            break;
        case "RasterItem":
            q = new RasterItemQuery(obj as RasterItem);
            break;
        case "PlacedItem":
            q = new PlacedItemQuery(obj as PlacedItem);
            break;
        case "GroupItem":
            q = new GroupItemQuery(obj as GroupItem);
            break;
        case "SymbolItem":
            q = new SymbolItemQuery(obj as SymbolItem);
            break;
        case "MeshItem":
            q = new MeshItemQuery(obj as MeshItem);
            break;
        case "PluginItem":
            q = new PluginItemQuery(obj as PluginItem);
            break;
        case "GraphItem":
            q = new GraphItemQuery(obj as GraphItem);
            break;
        case "Swatch":
            q = new SwatchQuery(obj as Swatch);
            break;
        case "Brush":
            q = new BrushQuery(obj as Brush);
            break;
        case "Tag":
            q = new TagQuery(obj as Tag);
            break;
        case "Symbol":
            q = new SymbolQuery(obj as Symbol);
            break;
        case "TextRange":
            q = new TextRangeQuery(obj as TextRange);
            break;
        case "InsertionPoint":
            q = new InsertionPointQuery(obj as InsertionPoint);
            break;
        case "CharacterStyle":
            q = new CharacterStyleQuery(obj as CharacterStyle);
            break;
        case "ParagraphStyle":
            q = new ParagraphStyleQuery(obj as ParagraphStyle);
            break;
        case "Pattern":
            q = new PatternQuery(obj as Pattern);
            break;
        case "Spot":
            q = new SpotQuery(obj as Spot);
            break;
        case "ArtStyle":
            q = new ArtStyleQuery(obj as ArtStyle);
            break;
        case "LegacyTextItem":
            q = new LegacyTextItemQuery(obj as LegacyTextItem);
            break;
        case "Vari":
            q = new VariableQuery(obj as Variable);
            break;
        case "DataSet":
            q = new DataSetQuery(obj as DataSet);
            break;
        case "View":
            q = new ViewQuery(obj as View);
            break;
        case "NonNativeItem":
            q = new NonNativeItemQuery(obj as NonNativeItem);
            break;
        case "PathPoint":
            q = new PathPointQuery(obj as PathPoint);
            break;
        default:
            // Handle unsupported or unknown typenames
            obj.q = Queryable.empty;
            q = obj.q;
            break;
    }
    return q;
};
