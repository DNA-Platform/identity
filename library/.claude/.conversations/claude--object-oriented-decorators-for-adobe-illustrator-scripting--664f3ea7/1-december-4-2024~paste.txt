class Query<TCollection extends Collection, TParent extends Item> {
    collection: TCollection;
    parent: TParent;
    constructor(collection: TCollection, parent: TParent) {
        this.collection = collection;
        this.parent = parent;
    }
}

class Model<TItem extends Item, TCollection extends Collection> {
    item: TItem;
    parent: TCollection;
    constructor(item: TItem, parent: TCollection) {
        this.item = item;
        this.parent = parent;
    }
}
