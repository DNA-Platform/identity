class $TestApp {
    x: any = 0;
    y: any = 0;
    name = "App";
    typename = "Application";
    documents: $TestDocument[] = [new $TestDocument("1", true), new $TestDocument("2", false)];
    activeDocument(): $TestDocument { return this.documents[0]; }
}

class $TestDocument {
    typename = "Document";
    layers: $TestLayer[] = [new $TestLayer("1", true), new $TestLayer("2", false)];
    activeLayer(): $TestLayer { return this.layers[0]; }
    get(index: number) { return this.layers[index]; }
    constructor(public name: string, public active: boolean) { }
}

class $TestLayer {
    constructor(public name: string, public active: boolean) { }
}

var $testApp = new $TestApp();

class $CommandTests extends TestClass {
    testConstructor() {
        var expression = "$testApp.x = $testApp.name;";
        var cmd = new $Command(expression);
        var actual = cmd.expression();
        var expected = expression;
        var m = this.assertEquals(actual, expected, "Command should store expression exactly", false); if (m) throw m;

        var ref = new $Reference("$testApp.name");
        var cmd = new $Command(ref);
        var actual = cmd.expression();
        var expected = "$testApp.name;";
        var m = this.assertEquals(actual, expected, "Command should add semicolon to references", false); if (m) throw m;
    }

    testAdd() {
        var expression = "$testApp.x = $testApp.name;";
        var cmd = new $Command(expression);
        var m = this.assertEquals(expression, cmd.expression(), "Command should store initial expression", false); if (m) throw m;

        var secondExpression = "$testApp.y = $testApp.x;";
        var newCmd = cmd.add(secondExpression);

        var actual = newCmd.expression();
        var expected = "$testApp.x = $testApp.name;\n$testApp.y = $testApp.x;";
        var m = this.assertEquals(actual, expected, "Add should combine expressions with newline", false); if (m) throw m;
    }

    testExecute() {
        var expression = "$testApp.x = $testApp.name;";
        var cmd = new $Command(expression);
        cmd.execute();
        
        var actual = $testApp.x;
        var expected = "App";
        var m = this.assertEquals(actual, expected, "Execute should run the command", false); if (m) throw m;
    }
}

class $PropertyReferenceTests extends TestClass {
    testAddress() {
        var obj = new $Reference("$testApp");

        var actual = ref.address();
        var expected = "$testApp.name";
        var m = this.assertEquals(actual, expected, "Property reference creates correct path", false); if (m) throw m;

        var actual = nested.address();
        var expected = "$testApp.documents[0].name";
        var m = this.assertEquals(actual, expected, "Property reference handles nested paths", false); if (m) throw m;
    }

    testObject() {
        var obj = new $Reference("$testApp");
        var ref = new $PropertyReference(obj, "name");
        var actual = ref.object().address();
        var expected = "$testApp";
        var m = this.assertEquals(actual, expected, "Property reference keeps object reference", false); if (m) throw m;
    }

    testResolve() {
        var obj = new $Reference("$testApp");
        var ref = new $PropertyReference(obj, "name");
        var actual = ref.$resolve();
        var expected = "App";
        var m = this.assertEquals(actual, expected, "Property reference.$resolves simple property", false); if (m) throw m;

        var nested = new $PropertyReference(obj, "documents[0].name");
        var actual = nested.$resolve();
        var expected = "1";
        var m = this.assertEquals(actual, expected, "Property reference.$resolves nested path", false); if (m) throw m;
    }
}
