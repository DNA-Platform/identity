class Diagram {
    chemical: $Chemical;
    bonds: Map<string, Bond> = new Map();
    get props() { return this.bonds.values().filter(b => b.isProp); }
    get chemicals() { return this.bonds.values().filter(b => b.isChemical); }
    constructor(chemical: $Chemical) {
        this.chemical = chemical;
        this.configureBonds()
    }

    configureBonds() {
        const prototypes: any[] = [];
        let proto = this.chemical;
        while (proto) {
            prototypes.unshift(proto);
            if (proto === $Chemical.prototype) break;
            proto = Object.getPrototypeOf(proto);
        }

        for (const p of prototypes) {
            for (const property in p) {
                const value = p[property];
                if (!isReactiveProperty(p, property, value)) continue;
                this.configureBond(property, value, p[decoratorsKey]);
            }
        }
    }

    configureBond(property: string, value: any, decorators?: DecoratorCache): Bond {
        let bond = this.bonds.get(property);
        if (!bond) {
            bond = new Bond(property);
            this.bonds.set(property, bond);
        }

        if (decorators?.inert.has(property)) {
            bond.isInert = true;
            return bond;
        }

        const parentType = decorators?.parent.get(property);
        if (parentType) {
            bond.type = parentType || $Chemical;
            bond.isChemical = true;
            bond.isInert = true;
            bond.isSettable = false; 
        }

        // Check for getter-only properties
        const descriptor = Object.getOwnPropertyDescriptor(this.chemical, property);
        if (descriptor?.get && !descriptor.set) {
            bond.isSettable = false;
            bond.symbolizer = symbolize;  // Getters always use symbolize
        }

        const representer = decorators?.dynamic.get(property);
        if (representer !== undefined) {
            bond.symbolizer = representer || symbolize;
            if (!bond.isChemical) bond.isMutable = true;
        }

        const $config = decorators?.$.get(property);
        if ($config) {
            if (bond.type && bond.type !== Object && bond.type !== $config.type)
                throw new Error(`${property} type conflict`);
            bond.type = $config.type;
            bond.isProp = property.startsWith('$');
            bond.isChemical = $config.type.prototype instanceof $Chemical;
            if (bond.isProp && (this.chemical as any)[property.substring(1)] === $Chemical.$set)
                bond.isExplicitlyBound = true;
            else if (bond.isChemical)
                bond.isImplicitlyBound = true;
            this.initializeBond(bond, $config.expecting, property, value);
        }

        // Auto-detect objects and collections from the value if not already set
        if (!bond.isChemical && typeof value === 'object' && !(value instanceof $Chemical)) {
            if (!bond.symbolizer) bond.symbolizer = symbolize;
            if (bond.isSettable) {
                bond.isValue = false;
                bond.isMutable = true;
            }
        }

        return bond;
    }

    private initializeBond(bond: Bond, expecting: '?' | '!' | '[]', property: string, value: any) {
        const chemical = this.chemical as any;
        if (expecting === '[]') {
            bond.isCollective = true;
            bond.isRequired = true;
            if (value && !Array.isArray(value)) throw new Error(`${property} marked array but isn't`);
            if (!value) chemical[property] = [];
            return;
        }
        bond.isRequired = expecting === '!';
        if (expecting === '?' && Array.isArray(value)) throw new Error(`${property} optional but is array`);
        if (bond.isRequired && bond.isChemical && !value) {
            const formula = (bond.type as any).formula || bond.type!.prototype;
            chemical[property] = Object.create(formula);
            chemical[property]._reactive = false;
        }
    }

    duplicate(): this {
        const diagram = Object.create(this) as this;
        const bonds: Map<string, Bond> = new Map();
        this.bonds.forEach((value, key) => bonds.set(key, value.duplicate()));
        diagram.bonds = bonds;
        return this;
    }
}

class Bond {
    private _id: number;
    get id() { return this._id; }
    type?: Type;  // Only for Chemical types, undefined otherwise
    property: string;
    isConfigured: boolean;
    
    isChemical = false;         // Property holds a chemical
    isMutable = false;          // Object/array that can change without reference change
    isValue = false;            // Has value semantics
    isCollective = false;       // Is a collection of chemicals 
    isSettable = true;          // Is a field, or a property with both a get and set method
    isRequired = false;         // Must not be null or undefinmed
    isInert = false;            // Never reactive
    isProp = false;             // Non-chemical property that Starts with $
    isDelayed = false           // Uses set-timeout to check if a value was changed after being accessed
    isImplicitlyBound = false;  // Chemical without setter
    isExplicitlyBound = false;  // Has $set setter
    symbolizer: (value: any) => any = (value) => value;
    
    private _isReactive = false;
    get isReactive() { return this.isInert ? false : this._isReactive; }
    set isReactive(value: boolean) { this._isReactive = value; }
    get isPolled() { return this.isMutable || !this.isSettable; }

    constructor(property: string, type?: Type) {
        this._id = Bond.getNextBondId();
        this.property = property;
        this.type = type;
        this.isConfigured = false;
        this.isReactive = false;
    }

    reactivate(chemical: $Chemical): any {
        if (chemical === chemical.getType().formula)
            throw new Error('Formulas should not have activated properties');
        // Imagine this is called on the instance of the class
        if (this.isConfigured && this.isReactive) return;
        if (this.isConfigured) {
            this.isReactive = true;
            return;
        }

        const thisBond = this;
        const chemical$: any = chemical;
        const backingField = Symbol(`_${this.property}${this.id}`);
        const getValue: () => any = () => { return chemical$[backingField]; }
        const setValue: (value: any) => void = (value) => { chemical$[backingField] = value; }
        setValue(chemical$[this.property]);

        const property = Object.defineProperty(chemical$, this.property, {
            get() {
                if (thisBond.isInert)
                    throw new Error(`${thisBond.type}.${chemical.chemicalId}.${thisBond.property} is inert`);

                const value = getValue();
                if (thisBond.isReactive && (thisBond.isPolled)) {
                    const symbol = thisBond.symbolizer(value);
                    setTimeout(() => {
                        if (!thisBond.isReactive) return;
                        const currentValue = getValue();
                        const currentSymbol = thisBond.symbolizer(currentValue);
                        if (symbol !== currentSymbol && !chemical$[updatingKey])
                            chemical.update();
                    });
                }
                return value;
            },
            set(value: any) {
                if (!thisBond.isSettable)
                    throw new Error(`${thisBond.type}.${chemical.chemicalId}.${thisBond.property} is not settable.`);
                if (thisBond.isInert)
                    throw new Error(`${thisBond.type}.${chemical.chemicalId}.${thisBond.property} is inert and should not be bound to a reactive property.`);
                if (!thisBond.isSettable)
                    throw new Error(`${thisBond.type}.${chemical.chemicalId}.${thisBond.property} is not settable.`)
                if (thisBond.isRequired && (value === null || value === undefined))
                    throw new Error(`${thisBond.type}.${chemical.chemicalId}.${thisBond.property} is required to have a value.`)
                if (thisBond.type)
                    thisBond.checkType(value);

                const previousValue = getValue();
                setValue(value);

                if (previousValue instanceof $Chemical) {
                    const previousUpdateFunctions: (() => void)[] = (previousValue as any)[updateFunctionsKey];
                    (previousValue as any)[updateFunctionsKey] = previousUpdateFunctions.filter(f => f !== this.update);
                }
                if (value instanceof $Chemical) {
                    const valueUpdateFunctions: (() => void)[] = (value as any)[updateFunctionsKey];
                    valueUpdateFunctions.push(this.update);
                }

                if (!thisBond.isReactive) return;
                if (chemical$[updatingKey]) return;

                const previousSymbol = thisBond.symbolizer(previousValue);
                const symbol = thisBond.symbolizer(value);
                if (!thisBond.equivalent(previousSymbol, symbol)) {
                    chemical$[updatingKey] = true;
                    setTimeout(() => {
                        chemical.update();
                    });
                }
            },
            enumerable: true,
        });

        this.isConfigured = true;
        this.isReactive = true;
        return property;
    }

    merge(value: any): void {
        if (value === null || value === undefined) return;
        
        if (value instanceof $Chemical) {
            if (this.isMutable) 
                throw new Error(`${this.property} was mutable, now Chemical`);
            this.isChemical = true;
            this.isValue = true;
            this.isMutable = false;
            this.symbolizer = (v) => v;
            this.type = value.constructor as Type;
            return;
        }
        
        const valueType = typeof value;
        if (valueType === 'string' || valueType === 'number' || valueType === 'boolean') {
            if (!this.isValue && !this.isMutable) {
                this.isValue = true;
                this.symbolizer = (v) => v;
            }
            return;
        }
        
        if (valueType === 'object') {
            if (this.isCollective && !Array.isArray(value))
                throw new Error(`${this.property} is collective but got non-array`);
            if (!this.isCollective && Array.isArray(value) && this.isRequired)
                throw new Error(`${this.property} got unexpected array`);
            
            if (!this.isMutable) {
                this.isMutable = true;
                this.isValue = false;
                this.symbolizer = symbolize;
            }
        }
    }

    equivalent(first: any, second: any) {
        return this.symbolizer(first) === this.symbolizer(second);
    }

    duplicate(): this {
        const double = Object.create(this);
        double._id = Bond.getNextBondId();
        return double;
    }

    private checkType(value: any): void {
        if (!this.type || value === null || value === undefined) return;

        if (this.isChemical) {
            if (!(value instanceof this.type))
                throw new Error(`${this.property} must be an instance of ${this.type.name}, got ${value.constructor.name}`);
            return;
        }

        const valueType = typeof value;
        if (this.type === String && valueType !== 'string')
            throw new Error(`${this.property} must be a string, got ${valueType}`);
        else if (this.type === Number && valueType !== 'number')
            throw new Error(`${this.property} must be a number, got ${valueType}`);
        else if (this.type === Boolean && valueType !== 'boolean')
            throw new Error(`${this.property} must be a boolean, got ${valueType}`);
        else if (this.type === Array && !Array.isArray(value))
            throw new Error(`${this.property} must be an array`);
        else if (this.type === Object && valueType !== 'object')
            throw new Error(`${this.property} must be an object, got ${valueType}`);
    }

    private static nextBondId = 1;
    private static getNextBondId(): number { return Bond.nextBondId++; }
}
