class Bond {
    private _id: number;
    get id() { return this._id; }
    type?: Type;  // Only for Chemical types, undefined otherwise
    property: string;
    isConfigured: boolean;
    
    isChemical = false;      // Property holds a Chemical instance
    isMutable = false;        // Object/array that can change without reference change
    isComparable = false;     // Has value semantics (primitives, immutables)
    isSettable = true;        // Can be set (false for getter-only)
    isRequired = false;       // Must have a value
    isInert = false;          // Never reactive
    isProp = false;           // Starts with $
    isImplicitlyBound = false;  // Chemical without setter
    isExplicitlyBound = false;  // Has $set setter
    symbolizer: (value: any) => any = (value) => value;
    
    private _isReactive = false;
    get isReactive() { return this.isInert ? false : this._isReactive; }
    set isReactive(value: boolean) { this._isReactive = value; }
    get isPolled() { return this.isMutable || !this.isSettable; }

    constructor(property: string, type?: Type) {
        this._id = Bond.getNextBondId();
        this.property = property;
        this.type = type;
        this.isConfigured = false;
        this.isReactive = false;
    }

    reactivate(chemical: $Chemical): any {
        if (chemical === chemical.getType().formula)
            throw new Error('Formulas should not have activated properties');
        // Imagine this is called on the instance of the class
        if (this.isConfigured && this.isReactive) return;
        if (this.isConfigured) {
            this.isReactive = true;
            return;
        }

        const thisBond = this;
        const chemical$: any = chemical;
        const backingField = Symbol(`_${this.property}${this.id}`);
        const getValue: () => any = () => { return chemical$[backingField]; }
        const setValue: (value: any) => void = (value) => { chemical$[backingField] = value; }
        setValue(chemical$[this.property]);

        const property = Object.defineProperty(chemical$, this.property, {
            get() {
                if (thisBond.isInert)
                    throw new Error(`${thisBond.type}.${chemical.chemicalId}.${thisBond.property} is inert`);

                const value = getValue();
                if (thisBond.isReactive && (thisBond.isPolled)) {
                    const symbol = thisBond.symbolizer(value);
                    setTimeout(() => {
                        if (!thisBond.isReactive) return;
                        const currentValue = getValue();
                        const currentSymbol = thisBond.symbolizer(currentValue);
                        if (symbol !== currentSymbol && !chemical$[updatingKey])
                            chemical.update();
                    });
                }
                return value;
            },
            set(value: any) {
                if (!thisBond.isSettable)
                    throw new Error(`${thisBond.type}.${chemical.chemicalId}.${thisBond.property} is not settable.`);
                if (thisBond.isInert)
                    throw new Error(`${thisBond.type}.${chemical.chemicalId}.${thisBond.property} is inert and should not be bound to a reactive property.`);
                if (!thisBond.isSettable)
                    throw new Error(`${thisBond.type}.${chemical.chemicalId}.${thisBond.property} is not settable.`)
                if (thisBond.isRequired && (value === null || value === undefined))
                    throw new Error(`${thisBond.type}.${chemical.chemicalId}.${thisBond.property} is required to have a value.`)
                if (thisBond.type)
                    thisBond.checkType(value);

                const previousValue = getValue();
                setValue(value);

                if (previousValue instanceof $Chemical) {
                    const previousUpdateFunctions: (() => void)[] = (previousValue as any)[updateFunctionsKey];
                    (previousValue as any)[updateFunctionsKey] = previousUpdateFunctions.filter(f => f !== this.update);
                }
                if (value instanceof $Chemical) {
                    const valueUpdateFunctions: (() => void)[] = (value as any)[updateFunctionsKey];
                    valueUpdateFunctions.push(this.update);
                }

                if (!thisBond.isReactive) return;
                if (chemical$[updatingKey]) return;

                const previousSymbol = thisBond.symbolizer(previousValue);
                const symbol = thisBond.symbolizer(value);
                if (!thisBond.equivalent(previousSymbol, symbol)) {
                    chemical$[updatingKey] = true;
                    setTimeout(() => {
                        chemical.update();
                    });
                }
            },
            enumerable: true,
        });

        this.isConfigured = true;
        this.isReactive = true;
        return property;
    }

    equivalent(first: any, second: any) {
        return this.symbolizer(first) === this.symbolizer(second);
    }

    duplicate(): this {
        const double = Object.create(this);
        double._id = Bond.getNextBondId();
        return double;
    }

    private checkType(value: any): void {
        if (!this.type || value === null || value === undefined) return;

        if (this.isChemical) {
            if (!(value instanceof this.type))
                throw new Error(`${this.property} must be an instance of ${this.type.name}, got ${value.constructor.name}`);
            return;
        }

        const valueType = typeof value;
        if (this.type === String && valueType !== 'string')
            throw new Error(`${this.property} must be a string, got ${valueType}`);
        else if (this.type === Number && valueType !== 'number')
            throw new Error(`${this.property} must be a number, got ${valueType}`);
        else if (this.type === Boolean && valueType !== 'boolean')
            throw new Error(`${this.property} must be a boolean, got ${valueType}`);
        else if (this.type === Array && !Array.isArray(value))
            throw new Error(`${this.property} must be an array`);
        else if (this.type === Object && valueType !== 'object')
            throw new Error(`${this.property} must be an object, got ${valueType}`);
    }

    private static nextBondId = 1;
    private static getNextBondId(): number { return Bond.nextBondId++; }
}
