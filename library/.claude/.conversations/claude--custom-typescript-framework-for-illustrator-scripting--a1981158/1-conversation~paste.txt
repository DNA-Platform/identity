// Sends a command to be evaluated in another context
function send(cmd: string,  onComplete?: OnCompleteHandler, onError?: OnErrorHandler, index: number = 0): void {
    var message = $BridgeTalkMessage();
    message.target = BridgeTalkApps.illustrator;
    message.body = `${cmd}; app.redraw();`;
    
    if (onComplete) message.onResult = () => onComplete(cmd, index);
    if (onError) message.onError = e => onError(cmd, index, e);
    
    message.send();
}

// Sends a script to be evaluated in another context
function sendScript(scriptPath: string, onComplete?: OnCompleteHandler, onError?: OnErrorHandler, index: number = 0) {
    scriptPath = combinePaths(SCRIPTS_PATH, scriptPath);
    send(`$.evalFile("${scriptPath}");`, onComplete, onError, index);
}

class Console {
    private _on: boolean = false;
    private _alert: boolean = true;
    on(type: "alert" | "terminal" = "alert"): this {
        this._on = true;
        this._alert = type == "alert";
        return this;
    }
    off(): this { this._on = false; return this; }
    alert(): this { this._alert = true; return this; }
    terminal(): this { this._alert = false; return this; }

    active(): boolean { return this._on; }
    alertType(): "alert" | "terminal" { return this._alert ? "alert" : "terminal"; }
    state(newState?: { active: boolean, type: "alert" | "terminal" }): { active: boolean, type: "alert" | "terminal" } {
        if (newState !== undefined) {
            this._on = newState.active;
            this._alert = newState.type === "alert" ? true : false;
        }
        return { active: this.active(), type: this.alertType() };
    }

    log(...args: any[]) {
        args = args ? args : [];
        this.show(false, ...args);
    }

    break(...args: any[]) {
        args = args ? args : [];
        this.show(true, ...args);
    }

    private show(halt: boolean, ...args: any[]) {
        if (!this._on) return;
        args = args ? args : [];
        args = f(args).select(a => typeof a === "string" ? `"${a}"` : a).a();
        var result = "";
        var arr = f(args).select(s => `${s}`);
        arr.do((s, i) => {
            i = i as number;
            result += `${s}\n`;
        })
        if (halt) {
            throw result;
        } else {
            this._alert ? alert(result) : $.write(result);
        }
    }
}

/**
 * Utility function to join an array of strings with a specified separator
 * @param array - The array of strings to join
 * @param separator - The separator to use between strings
 * @returns A single string formed by joining the array elements
 */
function stringJoin(array: string[] | A<string>, separator: string): Str {
    if (array instanceof Str) return array;
    if (array instanceof A) return stringJoin(array.a(), separator);
    var result = "";
    for (var i = 0; i < array.length; i++) {
        result += array[i];
        if (i < array.length - 1) {
            result += separator; // Add separator only between elements
        }
    }
    return new Str(result);
}

/**
 * Utility function to compare two arrays for equality
 * @param first - The first array to compare
 * @param second - The second array to compare
 * @returns True if the arrays are equal (same length and elements), otherwise false
 */
function arrayEquals(first: any[], second: any[]): boolean {
    if (!first && !second) return true;
    if (!first || !second) return false;
    if (first.length !== second.length) return false;

    for (var i = 0; i < first.length; i++) {
        var currentFirst = first[i];
        var currentSecond = second[i];
        var areEqual = (currentFirst instanceof Array && currentSecond instanceof Array) ?
            arrayEquals(currentFirst, currentSecond) :
            currentFirst === currentSecond;
        if (!areEqual) 
            return false;
    }

    return true;
}

/**
 * Deep clones an object in ExtendScript, handling arrays, objects, and common edge cases.
 * @param source - The object to clone.
 * @param seen - A record of already-seen objects to handle circular references.
 * @returns The deep-cloned object.
 */
function clone<T>(obj: T, primitivesOnly: boolean = false, seen?: A<[any, any]>): T {
    var source: any = obj;
    if (!seen) seen = f([]);

    // Check if the value is a primitive
    function isPrimitive(value: any): boolean {
        return value === null ||
            value === undefined ||
            Object.prototype.toString.call(value) === "[object Boolean]" ||
            Object.prototype.toString.call(value) === "[object Number]" ||
            Object.prototype.toString.call(value) === "[object String]" ||
            Object.prototype.toString.call(value) === "[object Function]";
    }

    // Handle primitives and primitive-only mode
    if (isPrimitive(source)) {
        return source;
    }
    if (primitivesOnly) {
        return undefined as T;
    }

    // Detect arrays
    if (Object.prototype.toString.call(source) === "[object Array]") {
        const copy: any[] = [];
        for (var i = 0; i < source.length; i++) {
            var currentSource = source[i];
            if (seen.any(v => v[0] == currentSource)) {
                copy[i] = seen.first(v => v[0] == currentSource)[1];
            } else {
                seen.push([currentSource, currentSource]);
                var clonedValue = clone(currentSource, primitivesOnly, seen)
                seen.at(-1)[1] = clonedValue;
                copy[i] = clonedValue;
            }
        }
        return copy as T;
    }

    // Handle objects (functions are included as properties)
    const copy: any = {};
    for (var key in source) {
        if (Object.prototype.hasOwnProperty.call(source, key) && key !== "parent") {
            var value = source[key];

            if (typeof value === "function") {
                // Rebind function to the cloned object
                copy[key] = function (...args: any[]) {
                    return value.apply(copy, args); // Bind `this` to the cloned object
                };
            } else {
                // Recursively clone other properties
                if (seen.any(v => v[0] == value)) {
                    copy[key] = seen.first(v => v[0] == value)[1];
                } else {
                    seen.push([value, value]);
                    var clonedValue = clone(value, primitivesOnly, seen)
                    seen.at(-1)[1] = clonedValue;
                    copy[key] = clonedValue;
                }
            }
        }
    }

    return copy as T;
}

/**
 * Rounds a number to a specific number of decimals
 * @param value - The number to round
 * @param decimals - The number of decimals of precision
 * @returns The rounded value
 */
function roundPrecisely(value: number, decimals: number = 6): number {
    var factor = Math.pow(10, decimals);
    return Math.round(value * factor) / factor;
}

/**
 * Normalizes file path separators to Windows style (backslashes)
 * and ensures no double separators
 * @param path - The file path to normalize
 * @returns Normalized path with proper separators
 */
function normalizePath(path: string): string {
    return str(path).replaceAll(/[\/\\]+/g, '\\').s();
}

/**
 * Combines two paths, ensuring proper separator handling
 * @param path1 - First path
 * @param path2 - Second path
 * @returns Combined path with proper separators
 *
 * @example
 * // Returns "C:\\Scripts\\documents\\test.ai"
 * combinePaths("C:\\Scripts", "documents/test.ai")
 *
 * @example
 * // Returns "C:\\Scripts\\documents\\test.ai"
 * combinePaths("C:\\Scripts\\", "\\documents\\test.ai")
 */
function combinePaths(path1: string, path2: string): string {
    var normalized1 = normalizePath(path1);
    var normalized2 = normalizePath(path2);

    // Remove trailing slash from first path and leading slash from second path
    var clean1 = str(normalized1).endsWith('\\') ? normalized1.slice(0, -1) : normalized1;
    var clean2 = str(normalized2).startsWith('\\') ? normalized2.slice(1) : normalized2;

    return clean1 + '\\' + clean2;
}

/**
 * Converts a negative index to a positive index within the bounds of the collection length.
 *
 * @param index - The index to evaluate (can be negative)
 * @param collectionLength - The length of the collection
 * @returns A valid positive index within the collection bounds
 * @throws Error if the evaluated index is out of bounds
 */
function evaluateNegativeIndex(index: number, collectionLength: number): number {
    var evaluatedIndex = index < 0 ? collectionLength + index : index;
    if (evaluatedIndex < 0 || evaluatedIndex >= collectionLength) {
        throw "Index " + index + " out of bounds for collection of length " + collectionLength;
    }
    return evaluatedIndex;
}

/**
 * Converts a negative index to a positive index within the bounds of the collection length but maps -1 to the length of the array so the item gets inserted at the end.
 *
 * @param index - The index to evaluate (can be negative)
 * @param collectionLength - The length of the collection
 * @returns A valid positive index within the collection bounds
 * @throws Error if the evaluated index is out of bounds
 */
function evaluateNegativeInsertIndex(index: number, collectionLength: number): number {
    var evaluatedIndex = index < 0 ? collectionLength + index + 1 : index;
    if (evaluatedIndex < 0 || evaluatedIndex > collectionLength) {
        throw "Index " + index + " out of insert bounds for collection of length " + collectionLength;
    }
    return evaluatedIndex;
}

function evaluateRangeSelector(selector?: string, collectionLength?: number): number[] {
    if (!selector) return []; // Empty selector
    selector = selector.replace(/\s+/g, "");

    if (collectionLength === undefined && selector === ":")
        return [];

    var start: number, step: number | undefined, end: number;

    if (selector === ":") {
        start = 0;
        end = -1;
        step = undefined;
    }
    // Single number (e.g., "5")
    else if (/^-?\d+$/.test(selector)) {
        start = end = parseInt(selector, 10);
        step = undefined;
    }
    // Start:end (e.g., "1:5")
    else if (/^-?\d+:-?\d+$/.test(selector)) {
        var parts = selector.split(":");
        start = parseInt(parts[0], 10);
        end = parseInt(parts[1], 10);
        step = undefined;
    }
    // Start:step:end (e.g., "1:2:5")
    else if (/^-?\d+:-?\d+:-?\d+$/.test(selector)) {
        var parts = selector.split(":");
        start = parseInt(parts[0], 10);
        step = parseInt(parts[1], 10);
        end = parseInt(parts[2], 10);
    }
    // :end (e.g., ":5")
    else if (/^:-?\d+$/.test(selector)) {
        start = 0;
        end = parseInt(selector.slice(1), 10);
        step = undefined;
    }
    // start: (e.g., "5:")
    else if (/^-?\d+:$/.test(selector)) {
        if (collectionLength === undefined)
            throw new Error(`Invalid range selector: "${selector}"`);
        var parts = selector.split(":");
        start = parseInt(parts[0], 10);
        end = collectionLength !== undefined ? collectionLength - 1 : -1;
        step = undefined;
    }
    // :step:end (e.g., ":2:5")
    else if (/^:-?\d+:-?\d+$/.test(selector)) {
        var parts = selector.slice(1).split(":");
        start = 0;
        step = parseInt(parts[0], 10);
        end = parseInt(parts[1], 10);
    }
    // Invalid cases
    else {
        throw new Error(`Invalid range selector: "${selector}"`);
    }

    // Adjust for negative indices if collection length is defined
    if (collectionLength !== undefined) {
        start = evaluateNegativeIndex(start, collectionLength);
        end = evaluateNegativeIndex(end, collectionLength);
    }

    // Validate step direction
    step = step ?? 1;
    if (step <= 0) {
        throw new Error(`Invalid step value in the range: selector: "${selector}", start: "${start}", step: "${step}", end: "${end}"`);
    }

    // Generate the range
    var range: number[] = [];
    if (start < end) {
        for (var i = start; i <= end; i += step) range.push(i);
    } else {
        for (var i = start; i >= end; i -= step) range.push(i);
    }

    return range;
}

class A<T> {

        return this.str().s();

    }

        return this.any(i => i == item);
    }

    }

    }

    pop(): T | undefined {

    }

    splice(start: number, deleteCount?: number, ...items: T[]): A<T> {

    }

        return new A(this.a().slice(start, end));

    }

    }

    reverse(): A<T> {
        var arr = this.a();
        var reversed: T[] = [];
        for (var i = arr.length - 1; i >= 0; i--) {
            reversed.push(arr[i]);
        }
        return new A<T>(reversed);

    skip(count: number): A<T> {
        if (count >= this.length()) return f([]);
        return this.get(`${count}:`)
    }

    take(count: number): A<T> {
        if (this.length() == 0) return f([]);
        count = Math.min(count, this.length());
        return this.get(`:${count - 1}`);
    }

    zip<U>(items: A<U>): A<[T, U]> {
        var len = this.length();
        if (len !== items.length())
            throw "Arrays must be the same length to be zipped";

        var result: A<[T, U]> = f([]);
        for (var i = 0; i < len; i++) {
            result.push([this.at(i), items.at(i)]);
        }

        return result;

        var items: [T, T][] = [];

                items.push([self[i], self[j]]);

        return new A(items);

        var array = this.copy().a();
        for (var i = array.length - 1; i > 0; i--) {
            var j = Math.floor(Math.random() * (i + 1)); // Random index from 0 to i
            [array[i], array[j]] = [array[j], array[i]]; // Swap elements
        }
        return new A<T>(array);
    }

    vector(): Vector {
        if (this.length() === 0) return v();
        if (VectorTypes.isNumberArrayExtension(this)) return v(this);
        if (VectorTypes.isPositionArrayExtension(this)) return v(this);
        throw new Error("The vector function only works on number and position arrays");
    }

        return stringJoin(this.select(x => x + "").a(), separator);
    }

    toString(): string {
        if (this.length() == 0) return "A[]";

            return `A<${typeof this.at(0)}>[${this.select(x => `${x}`).a()}]`;
        } catch {
            return `A<${typeof this.at(0)}>[${this.a()}]`;
        }
    }

    private static getTypedComparer<T>(arr: T[], selector?: (item: T) => any, desc: boolean = false): (a: T, b: T) => number {
        if (arr.length === 0) return () => 0;
        var getValue = selector || ((x: T) => x);
        var first = getValue(arr[0]);
        var type = typeof first;

        if (type === 'number' || type === 'boolean')
            return desc
                ? (a, b) => getValue(b) - getValue(a)
                : (a, b) => getValue(a) - getValue(b);

        if (type === 'string')
            return desc
                ? (a, b) => getValue(b).toString().localeCompare(getValue(a))
                : (a, b) => getValue(a).toString().localeCompare(getValue(b));

        if (first instanceof Date)
            return desc
                ? (a, b) => getValue(b).getTime() - getValue(a).getTime()
                : (a, b) => getValue(a).getTime() - getValue(b).getTime();

        return desc
            ? (a, b) => getValue(a) < getValue(b) ? 1 : getValue(a) > getValue(b) ? -1 : 0
            : (a, b) => getValue(a) < getValue(b) ? -1 : getValue(a) > getValue(b) ? 1 : 0;
    }
}

class Str extends A<string> {
    a(): string[] {

    /**
     * Utility function to check if a string starts with a specific prefix
     * @param str - The main string to check
     * @param prefix - The prefix to look for
     * @returns True if the string starts with the prefix, otherwise false
     */
    static stringStartsWith(str: string, prefix: string): boolean {
        return str.slice(0, prefix.length) === prefix;
    }

    /**
     * Utility function to check if a string ends with a specific suffix
     * @param str - The main string to check
     * @param suffix - The suffix to look for
     * @returns True if the string ends with the suffix, otherwise false
     */
    static stringEndsWith(str: string, suffix: string): boolean {
        return str.slice(-suffix.length) === suffix;
    }

    /**
     * Utility function to replace parts of a string using a pattern or substring
     * @param str - The original string
     * @param pattern - The pattern (string or RegExp) to replace
     * @param replacement - The string to replace the pattern with
     * @returns The resulting string after the replacement
     */
    static stringReplace(str: string, pattern: string | RegExp, replacement: string): string {
        if (typeof pattern === "string") {
            // Replace all occurrences of the string pattern
            while (str.indexOf(pattern) !== -1) {
                var index = str.indexOf(pattern);
                str = str.slice(0, index) + replacement + str.slice(index + pattern.length);
            }
            return str;
        } else if (pattern instanceof RegExp) {
            // Replace all occurrences of the regex pattern
            var result = "";
            var match;
            while ((match = pattern.exec(str)) !== null) {
                result += str.slice(0, match.index) + replacement;
                str = str.slice(match.index + match[0].length);
            }
            return result + str;
        }
        throw "Invalid pattern type. Must be a string or RegExp.";
    }

    /**
     * Utility function to check if a string contains a specific substring
     * @param str - The main string to check
     * @param search - The substring to look for
     * @returns True if the string contains the substring, otherwise false
     */
    static stringIncludes(str: string, search: string): boolean { return Str.stringContains(str, search) }
    static stringContains(str: string, search: string): boolean {
        if (str === undefined || str === null) return false;
        if (search === "") return true;

        var index = 0;
        var len = str.length;
        var searchLen = search.length;

        while (index + searchLen <= len) {
            if (str.slice(index, index + searchLen) === search) {
                return true;
            }
            index++;
        }
        return false;
    }

    /**
     * Utility function to pad the start of a string with a specific character
     * @param str - The main string to pad
     * @param targetLength - The desired total length of the output string
     * @param padChar - The character to use for padding (default is ' ')
     * @returns The padded string
     */
    static stringPadStart(str: string, targetLength: number, padChar: string = " "): string {
        str = String(str);
        if (padChar.length === 0) padChar = " ";
        var padding = "";
        var padLength = Math.max(targetLength - str.length, 0);

        while (padding.length < padLength) {
            padding += padChar;
        }

        return padding.slice(0, padLength) + str;
    }

    /**
     * Utility function to pad the end of a string with a specific character
     * @param str - The main string to pad
     * @param targetLength - The desired total length of the output string
     * @param padChar - The character to use for padding (default is ' ')
     * @returns The padded string
     */
    static stringPadEnd(str: string, targetLength: number, padChar: string = " "): string {
        str = String(str);
        if (padChar.length === 0) padChar = " ";
        var padding = "";
        var padLength = Math.max(targetLength - str.length, 0);

        while (padding.length < padLength) {
            padding += padChar;
        }

        return str + padding.slice(0, padLength);
    }

function f<T>(array: T[] | A<T> = A.empty()): A<T> {
    if (array instanceof A) return array as A<T>;
    return new A(array);
}

function r(selector: string | number, length?: number): A<number> {
    if (typeof selector === "number") selector = `0:${selector}`;
    return f(evaluateRangeSelector(selector, length));
}

function str(s: Str | string | A<string> = ""): Str {
    if (s instanceof Str) return s;
    else if (typeof s === "string") return new Str(s);
    else if (s instanceof A) return s.stringJoin();
    return (s as any || "").toString();
}

class Subscribable<TSender, TContext = undefined> {
    private _handlers: A<(sender: TSender, context?: TContext) => void> = f([]);
    private _off: boolean = false;

    active(): boolean { return !this.off; }
    on(): this { this._off = false; return this; }
    off(): this { this._off = true; return this; }

    handlers() { return this._handlers; }

    constructor() {
        this.reset();
    }

    subscribe(handler: (sender: TSender, context?: TContext) => void): void {
        this._handlers.push(handler);
    }

    unsubscribe(handler: (sender: TSender, context?: TContext) => void): void {
        this._handlers.delete(handler);
    }

    fire(sender: TSender, context?: TContext): void {
        if (this._off) return;
        this._handlers.do((handler) => handler(sender, context));
    }

    reset() {
        this._handlers = new A<(sender: TSender, context?: TContext) => void>();
    }
}

class TestClass {
    tolerance: number = 0.0001

    init() { }

    runTests(): void {
        trace(this);
        var self = this as any;
        for (var key in self) {
            if (str(key).startsWith("test") && typeof self[key] === "function") {
                this.init();
                self[key]();
            }
        }
    }

    assertEquals<T>(actual: T, expected: T, message: string = "", shouldThrow: boolean = true): string | undefined {
        if (typeof actual === "number" && typeof expected === "number") {
            // Compare numbers within tolerance
            if (Math.abs(actual - expected) > this.tolerance) {
                var message = this.formatError(message, actual, expected, "==");
                if (shouldThrow) throw message;
                return message; 
            }
        } else if (actual instanceof Array && expected instanceof Array) {
            // Compare arrays deeply
            if (!this.arrayEquals(actual, expected)) {
                var message = this.formatError(message, actual, expected, "==");
                if (shouldThrow) throw message;
                return message; 
            }
        } else if (actual instanceof A && expected instanceof A) {
            // Compare wrapped arrays deeply
            if (!this.arrayEquals(actual.a(), expected.a())) {
                var message = this.formatError(message, actual, expected, "!=");
                if (shouldThrow) throw message;
                return message; 
            }
        } else if (actual !== expected) { // Use '!=' for type coercion if necessary
            var message = this.formatError(message, actual, expected, "==");
            if (shouldThrow) throw message;
                return message; 
        }
    }

    assertNotEqual<T>(actual: T, expected: T, message: string = "", shouldThrow: boolean = true): string | undefined {
        if (typeof actual === "number" && typeof expected === "number") {
            // Compare numbers within tolerance
            if (Math.abs(actual - expected) <= this.tolerance) {
                var message = this.formatError(message, actual, expected, "!=");
                if (shouldThrow) throw message;
                return message; 
            }
        } else if (actual instanceof Array && expected instanceof Array) {
            // Compare arrays deeply
            if (this.arrayEquals(actual, expected)) {
                var message = this.formatError(message, actual, expected, "!=");
                if (shouldThrow) throw message;
                return message; 
            }
        } else if (actual instanceof A && expected instanceof A) {
            // Compare wrapped arrays deeply
            if (this.arrayEquals(actual.a(), expected.a())) {
                var message = this.formatError(message, actual, expected, "!=");
                if (shouldThrow) throw message;
                return message; 
            }
        } else if (actual === expected) { // Use '!=' for type coercion if necessary
            var message = this.formatError(message, actual, expected, "!=");
            if (shouldThrow) throw message;
                return message; 
        }
    }

    assertTrue(actual: boolean, message: string = "", shouldThrow: boolean = true): string | undefined {
        return this.assertEquals(actual, true, message, shouldThrow);
    }

    assertFalse(actual: boolean, message: string = "", shouldThrow: boolean = true): string | undefined {
        return this.assertEquals(actual, false, message, shouldThrow);
    }

    assertDefined(value: any, message: string = "", shouldThrow: boolean = true): string | undefined {
        return this.assertTrue(value != null && value != undefined, message, shouldThrow);
    }

    assertNullOrUndefined(value: any, message: string = "", shouldThrow: boolean = true): string | undefined {
        return this.assertTrue(value == null || value == undefined, message, shouldThrow);
    }

    assertThrows(test: () => void, message: string, shouldThrow: boolean = true): string | undefined {
        try {
            test()
        } catch {
            return;
        }
        if (shouldThrow) throw new Error(message);
        return message;
    }

    formatError(message: string, actual: any, expected: any, equals: "==" | "!="): string {
        var formattedMessage = message ? `${message}  :  ` : "";
        return equals === "==" ? 
            `${formattedMessage}[Expected: ${this.toString(expected)}, but got: ${this.toString(actual)}]` : 
            `${formattedMessage}[Expected: ${this.toString(expected)} and ${this.toString(actual)} to be different]`
    }

    toString(value: any): string {
        if (value instanceof Array) {
            return "[" + f(value).select(x => x?.toString()).stringJoin(", ") + "]";
        } else if (typeof value === "object" && value !== null) {
            var result = [];
            for (var key in value) {
                if (value.hasOwnProperty(key)) {
                    result.push(key + ": " + value[key]?.toString());
                }
            }
            return "{" + result.join(", ") + "}";
        } else {
            return value !== undefined ? value.toString() : "undefined";
        }
    }

    private arrayEquals(first: any[], second: any[]): boolean {
        if (!first && !second) return true;
        if (!first || !second) return false;
        if (first.length !== second.length) return false;
    
        for (var i = 0; i < first.length; i++) {
            var currentFirst = first[i];
            var currentSecond = second[i];
            var areEqual = 
                (typeof currentFirst === "number" && typeof currentSecond === "number") ? 
                    Math.abs(currentFirst - currentSecond) < this.tolerance :
                (currentFirst instanceof Array && currentSecond instanceof Array) ?
                    this.arrayEquals(currentFirst, currentSecond) :
                    currentFirst === currentSecond;
            if (!areEqual) 
                return false;
        }
    
        return true;
    }
}
