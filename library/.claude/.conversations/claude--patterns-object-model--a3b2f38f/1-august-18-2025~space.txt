
type RegionBounds = [left: number, right: number, bottom: number, top: number]

class Region {
    private _bounds: RegionBounds;
    private _boundsParent: RegionBounds;
    protected _parent: () => Region;
    private _isBase: boolean;
    private _onChange: Subscribable<Region> = new Subscribable<Region>();

    parent(): Region { return this.parent !== undefined ? (this._parent as any)() : undefined as any; }
    base(): Region { return this.isBase() ? this : this.parent().base(); }
    isBase(): boolean { return this._isBase; }
    onChange(): Subscribable<Region> { return this._onChange; }

    unit(): number { return this.rightParent() - this.leftParent(); }
    unitAbs(): number { return this.base().unit(); }

    left() { return this._bounds[0]; }
    leftParent(): number { return this._boundsParent[0]; }
    leftAbs(): number { return this.base().leftParent(); }

    right() { return this._bounds[1]; }
    rightParent(): number { return this._boundsParent[1]; }
    rightAbs(): number { return this.base().rightParent(); }

    bottom() { return this._bounds[2]; }
    bottomParent(): number { return this._boundsParent[2]; }
    bottomAbs(): number { return this.base().bottomParent(); }

    top() { return this._bounds[3]; }
    topParent(): number { return this._boundsParent[3]; }
    topAbs(): number { return this.base().topParent(); }
    
    bounds(): RegionBounds { return clone(this._bounds); }
    boundsParent(): RegionBounds { return clone(this._boundsParent); }
    boundsAbs(): RegionBounds { return this.base().boundsParent(); }

    rect(): Rect { return [this.leftParent(), this.topParent(), this.rightParent(), this.bottomParent()]; }
    rectAbs(): Rect { return this.base().rect(); }

    centerParent(): Position { return [ (this.leftParent() + this.rightParent()) * 0.5, (this.bottomParent() + this.topParent()) * 0.5] }
    centerAbs(): Position { return this.base().centerParent(); }
    center(newCenter?: Position): Position { 
        if (newCenter) {
            this.moveTo(newCenter);
            this.onChange().fire(this);
        }
        return [ (this.left() + this.right()) * 0.5, (this.bottom() + this.top()) * 0.5] 
    }

    widthParent(): number { return this.rightParent() - this.leftParent(); }
    widthAbs(): number { return this.base().widthParent(); }
    width(newWidth?: number, proportional = true) {
        if (newWidth) {
            var scaleFactor = newWidth / this.width();
            this.size([newWidth, proportional ? this.height() * scaleFactor : this.height()]);
            this.onChange().fire(this);
        }
        return this.right() - this.left();
    }
    
    heightParent(): number { return this.topParent() - this.bottomParent(); }
    heightAbs(): number { return this.base().heightParent(); }
    height(newHeight?: number, proportional = true) {
        if (newHeight) {
            var scaleFactor = newHeight / this.height();
            this.size([proportional ? this.width() * scaleFactor : this.width(), newHeight]);
            this.onChange().fire(this);
        }
        return this.top() - this.bottom();
    }

    sizeParent(): Size { return [this.widthParent(), this.heightParent()]; }
    sizeAbs(): Size { return this.base().sizeParent(); }
    size(newSize?: Size): Size {
        if (newSize) {
            var center = this.centerParent();
            var newWidth = this.toParentUnits(newSize[0]);
            var newHeight = this.toParentUnits(newSize[1]);
            this._boundsParent = [
                center[0] - newWidth * 0.5,  // left
                center[0] + newWidth * 0.5,  // right
                center[1] - newHeight * 0.5, // bottom
                center[1] + newHeight * 0.5  // top
            ];
            this._bounds = this.toRelativeBounds(this._boundsParent);
            this.onChange().fire(this);
        }
        return [this.width(), this.height()];
    }

    constructor(bounds: RegionBounds, parent?: () => Region) {
        this._boundsParent = bounds;
        this._isBase = parent === undefined;
        this._parent = parent !== undefined ? parent : () => undefined as any;

        if (this._boundsParent[0] > this._boundsParent[1] || this._boundsParent[2] > this._boundsParent[3]) 
            throw `Invalid Region: [${bounds}]`;
        if (parent) 
            parent().onChange().subscribe(() => this.onChange().fire(this));

        this._bounds = this.toRelativeBounds(this._boundsParent);
    }

    equals(region: Region): boolean {
        return f(this._boundsParent).equals(region._boundsParent);
    }

    floor(): Region {
        if (!this.isBase())
            throw new Error("Only the base region needs to be floored");

        this._boundsParent = f(this._boundsParent).select(b => Math.floor(b)).a() as RegionBounds;
        this._bounds = this.toRelativeBounds(this._boundsParent);
        this.onChange().fire(this);
        return this;
    }

    moveTo(newCenter: Position): Region {
        var newCenterParent = this.toParentPosition(newCenter);
        var deltaX = newCenterParent[0] - this.centerParent()[0];
        var deltaY = newCenterParent[1] - this.centerParent()[1];
        this._boundsParent = [
            this._boundsParent[0] + deltaX,
            this._boundsParent[1] + deltaX,
            this._boundsParent[2] + deltaY,
            this._boundsParent[3] + deltaY
        ];
        this._bounds = this.toRelativeBounds(this._boundsParent);
        this.onChange().fire(this);
        return this;
    }

    // Creates a region in units of the current region
    createRegion(bounds: RegionBounds) {
        var bounds = this.toParentBounds(bounds);
        return new Region(bounds);
    }

    createRegionRect(rect: Rect) {
        return this.createRegion(Region.rectToBounds(rect));
    }

    relativeTo(region: Region): RegionBounds {
        var boundsParent = this.boundsParent();
        return region.toRelativeBounds(boundsParent);
    }

    toRelativeUnits(parentLength: number): number {
        return parentLength / this.widthParent();
    }

    toParentUnits(length: number): number {
        return length * this.widthParent();
    }

    toAbsUnits(length: number): number {
        var newLength = this.toParentUnits(length);
        if (this.isBase()) 
            return newLength;

        newLength = this.parent().toRelativeUnits(newLength);
        return this.parent().toAbsUnits(newLength);
    }

    toRelativeLocation(parentLocation: number, axis: "x" | "y"): number {
        return axis === "x" ? 
            (parentLocation - this.leftParent()) / this.unit() : 
            (parentLocation - this.bottomParent()) / this.unit()
    }

    toParentLocation(location: number, axis: "x" | "y"): number {
        var loc = axis === "x" ?
            this.leftParent() + location * this.unit() :
            this.bottomParent() + location * this.unit()

        return loc;
    }

    toAbsLocation(location: number, axis: "x" | "y"): number {
        var newLocation = this.toParentLocation(location, axis);
        if (this.isBase()) 
            return newLocation;

        newLocation = this.parent().toRelativeLocation(newLocation, axis);
        return this.parent().toAbsLocation(newLocation, axis);
    }

    toRelativePosition(parentPosition: Position): Position {
        return [
            this.toRelativeLocation(parentPosition[0], "x"),
            this.toRelativeLocation(parentPosition[1], "y")
        ];
    }

    toParentPosition(position: Position): Position {
        var x = this.toParentLocation(position[0], "x");
        var y = this.toParentLocation(position[1], "y");
        return [x, y];
    }

    toAbsPosition(position: Position): Position {
        var newPosition = this.toParentPosition(position);
        return this.isBase() ? newPosition : this.parent().toAbsPosition(newPosition);
    }

    toRelativeBounds(parentBounds: RegionBounds): RegionBounds {
        return [
            this.toRelativeLocation(parentBounds[0], "x"),
            this.toRelativeLocation(parentBounds[1], "x"),
            this.toRelativeLocation(parentBounds[2], "y"),
            this.toRelativeLocation(parentBounds[3], "y")
        ];
    }

    toParentBounds(bounds: RegionBounds): RegionBounds {
        return [
            this.toParentLocation(bounds[0], "x"),
            this.toParentLocation(bounds[1], "x"),
            this.toParentLocation(bounds[2], "y"),
            this.toParentLocation(bounds[3], "y")
        ];
    }

    toAbsBounds(bounds: RegionBounds): RegionBounds {
        var newBounds = this.toParentBounds(bounds);
        return this.isBase() ? newBounds : this.parent().toAbsBounds(newBounds);
    }

    angle(position: Position, units: "degrees" | "radians" = "degrees"): number {
        var center = this.center();
        var deltaX = position[0] - center[0];
        var deltaY = position[1] - center[1];
        var radians = Math.atan2(deltaY, deltaX); // Returns angle in radians

        // Normalize and convert based on units
        if (units === "degrees") return Region.to360(Region.toDegrees(radians));
        return roundPrecisely(Region.to2Pi(radians));
    }

    // Returns the radius (distance from the center) in relative units
    radius(position: Position): number {
        var center = this.center();
        var deltaX = position[0] - center[0];
        var deltaY = position[1] - center[1];
        return Math.sqrt(deltaX ** 2 + deltaY ** 2);
    }

    toString() {
        return this.isBase() ?
            `[${this._bounds}], [${this._boundsParent}]` :
            `[${this._bounds}], [${this._boundsParent}], [${this.boundsAbs()}]`
    }

    static rectToBounds(rect: Rect): RegionBounds {
        return [rect[0], rect[2], rect[3], rect[1]];
    }

    static fromRect(rect: Rect): Region {
        return new Region(this.rectToBounds(rect));
    }

    static fromArray(bounds: number[] | [number, number, number, number] | RegionBounds): Region {
        return Region.fromRect(bounds as any);
    }

    static toRadians(degrees: number): number {
        return Region.to2Pi((degrees * Math.PI) / 180);
    }

    // Converts radians to degrees
    static toDegrees(radians: number): number {
        return Region.to360((radians * 180) / Math.PI);
    }

    static to2Pi(radians: number): number {
        var twoPi = 2 * Math.PI;
        return ((radians % twoPi) + twoPi) % twoPi;
    }

    // Normalizes degrees to the range [0, 360)
    static to360(degrees: number): number {
        var fullCircle = 360;
        return ((degrees % fullCircle) + fullCircle) % fullCircle;
    }

    private static round(value: number) {
        return roundPrecisely(value, 8);
    }
}

class Regions {
    constructor() {
        throw new Error("Regions is a static class");
    }

    static union(first: Region, second: Region): Region
    static union(...regions: Region[]): Region
    static union(...regions: Region[]): Region {
        if (regions.length === 0) throw "No regions to union";
        var left = f(regions).select(r => r.leftParent()).orderByAscending().first();
        var right = f(regions).select(r => r.rightParent()).orderByDescending().first();
        var top = f(regions).select(r => r.topParent()).orderByDescending().first();
        var bottom = f(regions).select(r => r.bottomParent()).orderByAscending().first();

        return new Region([left, right, bottom, top]);
    }

    static intersection(first: Region, second: Region): Region
    static intersection(...regions: Region[]): Region
    static intersection(...regions: Region[]): Region {
        if (regions.length === 0) throw "No regions to intersect";
        var left = f(regions).select(r => r.leftParent()).orderByDescending().first();
        var right = f(regions).select(r => r.rightParent()).orderByAscending().first();
        var top = f(regions).select(r => r.topParent()).orderByAscending().first();
        var bottom = f(regions).select(r => r.bottomParent()).orderByDescending().first();

        if (left >= right || bottom >= top)
            throw "Regions do not intersect";

        return new Region([left, right, bottom, top]);
    }

    // Lines count from the top down for the y dimension
    static lines(region: Region, orientation: "row" | "column", count: number): Region[]
    static lines(region: Region, orientation: "row" | "column", sequence: Vector, norm?: boolean): Region[]
    static lines(region: Region, orientation: "row" | "column", sequence: Vector | number, norm: boolean = true): Region[] {
        var rng: Range = [0, orientation === "row" ? region.width() : region.height()];
        if (typeof sequence === 'number')
            sequence = Vector.linspace(rng, sequence + 1);

        if (sequence.dim() !== 1) throw "Vector must be 1-dimensional";
        if (norm) sequence = sequence.norm(rng);

        
        var regions = new A<Region>();
        sequence.xa().do((x2, i) => {
            if (i === 0) return;
            var x1 = sequence.att(i - 1);
            var [start, end] = [Math.min(x1, x2), Math.max(x1, x2)];
            regions.push(orientation === "row"
                ? region.createRegion([start, end, region.bottom(), region.top()])
                : region.createRegion([region.left(), region.right(), start, end]));
        });

        if (orientation == "column")
            regions = regions.reverse();

        return regions.a();
    }

    // Returns a row by column grid
    static grid(region: Region, xCount: number, yCount?: number): Region[][]
    static grid(region: Region, xSequence: Vector, ySequence?: Vector, norm?: boolean): Region[][]
    static grid(region: Region, xCount: number, ySequence: Vector, norm?: boolean): Region[][]
    static grid(region: Region, xSequence: Vector, yCount: number, norm?: boolean): Region[][]
    static grid(region: Region, xSequence: Vector | number, ySequence?: Vector | number, norm: boolean = true): Region[][] {
        if (ySequence === undefined) ySequence = xSequence;

        var xRows = f(Regions.lines(region, "row", xSequence as Vector, norm));
        var yColumns = f(Regions.lines(region, "column", ySequence as Vector, norm));

        var result = yColumns.select(yStrip => 
            xRows.select(xStrip =>
                Regions.intersection(xStrip, yStrip)
            )
        );

        return result.select(r => r.a()).a();
    }
}
