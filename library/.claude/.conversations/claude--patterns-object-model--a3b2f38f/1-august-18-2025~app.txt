/**
 * Base class for all document-related references.
 * Provides common methods to access the application, document, typename, and parent object.
 */
class $ApplicationReference<IItem = any, TParent extends $ApplicationItem = any> extends $Reference<IItem> {
    private _typename: string;

    app(): $Application { return $app; }
    typename(): string { return this._typename; }
    parent(): TParent { return $docref<TParent>(this.$get(o => (o as Layer).parent), this.typename()); }

    constructor(ref: $Reference<IItem> | string, typename: string) {
        super(ref instanceof $Reference ? ref.address() : ref)
        this._typename = typename;
    }

    $next(): this {
        var ref = super.$next();
        return $docref(ref, this.typename());
    }
}

/**
 * Represents an object within a document that has a name and can be removed.
 * Used for individual objects like layers, items, and styles.
 */
class $ApplicationItem<TItem = any, TParent extends $ApplicationItem = any> extends $ApplicationReference<TItem, TParent> {
    protected _name(newValue?: $Str) { return this.$get<string>(o => (o as Layer).name, newValue); }

    collection<T extends $ApplicationCollection = $ApplicationCollection>(): T { 
        var collectionType = `${this.typename()}s`;
        var collectionName = `${str(collectionType[0]).toLowerCase()}${collectionType.slice(1)}`;
        return $docref(this.$dot("parent").$dot(collectionName), collectionType) as any;
    }

    index(): $Reference<number> {
        var collection = this.collection();
        if (!collection) return $arg(-1);
        return this.$func((self, collection) => {
            for (var i = 0; i < collection.length; i++)
                if (self === collection[i])
                    return i;
            return -1;
        }, collection);
    }

    protected _remove(): void
    protected _remove(...args: $Reference[]): void {
        this.$call(o => (o as Layer).remove, args).$();
    }
}

/**
 * Represents a collection of objects within a application.
 * Provides methods for adding, accessing, and removing objects in the collection.
 */
class $ApplicationCollection<TCollection extends Array<TElement> = any, $TElement extends $ApplicationItem = any, TElement extends ApplicationItem = any, TParent extends $ApplicationItem = any> extends $ApplicationReference<TCollection, TParent> {
    private _etypename: string;
    protected _selector: (index: $Num) => $TElement;

    app(): $Application { return $app; }
    length() { return this.$get<number>(o => o.length); }

    constructor(ref: $Reference<TCollection> | string, typename: string) {
        super(ref, typename)
        this._etypename = typename.slice(0, -1);
        this._selector = index => $docref<$TElement>(this.$of(index), this._etypename);
    }

    at(index: $Num): $TElement { return this._selector(index); }

    a(): $ArrayReference<TElement> {
        return $arr(this.$func(self => {
            var collection = self;
            var elements = [];
            for (var i = 0; i < collection.length; i++)
                elements.push(collection[i]);
            return elements;
        }));
    }

    protected _add(): $TElement
    protected _add(...args: $Any[]): $TElement
    protected _add(...args: $Any[]): $TElement {
        var ref = $var().equals(this.$call("add", ...args));
        return $docref(ref, this._etypename);
    }

    protected _removeAll(): $Effect {
        return this.a().do(item => {
            if ("remove" in item) {
                (item as any).remove();
            }
        });
    }

    protected _getByName(name: $Str): $TElement {
        var ref = $var().equals(this.$call("getByName", name));
        return $docref<$TElement>(ref, this._etypename);
    }

    private static nextId = 1;
    private static variableName = "element";
    protected static getNextVariableName() { return `${this.variableName}${this.nextId++}`; }
}

/**
 * Represents an object inside a document that **can be moved, duplicated, and reordered**.
 * Used for `Layers`, `PathItems`, `PageItems`, `GroupItems`.
 */
class $DocumentItem<TObject extends ApplicationItem = any, TParent extends $ApplicationItem = any> extends $ApplicationItem<TObject> {
    name(newValue?: $Str) { return this._name(newValue); }
    move(relativeObject: this | TParent, insertionLocation: $Arg<ElementPlacement>): void { this.$call("move", relativeObject, insertionLocation).$(); }
    duplicate(relativeObject: this | TParent, insertionLocation: $Arg<ElementPlacement>): void { this.$call("duplicate", relativeObject, insertionLocation).$(); }
    zOrder(command: $Arg<ZOrderMethod>): void { this.$call("zOrder", command).$(); }

    moveTo(index: $Num): $Effect {
        return this.$func((self, currentIndex, targetIndex, collection, placeAfter, placeBefore) => {
            if (currentIndex === -1 || currentIndex === targetIndex) return;
            var relative = collection[targetIndex];
            var location = currentIndex < targetIndex ? placeAfter : placeBefore;
            (self as any).move(relative, location);
        }, this.index(), index, this.collection(), ElementPlacement.PLACEAFTER, ElementPlacement.PLACEBEFORE).$() as any; 
    }
}

        }, relativePath), "Document");
    }

        return this.find(relativePath).$dot(o => o?.activate());

        $app.documents().add(DocumentColorSpace.RGB, width, height, 1).$$();

            var file = new File(combinePaths(DOCUMENTS_PATH, relativePath));
            var doc = app.activeDocument;
            doc.saveAs(file);
            return doc;
        }, relativePath).$$();
        return this.find(relativePath);
    }

    static open(relativePath: $Str): $Document {
        this.find(relativePath).$func((doc, relativePath) => {
            if (doc) {
                doc.activate();
                return doc;
            }
            if (this.exists(relativePath).val()) {
                var file = new File(combinePaths(DOCUMENTS_PATH, relativePath));
                return app.open(file);
            }
            return this.create(relativePath);
        }, relativePath).$();
        return $app.activeDocument();
    }

function $appref<T extends $ApplicationItem = any>(item: ApplicationItem): T {
    var [ref, document] = _$appref(item);
    return $docref<T>(ref, item.typename || "");
}

function _$appref(item: ApplicationItem): [string, Document] {
    if (!item) throw Error("$appref(item) undefined");
    if (!item.typename) throw Error("item.typename undefined");
    if (!item.parent) return [$app.address(), app.activeDocument];
    
    var parent: ApplicationItem = item.parent;
    var collectionName = `${str(item.typename[0]).toLowerCase()}${item.typename.slice(1)}s`;
    var collection = item.parent[collectionName] as ApplicationCollection<any>;
    var index = -1;
    for (var i = 0; i < collection.length; i++) {
        if (collection[i] === item) {
            index = i;
            break;
        }
    }

    if (index == -1) 
        throw Error(`$appref(item) of type ${item.typename} not found on ${parent.typename}.${collectionName}`);

    var document = parent.typename === "Application" ? app.documents[index] : app.activeDocument;
    return [`${_$appref(parent)[0]}.${collectionName}[${index}]`, document];
}
