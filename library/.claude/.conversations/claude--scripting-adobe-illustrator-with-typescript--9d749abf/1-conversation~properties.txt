class SetProperties<TQueryItem extends QueryItem = QueryItem> {
    protected _query: TQueryItem;
    
    constructor(query: TQueryItem, queryName?: string) {
        if (query === undefined) throw "queryItem undefined";
        this._query = query;
        if (query !== undefined)
            this._query.name(queryName);
    }

    name(name: string): this { this._query.name(name); return this; }
    index(newIndex: number): this { this._query.moveTo(newIndex); return this; }
    locationToFront(): this { this._query.moveTo(0); return this; }
    locationToBack(): this { this._query.moveTo(-1); return this; }
    locationBefore(referenceItem: QueryItem) { this._query.move(referenceItem, ElementPlacement.PLACEBEFORE); }
    locationAfter(referenceItem: QueryItem) { this._query.move(referenceItem, ElementPlacement.PLACEAFTER); }
    q():QueryItem { return this._query; }
}

class SetPropertiesTests extends AppTestClass {
    constructor() {
        super();
        this.decorateExceptions = false;
    }

    testIndex(): void {
        var doc = api.activeDocument();
        var layer = doc.activeLayer();

        // Test moving in empty layer
        const path1 = layer.pathItems().createPathItem("Path 1");
        path1.set().index(0);
        this.assertEquals(path1.index(), 0, "Single item should stay at index 0");

        // Test moving with multiple items
        const path2 = layer.pathItems().createPathItem("Path 2");
        const path3 = layer.pathItems().createPathItem("Path 3");
        const path4 = layer.pathItems().createPathItem("Path 4");
        
        path4.set().index(1);  // Move to middle
        this.assertEquals(path4.index(), 1, "Path4 should be at index 1");
        
        path2.set().index(0);  // Move to front
        this.assertEquals(path2.index(), 0, "Path2 should be at index 0");
        
        path3.set().index(3);  // Move to end
        this.assertEquals(path3.index(), 3, "Path3 should be at back");

        // Test with new layer
        const layer2 = doc.layers().create();
        layer2.set().index(0);
        this.assertEquals(layer2.index(), 0, "New layer should move to front");
        
        // Cleanup
        layer.pathItems().removeAll();
        layer2.remove();
    }

    testLocationToFront(): void {
        var doc = api.activeDocument();
        var layer = doc.activeLayer();

        // Test with single item
        const single = layer.pathItems().create("Single").set().q();
        single.set().locationToFront();
        this.assertEquals(single.index(), 0, "Single item should stay at front");

        // Test with three items
        const back = layer.pathItems().create().set("Back").q();
        const middle = layer.pathItems().create("Middle").set().q();
        const front = layer.pathItems().createPathItem("Front");
        
        back.set().locationToFront();
        this.assertEquals(back.index(), 0, "Back item should move to front");
        this.assertEquals(middle.index(), 2, "Middle item should shift back");
        
        middle.set().locationToFront();
        this.assertEquals(middle.index(), 0, "Middle item should move to front");
        this.assertEquals(back.index(), 1, "Previous front item should shift back");

        // Test with layers
        const newLayer = doc.layers().create();
        newLayer.set().locationToFront();
        this.assertEquals(newLayer.index(), 0, "New layer should move to front");
        
        // Cleanup
        layer.pathItems().removeAll();
        newLayer.remove();
    }

    testLocationToBack(): void {
        var doc = api.activeDocument();
        var layer = doc.activeLayer();

        // Create several items in known positions
        const items = [
            layer.pathItems().create("Item 1"),
            layer.pathItems().create("Item 2"),
            layer.pathItems().create("Item 3"),
            layer.pathItems().create("Item 4"),
            layer.pathItems().create("Item 5")
        ];

        // Test moving front item to back
        items[0].set().locationToBack();
        this.assertEquals(items[0].index(), 4, "Front item should move to back");
        
        // Test moving middle item to back
        items[2].set().locationToBack();
        this.assertEquals(items[2].index(), 4, "Middle item should move to back");
        this.assertEquals(items[0].index(), 3, "Previous back item should shift up");

        // Test with layers
        const newLayer = doc.layers().create();
        const newLayer2 = doc.layers().create();
        newLayer.set().locationToBack();
        this.assertEquals(newLayer.index(), doc.layers().length() - 1, "Layer should move to back");

        // Cleanup
        layer.pathItems().removeAll();
        newLayer.remove();
        newLayer2.remove();
    }

    testLocationBefore(): void {
        var doc = api.activeDocument();
        var layer = doc.activeLayer();

        // Create test items
        const items = [
            layer.pathItems().create("Item 1"),
            layer.pathItems().create("Item 2"),
            layer.pathItems().create("Item 3"),
            layer.pathItems().create("Item 4")
        ];

        // Test moving to before first item
        items[3].set().locationBefore(items[0]);
        this.assertEquals(items[3].index(), 0, "Should move before first item");

        // Test moving to middle
        items[1].set().locationBefore(items[3]);
        this.assertEquals(items[1].index(), items[3].index() - 1, "Should move before target item");

        // Test with adjacent items
        items[2].set().locationBefore(items[1]);
        this.assertEquals(items[2].index(), items[1].index() - 1, "Should move before adjacent item");

        // Test with layers
        const newLayer = doc.layers().create();
        const targetLayer = doc.layers().create();
        newLayer.set().locationBefore(targetLayer);
        this.assertEquals(newLayer.index(), targetLayer.index() - 1, "Layer should move before target");

        // Cleanup
        layer.pathItems().removeAll();
        newLayer.remove();
        targetLayer.remove();
    }

    testLocationAfter(): void {
        var doc = api.activeDocument();
        var layer = doc.activeLayer();

        // Create test items
        const items = [
            layer.pathItems().create("First"),
            layer.pathItems().create("Second"),
            layer.pathItems().create("Third"),
            layer.pathItems().create("Fourth")
        ];

        // Test moving to after last item
        items[0].set().locationAfter(items[3]);
        this.assertEquals(items[0].index(), 3, "Should move after last item");

        // Test moving to middle
        items[2].set().locationAfter(items[1]);
        this.assertEquals(items[2].index(), items[1].index() + 1, "Should move after middle item");

        // Test moving adjacent items
        items[1].set().locationAfter(items[2]);
        this.assertEquals(items[1].index(), items[2].index() + 1, "Should move after adjacent item");

        // Test moving multiple times
        items[3].set().locationAfter(items[0]);
        items[3].set().locationAfter(items[2]);
        this.assertEquals(items[3].index(), items[2].index() + 1, "Should handle multiple moves");

        // Test with layers
        const newLayer = doc.layers().create();
        const referenceLayer = doc.layers().create();
        newLayer.set().locationAfter(referenceLayer);
        this.assertEquals(newLayer.index(), referenceLayer.index() + 1, "Layer should move after reference");

        // Cleanup
        layer.pathItems().removeAll();
        newLayer.remove();
        referenceLayer.remove();
    }
}
