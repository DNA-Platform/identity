class Console {
    private _on: boolean = false;
    on(): this { this._on = true; return this; }
    off(): this {  this._on = false; return this; }
    active(): boolean { return this._on; }

    log(...args: any[]) {
        var turnOff = false;
        if (args && args.length > 0 && args[0] == "on") {
            this.on();
            args = f(args).skip(1).a;
        }
        if (args && args.length > 0 && args[0] == "off") {
            turnOff = true;
            args = f(args).skip(1).a;
        }
        args = args ? args : [];
        args.unshift("debug:");
        this.show(false, ...args);
        if (turnOff) this.off();
    }

    break(...args: any[]) {
        var turnOff = false;
        if (args && args.length > 0 && args[0] == "on") {
            this.on();
            args = f(args).skip(1).a;
        }
        if (args && args.length > 0 && args[0] == "off") {
            turnOff = true;
            args = f(args).skip(1).a;
        }
        args = args ? args : [];
        args.unshift("debug:");
        this.show(true, ...args);
        if (turnOff) this.off();
    }

    private show(halt: boolean, ...args: any[]) {
        if (!this._on) return;
        app.redraw();
        args = args ? args : [];
        var result = "";
        var arr = f(args).select(s => `${s}`).where(s => s.length > 0)
        arr.do((s, i) => {
            i = i as number;
            result += (i < arr.length()-1 ? `${s}\n ` : s);
        })
        if (halt) throw result;
        else alert(result);
    }
}

/**
 * Utility function to check if a string starts with a specific prefix
 * @param str - The main string to check
 * @param prefix - The prefix to look for
 * @returns True if the string starts with the prefix, otherwise false
 */
function stringStartsWith(str: string, prefix: string): boolean {
    return str.slice(0, prefix.length) === prefix;
}

/**
 * Utility function to check if a string ends with a specific suffix
 * @param str - The main string to check
 * @param suffix - The suffix to look for
 * @returns True if the string ends with the suffix, otherwise false
 */
function stringEndsWith(str: string, suffix: string): boolean {
    return str.slice(-suffix.length) === suffix;
}

/**
 * Utility function to join an array of strings with a specified separator
 * @param array - The array of strings to join
 * @param separator - The separator to use between strings
 * @returns A single string formed by joining the array elements
 */
function stringJoin(array: string[], separator: string): string {
    var result = "";
    for (var i = 0; i < array.length; i++) {
        result += array[i];
        if (i < array.length - 1) {
            result += separator; // Add separator only between elements
        }
    }
    return result;
}

/**
 * Utility function to check if a string contains a specific substring
 * @param str - The main string to check
 * @param search - The substring to look for
 * @returns True if the string contains the substring, otherwise false
 */
function stringIncludes(str: string, search: string): boolean { return stringContains(str, search) }
function stringContains(str: string, search: string): boolean {
    if (str === undefined || str === null) return false;
    if (search === "") return true;
    
    var index = 0;
    var len = str.length;
    var searchLen = search.length;
    
    while (index + searchLen <= len) {
        if (str.slice(index, index + searchLen) === search) {
            return true;
        }
        index++;
    }
    return false;
}

/**
 * Utility function to compare two arrays for equality
 * @param first - The first array to compare
 * @param second - The second array to compare
 * @returns True if the arrays are equal (same length and elements), otherwise false
 */
function arrayEquals(first: any[], second: any[]): boolean {
    if (!first && !second) return true;
    if (!first || !second) return false;
    if (first.length !== second.length) return false;

    for (let i = 0; i < first.length; i++) {
        if (first[i] !== second[i]) return false;
    }

    return true; 
}

/**
 * Deep clones an object in ExtendScript, handling arrays, objects, and common edge cases.
 * @param source - The object to clone.
 * @param seen - A record of already-seen objects to handle circular references.
 * @returns The deep-cloned object.
 */
function clone<T>(obj: T): T {
    // Handle null, undefined, or primitive types directly
    var source: any = obj;
    function isPrimitive(value: any): boolean {
        return value === null || 
            value === undefined || 
            Object.prototype.toString.call(value) === "[object Boolean]" || 
            Object.prototype.toString.call(value) === "[object Number]" || 
            Object.prototype.toString.call(value) === "[object String]" || 
            Object.prototype.toString.call(value) === "[object Function]" || 
            Object.prototype.toString.call(value) === "[object Date]" || 
            Object.prototype.toString.call(value) === "[object RegExp]";
    }

    if (isPrimitive(source))
        return obj;

    // Detect arrays (primitive approach for ES3/ES5)
    if (Object.prototype.toString.call(source) === "[object Array]") {
        var copy: any = [];
        for (var i = 0; i < (source as any[]).length; i++) {
            copy[i] = clone((source as any[])[i]);
        }
    }
    // Handle objects with no prototype (created via Object.create(null))
    else {
        var copy: any = {};
        for (var key in source) {
            if (Object.prototype.hasOwnProperty.call(source, key) && key != "parent") {
                try {
                    copy[key] = clone(source[key]);
                } catch (e) {
                    throw "key " + key + ", source " + source;
                }
            }
        }
    }

    return copy as T;
}

/**
 * Rounds a number to a specific number of decimals
 * @param value - The number to round
 * @param decimals - The number of decimals of precision
 * @returns The rounded value
 */
function roundPrecisely(value: number, decimals: number = 6): number {
    const factor = Math.pow(10, decimals);
    return Math.round(value * factor) / factor;
}

/**
 * Normalizes file path separators to Windows style (backslashes)
 * and ensures no double separators
 * @param path - The file path to normalize
 * @returns Normalized path with proper separators
 */
function normalizePath(path: string): string {
    return path.replace(/[\/\\]+/g, '\\');
}

/**
 * Combines two paths, ensuring proper separator handling
 * @param path1 - First path
 * @param path2 - Second path
 * @returns Combined path with proper separators
 * 
 * @example
 * // Returns "C:\\Scripts\\documents\\test.ai"
 * combinePaths("C:\\Scripts", "documents/test.ai")
 * 
 * @example
 * // Returns "C:\\Scripts\\documents\\test.ai"
 * combinePaths("C:\\Scripts\\", "\\documents\\test.ai")
 */
function combinePaths(path1: string, path2: string): string {
    const normalized1 = normalizePath(path1);
    const normalized2 = normalizePath(path2);

    // Remove trailing slash from first path and leading slash from second path
    const clean1 = stringEndsWith(normalized1, '\\') ? normalized1.slice(0, -1) : normalized1;
    const clean2 = stringStartsWith(normalized2, '\\') ? normalized2.slice(1) : normalized2;

    return clean1 + '\\' + clean2;
}

/**
 * Converts a negative index to a positive index within the bounds of the collection length.
 *
 * @param index - The index to evaluate (can be negative)
 * @param collectionLength - The length of the collection
 * @returns A valid positive index within the collection bounds
 * @throws Error if the evaluated index is out of bounds
 */
function evaluateNegativeIndex(index: number, collectionLength: number): number {
    var evaluatedIndex = index < 0 ? collectionLength + index : index;
    if (evaluatedIndex < 0 || evaluatedIndex >= collectionLength) {
        throw "Index " + index + " out of bounds for collection of length " + collectionLength;
    }
    return evaluatedIndex;
}

/**
 * Converts a negative index to a positive index within the bounds of the collection length but maps -1 to the length of the array so the item gets inserted at the end.
 *
 * @param index - The index to evaluate (can be negative)
 * @param collectionLength - The length of the collection
 * @returns A valid positive index within the collection bounds
 * @throws Error if the evaluated index is out of bounds
 */
function evaluateNegativeInsertIndex(index: number, collectionLength: number): number {
    var evaluatedIndex = index < 0 ? collectionLength + index + 1 : index;
    if (evaluatedIndex < 0 || evaluatedIndex > collectionLength) {
        throw "Index " + index + " out of insert bounds for collection of length " + collectionLength;
    }
    return evaluatedIndex;
}

/**
 * Evaluates a selector against a collection length and returns either:
 * - A string (when selector is a name)
 * - An array of indices (for all other valid selectors)
 * All indices are guaranteed to be within bounds [0, collectionLength-1]
 * 
 * @param selector - The selector to evaluate
 * @param collectionLength - Length of the collection being indexed
 * @returns String name or array of valid indices
 * @throws Error for invalid selectors or out of bounds indices
 * 
 * @example
 * evaluateSelector("MyName", 5)      // returns "MyName"
 * evaluateSelector(1, 5)             // returns [1]
 * evaluateSelector(-1, 5)            // returns [4]
 * evaluateSelector("1:3", 5)         // returns [1, 2, 3]
 * evaluateSelector("-2:", 5)         // returns [3, 4]
 * evaluateSelector([0, -1], 5)       // returns [0, 4]
 * evaluateSelector(["a", "b"], 5)    // returns ["a", "b"]
 */
function evaluateSelector(selector: Selector, collectionLength: number): string[] | number[] {
    // Handle undefined case
    if (selector === undefined) {
        return [0];  // Default to first item
    }

    // Handle single number
    if (typeof selector === "number") {
        return [evaluateNegativeIndex(selector, collectionLength)];
    }

    // Handle arrays
    if (selector instanceof Array) {
        // If it's a string array, return as is
        if (typeof selector[0] === "string") {
            return selector as string[];
        }

        // Convert number array with negative index handling
        var resolvedIndices = [];
        for (var i = 0; i < selector.length; i++) {
            resolvedIndices.push(evaluateNegativeIndex(selector[i] as number, collectionLength));
        }
        return resolvedIndices;
    }

    // Handle string
    if (typeof selector === "string") {
        // If doesn't start with number or :, it's a name
        if (!/^[0-9:-]/.test(selector)) {
            return [selector];
        }

        // Handle empty string
        if (selector.length === 0) {
            throw "Empty selector";
        }

        return evaluateRangeSelector(selector, collectionLength);
    }

    throw "Invalid selector type";
}

function evaluateRangeSelector(selector: string, collectionLength?: number): number[] {
    if (!selector) return []; // Empty selector returns an empty array
    if (collectionLength === 0 && selector == ":") return [];

    // Split the selector by ":" and parse parts into numbers
    var parts: any[] = selector.split(":");
    if (parts.length == 0 || parts.length > 3) {
        "Selector has the wrong number of parts"
    }
    if (collectionLength !== undefined) {
        if (parts.length >= 1 && parts[0] == "")
            parts[0] = "0";
        if (parts.length == 2 && parts[1] == "")
            parts[1] = `${collectionLength-1}`;
        if (parts.length == 3 && parts[1] == "")
            parts[1] = "1";
        if (parts.length == 3 && parts[2] == "")
            parts[2] = `${collectionLength-1}`;
    }
    for (var i = 0; i < parts.length; i++) {
        parts[i] = parseInt(parts[i], 10);
        if (isNaN(parts[i])) {
            throw "Selector contains invalid numbers";
        }
    }

    var len = parts.length;
    if (len === 1) {
        // Single number, return it as an array
        return [
            collectionLength !== undefined ? 
            evaluateNegativeIndex(parts[0], collectionLength) : 
            parts[0]
        ];
    } else if (len === 2) {
        // Range with a single step (inclusive range)
        var start: number = parts[0];
        var end: number = parts[1];
        if (collectionLength !== undefined) {
            start = evaluateNegativeIndex(start, collectionLength);
            end = evaluateNegativeIndex(end, collectionLength);
        }
        var range = [];
        if (start <= end) {
            for (var i = start; i <= end; i++) {
                range.push(i);
            }
        } else {
            // Decreasing range
            for (var i = start; i >= end; i--) {
                range.push(i);
            }
        }
        return range;
    } else if (len === 3) {
        // Range with custom step
        var start: number = parts[0];
        var step: number = parts[1];
        var end: number = parts[2];

        if (collectionLength !== undefined) {
            start = evaluateNegativeIndex(start, collectionLength);
            end = evaluateNegativeIndex(end, collectionLength);
        }
        if (step === 0 || (start < end && step < 0) || (start > end && step > 0)) {
            throw new Error("Invalid step value in the range!");
        }

        var range = [];
        if (start < end) {
            for (var i = start; i <= end; i += step) {
                range.push(i);
            }
        } else {
            for (var i = start; i >= end; i += step) {
                range.push(i);
            }
        }
        return range;
    } else {
        throw new Error("Invalid range selector format!");
    }
}

class A<T> {
    a: T[];

    constructor(array?: T[]) {
        if (array === undefined) array = []; 
        this.a = array;
    }

    equals(other: T[] | A<T>): boolean {
        if (!other) return false;
        var that = other instanceof Array ? other : other.a
        return arrayEquals(this.a, that);
    }

    doesntEqual(other: T[] | A<T>): boolean {
        return !this.equals(other);
    }

    at(index: number): T {
        index = evaluateNegativeIndex(index, this.a.length);
        return this.a[index];
    }

    get(selector: string): A<T> {
        var indexes = evaluateRangeSelector(selector, this.a.length);
        var result = new A<T>();
        for (var i = 0; i < indexes.length; i++) {
            result.push(this.a[indexes[i]]);
        }
        return result;
    }

    set(index: number, item: T): A<T> {
        index = evaluateNegativeIndex(index, this.a.length);
        this.a[index] = item;
        return this;
    }

    length(): number { return this.a.length; }

    insertAt(index: number, item: T): A<T> {
        const insertAt = evaluateNegativeInsertIndex(index, this.a.length);
        this.a.splice(insertAt, 0, item);
        return this;
    }

    insertManyAt(index: number, items: T[]): A<T> {
        const insertAt = evaluateNegativeInsertIndex(index, this.a.length);
        this.a.splice(insertAt, 0, ...items);
        return this;
    }

    delete(item: T): A<T> {
        for (let i = 0; i < this.a.length; i++) {
            if (this.a[i] === item) {
                this.a.splice(i, 1);
                break;
            }
        }
        return this;
    }

    deleteMany(items: T[]): A<T> {
        f(items).forEach(item => this.delete(item));
        return this;
    }

    deleteAt(index: number, count: number = 1): A<T> {
        const removeFrom = evaluateNegativeIndex(index, this.a.length);
        this.a.splice(removeFrom, count);
        return this;
    }

    deleteAll(): A<T> {
        this.a.length = 0;
        return this;
    }

    forEach(callback: (item: T, index?: number) => any | void): any | void {
        let result: any;
        for (let i = 0; i < this.a.length; i++) {
            result = callback(this.a[i], i);
        }
        return result;
    }

    do(callback: (item: T, index?: number) => any | void): A<T> {
        this.forEach(callback);
        return this;
    }

    doOnce(callback: (item: T, index?: number) => boolean): boolean {
        for (let i = 0; i < this.a.length; i++) {
            if (callback(this.a[i], i)) {
                return true;
            }
        }
        return false;
    }

    select<U>(selector: (element: T) => U): A<U> {
        const result = new A<U>();
        for (let i = 0; i < this.a.length; i++) {
            result.push(selector(this.a[i]));
        }
        return result;
    }

    where(predicate: (element: T) => boolean): A<T> {
        const result = new A<T>();
        for (let i = 0; i < this.a.length; i++) {
            if (predicate(this.a[i])) {
                result.push(this.a[i]);
            }
        }
        return result;
    }

    first(predicate?: (element: T) => boolean): T {
        if (!predicate) return this.a.length > 0 ? this.a[0] : undefined as any as T;
        for (let i = 0; i < this.a.length; i++) {
            if (predicate(this.a[i])) return this.a[i];
        }
        return undefined as any as T;
    }

    firstIndex(predicate?: (element: T) => boolean): number {
        if (!predicate) return this.a.length > 0 ? 0 : -1;
        for (let i = 0; i < this.a.length; i++) {
            if (predicate(this.a[i])) return i;
        }
        return -1;
    }

    last(predicate?: (element: T) => boolean): T | undefined {
        if (!predicate) return this.a.length > 0 ? this.a[this.a.length - 1] : undefined;
        for (let i = this.a.length - 1; i >= 0; i--) {
            if (predicate(this.a[i])) return this.a[i];
        }
        return undefined;
    }

    lastIndex(predicate?: (element: T) => boolean): number {
        if (!predicate) return this.a.length > 0 ? this.a.length - 1 : -1;
        for (let i = this.a.length - 1; i >= 0; i--) {
            if (predicate(this.a[i])) return i;
        }
        return -1;
    }

    any(predicate?: (element: T) => boolean): boolean {
        if (!predicate) return this.a.length > 0;
        for (let i = 0; i < this.a.length; i++) {
            if (predicate(this.a[i])) return true;
        }
        return false;
    }

    all(predicate: (element: T) => boolean): boolean {
        return this.length() == this.where(predicate).length();
    }

    includes(item: T): boolean {
        return this.any(i => i == item);
    }

    // Array methods
    push(...items: T[]): A<T> {
        this.a.push(...items);
        return this;
    }

    pop(): T | undefined {
        return this.a.pop();
    }

    splice(start: number, deleteCount?: number, ...items: T[]): A<T> {
        this.a.splice(start, deleteCount, ...items);
        return this;
    }

    slice(start?: number, end?: number): A<T> {
        return new A(this.a.slice(start, end));
    }

    shift(): T | undefined {
        return this.a.shift();
    }

    unshift(...items: T[]): A<T> {
        this.a.unshift(...items);
        return this;
    }

    reverse(): A<T> {
        this.a.reverse();
        return this;
    }

    sort(compareFunction?: (a: T, b: T) => number): A<T> {
        this.a.sort(compareFunction);
        return this;
    }

    join(separator: string): string {
        return stringJoin(this.select(x => x + "").a, separator);
    }

    skip(count: number): A<T> {
        if (count >= this.length()) return f([]);
        return this.get(`${count}:`)
    }

    take(count: number): A<T> {
        count = Math.min(count, this.length());
        return this.get(`0:${count-1}`);
    }

    toString() {
        if (this.a === undefined) return "[undefined]";
        return this.a.toString();
    }

    static empty(): A<any> { return new A<any>([]) };
}

function f<T>(array: T[] | A<T> = A.empty()): A<T> {
    if (array instanceof A) return array as A<T>;
    return new A(array);
}

function r(selector: string, length?: number): A<number> {
    return f(evaluateRangeSelector(selector, length));
}

class Region {
    private _leftAbs: number;
    private _topAbs: number;
    private _rightAbs: number;
    private _bottomAbs: number;

    // Measure
    unit(): number { return this._rightAbs - this._leftAbs; }

    // Absolute positions
    boundsAbs(): [number, number, number, number] { return [this.leftAbs(), this.topAbs(), this.rightAbs(), this.bottomAbs()]; }
    rectAbs(): Rect { return this.boundsAbs(); }
    topAbs() { return this._topAbs; }
    leftAbs() { return this._leftAbs; }
    rightAbs() { return this._rightAbs; }
    bottomAbs() { return this._bottomAbs; }
    centerAbs(): Position { return [(this._rightAbs + this._leftAbs) / 2, (this._topAbs + this._bottomAbs) / 2]; }
    widthAbs() { return this.rightAbs() - this.leftAbs(); }
    heightAbs() { return this.topAbs() - this.bottomAbs(); }

    // Relative positions
    bounds(): [number, number, number, number] { return [this.left(), this.top(), this.right(), this.bottom()]; }
    rect(): Rect { return this.bounds(); }
    left() { return this.toRelativeLocation(this.leftAbs(), "x"); }
    right() { return this.toRelativeLocation(this.rightAbs(), "x"); }
    top() { return this.toRelativeLocation(this.topAbs(), "y"); }
    bottom() { return this.toRelativeLocation(this.bottomAbs(), "y"); }
    center(newCenter?: Position) {
        if (newCenter) this.moveTo(this.toAbsolutePosition(newCenter));
        return this.toRelativePosition(this.centerAbs());
    }

    // Width and height with optional resizing
    width(newWidth?: number, proportional = true) {
        if (newWidth) {
            const scaleFactor = newWidth / this.width();
            this.size([newWidth, proportional ? this.height() * scaleFactor : this.height()]);
        }
        return Math.abs(this.left() - this.right());
    }

    height(newHeight?: number, proportional = true) {
        if (newHeight) {
            const scaleFactor = newHeight / this.height();
            this.size([proportional ? this.width() * scaleFactor : this.width(), newHeight]);
        }
        return Math.abs(this.top() - this.bottom());
    }

    // Set new size
    size(newSize?: Size) {
        if (newSize) {
            const center = this.centerAbs();
            const newWidth = this.toAbsoluteUnits(newSize[0]);
            const newHeight = this.toAbsoluteUnits(newSize[1]);

            this._leftAbs = center[0] - newWidth / 2;
            this._rightAbs = center[0] + newWidth / 2;
            this._topAbs = center[1] + newHeight / 2;
            this._bottomAbs = center[1] - newHeight / 2;
        }
        return [this.width(), this.height()];
    }

    constructor(left: number, top: number, right: number, bottom: number) {
        if (left > right || bottom > top) 
            throw `Invalid Region: ${[left, top, right, bottom]}`;
        this._leftAbs = left;
        this._topAbs = top;
        this._bottomAbs = bottom;
        this._rightAbs = right;
    }

    floor(): Region {
        this._leftAbs = Math.floor(this._leftAbs);
        this._topAbs = Math.floor(this._topAbs);
        this._rightAbs = Math.floor(this._rightAbs);
        this._bottomAbs = Math.floor(this._bottomAbs);
        return this;
    }

    moveTo(newCenter: Position) {
        const deltaX = newCenter[0] - this.centerAbs()[0];
        const deltaY = newCenter[1] - this.centerAbs()[1];
        this._leftAbs += deltaX;
        this._rightAbs += deltaX;
        this._topAbs += deltaY;
        this._bottomAbs += deltaY;
    }

    toRelativeUnits(length: number): number {
        // Normalize length based on width (unit becomes total width for scaling)
        return this.round(length / this.widthAbs());
    }
    
    toRelativeLocation(location: number, axis: "x" | "y"): number {
        const left = this.leftAbs();
        const bottom = this.bottomAbs();
        const unit = this.unit();
    
        return this.round
        (
            axis === "x"
                ? (location - left) / unit // Normalize x-axis [0, 1]
                : (location - bottom) / unit // Normalize y-axis proportional to width
        );
    }
    
    toRelativePosition(position: Position): Position {
        const x = this.toRelativeLocation(position[0], "x");
        const y = this.toRelativeLocation(position[1], "y");
        return [
            this.round(x), 
            this.round(y)
        ];
    }
    
    toAbsoluteUnits(length: number): number {
        // Scale relative length back to absolute based on width
        return roundPrecisely(length * this.widthAbs());
    }
    
    toAbsoluteLocation(location: number, axis: "x" | "y"): number {
        const left = this.leftAbs();
        const bottom = this.bottomAbs();
        const unit = this.unit();
    
        return this.round(
            axis === "x"
                ? left + location * unit // Convert back to absolute x
                : bottom + location * unit // Convert back to absolute y (proportional to width)
        );
    }
    
    toAbsolutePosition(position: Position): Position {
        const x = this.toAbsoluteLocation(position[0], "x");
        const y = this.toAbsoluteLocation(position[1], "y");
        return [this.round(x), this.round(y)];
    }

    angle(position: Position, units: "degrees" | "radians" = "degrees"): number {
        const center = this.center();
        const deltaX = position[0] - center[0];
        const deltaY = position[1] - center[1];
        const radians = Math.atan2(deltaY, deltaX); // Returns angle in radians
    
        // Normalize and convert based on units
        if (units === "degrees") return Region.to360(Region.toDegrees(radians));
        return Region.to2Pi(radians);
    }
    
    // Returns the radius (distance from the center) in relative units
    radius(position: Position): number {
        const center = this.center();
        const deltaX = position[0] - center[0];
        const deltaY = position[1] - center[1];
        return Math.sqrt(deltaX ** 2 + deltaY ** 2);
    }

    round(value: number) {
        return roundPrecisely(value, 3);
    }

    toString() {
        return `[${this.bounds()}], [${this.boundsAbs()}]`;
    }

    static fromRect(rect: Rect): Region {
        return new Region(rect[0], rect[1], rect[2], rect[3]);
    }

    static fromArray(rect: [number, number, number, number]): Region {
        return Region.fromRect(rect);
    }

    static toRadians(degrees: number): number {
        return Region.to2Pi((degrees * Math.PI) / 180);
    }
    
    // Converts radians to degrees
    static toDegrees(radians: number): number {
        return Region.to360((radians * 180) / Math.PI);
    }

    static to2Pi(radians: number): number {
        const twoPi = 2 * Math.PI;
        return ((radians % twoPi) + twoPi) % twoPi;
    }
    
    // Normalizes degrees to the range [0, 360)
    static to360(degrees: number): number {
        const fullCircle = 360;
        return ((degrees % fullCircle) + fullCircle) % fullCircle;
    }
}

class TestClass {
    decorateExceptions: boolean = true;
    init() {}

    runTests(): void {
        var self = this as any;
        const className = this.constructor.name;

        for (var key in self) {
            if (stringStartsWith(key, "test") && typeof self[key] === "function") {
                if (this.decorateExceptions) {
                    try {
                        this.init();
                        self[key]();
                    } catch (e) {
                        throw `${className}.${key}: ${e}`;
                    }
                } else {
                    this.init();
                    self[key]();
                }
            }
        }
    }

    assertEquals(actual: any, expected: any, message: string = "", tolerance: number = 0.0001): void {
        const areNumbers = typeof actual === "number" && typeof expected === "number";
        const areArrays = actual instanceof Array && expected instanceof Array;

        if (areNumbers) {
            // Compare numbers within tolerance
            if (Math.abs(actual - expected) > tolerance) {
                throw this.formatError(message, actual, expected);
            }
        } else if (areArrays) {
            // Compare arrays deeply
            if (actual.length !== expected.length) {
                throw this.formatError(message, actual, expected);
            }
            for (let i = 0; i < actual.length; i++) {
                this.assertEquals(actual[i], expected[i], `${message} (index ${i})`, tolerance);
            }
        } else if (actual != expected) { // Use '!=' for type coercion if necessary
            throw this.formatError(message, actual, expected);
        }
    }

    assertThrows(test: () => void, error: string) {
        try {
            test()
        } catch {
            return;
        }
        throw error;
    }

    formatError(message: string, actual: any, expected: any): string {
        const formattedMessage = message ? `${message} - ` : "";
        return `${formattedMessage}Expected: ${this.toString(expected)}, but got: ${this.toString(actual)}`;
    }

    toString(value: any): string {
        if (value instanceof Array) {
            return "[" + f(value).select(x => x?.toString()).join(", ") + "]";
        } else if (typeof value === "object" && value !== null) {
            const result = [];
            for (var key in value) {
                if (value.hasOwnProperty(key)) {
                    result.push(key + ": " + value[key]?.toString());
                }
            }
            return "{" + result.join(", ") + "}";
        } else {
            return value !== undefined ? value.toString() : "undefined";
        }
    }
}
