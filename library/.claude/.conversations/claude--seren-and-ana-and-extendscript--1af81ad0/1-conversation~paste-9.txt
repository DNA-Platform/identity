class CommandQuueue implements Queue {
    private complete: boolean = true;
    private commands: A<string> = f();

    length(): number { return this.commands.length(); }

    peek(): string | undefined {
        if (this.length() == 0) return undefined;
        return this.commands.first();
    }

    dequeue(): string | undefined {
        return this.commands.shift();
    }

    enqueue(task: string, sync: boolean): void {
        this.commands.push(task);
        if (sync) this.sync();
    }

    sync(onComplete?: OnCompleteHandler, onError?: OnErrorHandler): void {
        if (this.length() == 0) return;
        this.complete = false;
        while (this.length() > 0) {
            var cmd = this.dequeue() as string;
            console.log("i", this.length())
            send(cmd, 
                (cmd, i) => {
                    if (i == 0) this.complete = true;
                    if (onComplete) onComplete(cmd, i);
                },
                (cmd, i, error) => {
                    this.complete = true;
                    this.clear();
                    if (onError) onError(cmd, i, error);
                    else throw new Error(`${error.headers}\n${error.body}`);
                },
                this.length()
            );
        }

        while (!this.complete) {
            $BridgeTalk().pump();
            $.sleep(1000);
        }
    }

    clear(): void {
        this.commands = f([]);
    }
}

class Reference<Type = any> {
    _expression: string;
    address(): string { return this._expression; }
    loc(): string { return this.address(); }

    constructor(expression?: string) {
        this._expression = expression || "$.global";
    }

    dot<TSelection = any>(selector: (o: Type) => TSelection): PropertyReference<this, TSelection> {
        var path = str(stringJoin(str(selector.toString()).split(".").skip(1), ".").split(";").first()).replaceAll(" ", "");
        return new PropertyReference<this, TSelection>(this, path.s());
    }

    dott<TSelection = any>(expression: string): PropertyReference<this, TSelection> {
        return new PropertyReference<this, TSelection>(this, expression);
    }

    call<TResult = any>(...args: (Reference | string)[]): FunctionCallReference<TResult> {
        var refs = f(args).select(r => $ref(r)).a();
        return new FunctionCallReference<TResult>(this, refs);
    }

    of<TElement = any>(index: number): IndexReference<this, TElement> {
        return new IndexReference<this, TElement>(this, index);
    }

    cast<T extends Reference = any>(convert?: (ref: this) => T): T {
        if (convert) return convert(this);
        return this as any;
    }

    resolve(): Type {
        var referent: Type = undefined as any;
        eval(`var referent = ${this.address()};`);
        return referent;
    }

    equals(ref: Reference): boolean {
        var thisObject = this.resolve();
        var otherObj = ref.resolve();
        return thisObject === otherObj;
    }

    toString(): string {
        return this.address();
    }
}

class Command {
    _expression: string = "";
    expression(): string { return this._expression; }
    empty(): boolean { return !this.expression(); }

    constructor(expression?: string | Reference) {
        if (expression instanceof Reference) expression = `${expression.address()};`;
        this._expression = expression || "";
    }

    add(cmd: string | Command): Command {
        if (typeof cmd === "string") cmd = new Command(cmd);
        return cmd.empty() ? 
            new Command(this.expression()) : 
            new Command(`${this.expression()}\n${cmd.expression()}`);
    }

    execute(global: boolean = false): void {
        if (global) {
            // Explicitly run eval in the global scope
            eval.call($.global || globalThis, this.expression());
        } else {
            eval(this.expression());
        }
    }

    toString(): string {
        return this.expression();
    }
}

class VariableReference<TVariable = any> extends Reference<TVariable> {
    protected _name: string;

    address(): string { return this.name(); }

    constructor(name: string) {
        super();
        if (name === undefined || name.length === 0) 
            throw new Error("A variable reference must have a name");
        this._name = name;
    }
}

class PropertyReference<TObject extends Reference = any, TProperty = any> extends Reference<TProperty> {
    protected _object: TObject;
    protected _property: string;
    property(): string { return this._property; }
    object(): TObject { return this._object; }
    address(): string { return `${this.object().address()}.${this.property()}` }

    constructor(object: TObject, property: string) {
        super();
        if (property === undefined || property.length == 0 || object === undefined)
            throw new Error("The object and property both need to be defined in a property reference");
        this._object = object;
        this._property = property;
    }
}

class ArrayReference<TObject extends Reference = any, TArray = any, TElement extends IndexReference = any> extends PropertyReference<TObject, TArray> {

    constructor(object: TObject, array: string, selector: (array: ArrayReference, index: number) => TElement) {
        super(object, array);
        this._selector = selector;
    }

    at(index: number): TElement {
        return this._selector(this, index);
    }
}

class IndexReference<TArray extends Reference = any, TElement = any> extends Reference<TElement> {
    protected _array: TArray;
    index: number;
    array(): Reference { return this._array; }
    property(): string { 
        return `[${this.index}]`; 
    }
    address(): string { 
        return `${this.array().address()}[${this.index}]`;
    }

    constructor(array: TArray, index: number) {
        super();
        if (index < 0) throw new Error("Referential indexing does not support negative indexing");
        if (array === undefined || index == undefined)
            throw new Error("Index references must have an array and an index");
        this._array = array;
        this.index = index;
    }
}

class FunctionCallReference<TResult = any> extends Reference<TResult> {
    _func: Reference;
    _args: A<Reference>;
    func(): Reference { return this._func; }
    args(): A<Reference> { return this._args; }
    address(): string {
        var args = this.args().select(a => a.address()).stringJoin(",");
        return `${this.func().address()}(${args})`;
    }

    constructor(func: Reference, args: Reference[]) {
        super();
        if (func === undefined) throw new Error("A FunctionCallReference must have an function")
        if (args === undefined)  
            args = [];
        this._func = func;
        this._args = f(args);
    }
}

class StringLiteralReferece extends Reference<string> {
    _value: string;
    value(): string { return this._value; }

    constructor(value: string | Str| Reference, quoteType: `"` | `'` = `"`) {
        if (value === undefined) throw new Error("String literal unspecified");
        if (value instanceof Reference) value = value.toString();
        if (value instanceof Str) value = value.s();
        super(`${quoteType}${value}${quoteType}`);
        this._value = value;
    }
}

class Var {
    static $<Type = any>(name: string): { equals: (ref: Reference<Type> | string) => VariableAssingment<Type> } {
        return { 
            equals: (ref: Reference<Type> | string) => {
                if (typeof ref === "string") ref = $ref<Type>(ref);
                return new VariableAssingment<Type>(name, ref);
            }
        };
    }

    static $$<Type = any>(name: string): Reference<Type> {
        return VariableAssingment.lookup(name);
    }
}

class VariableAssingment<Type> extends Command {
    _name: string;
    _value: Reference<Type>;
    _variable: Reference<Type>;

    name(): string { return this._name; }
    ref(): Reference<Type> { return this._value as any; }
    expression(): string { return `${this.variable().address()} = ${this.value().address()};` }
    value(): Reference<Type> { return this._value; }
    variable(): Reference<Type> { return this._variable; }

    constructor(name: string, value: Reference<Type>) {
        super();
        this._name = name || `var${VariableAssingment.nextId++}`;
        this._value = value;
        this._variable = VariableAssingment.createVariable(this.name());
    }

    private static nextId = 1;
    private static variables = dictionary<Reference>();
    private static createVariable<Type = any>(name: string): Reference<Type> {
        var variableName = `_____${name}_____`;
        var ref =  new Reference(`$.global.${variableName}`);
        this.variables[name] = ref;
        return ref;
    }

    public static lookup<Type = any>(variableName: string): Reference<Type> {
        var variable = this.variables[variableName];
        if (variable === undefined) throw new Error(`The variable ${variableName} has not been created`);
        return variable;
    }
}

function $ref<Type = any>(expression: string | Reference): Reference<Type> {
    if (expression instanceof Reference) return expression;
    return new Reference<Type>(expression);
}

function $cmd(expression: string | Reference): Command {
    return new Command(expression);
}

function $str(...args: (string | Str | Reference)[]): StringLiteralReferece {
    var value = f(args).stringJoin(" ");
    return new StringLiteralReferece(value);
}

type ExecutionMode = "Remote" | "Local"

class $Eval {
    private static _commands: CommandQuueue | undefined;
    private static _mode: ExecutionMode = "Remote";

    private static commands() { if (!this._commands) this._commands = new CommandQuueue(); return this._commands; }
    private static mode(newMode?: ExecutionMode) { if (newMode !== undefined) this._mode = newMode; return this._mode; }

    static debug: boolean = false;

    static cmd(cmd: Command | Reference | string, onComplete?: OnCompleteHandler, onError?: OnErrorHandler): void {
        onComplete = onComplete || this.onCompleteDefault;
        onError = onError || this.onErrorDefault;
        if (this.mode() === "Remote") {
            if (cmd instanceof Reference) cmd = cmd.address();
            if (cmd instanceof Command) cmd = cmd.expression();
            this.commands().enqueue(cmd, false);
        } else {
            if (!(cmd instanceof Command)) cmd = $cmd(cmd);
            if (this.commands().length() > 0) 
                this.sync(onComplete, onError);

            var cmdExpression = cmd.expression();
            if (!onError) {
                cmd.execute();
                if (onComplete) onComplete(cmdExpression, 0);
            } else {
                try {
                    cmd.execute();
                    if (onComplete) onComplete(cmdExpression, 0);
                } catch (e) {
                    onError(cmdExpression, 0, { 
                        body: (e as any || "Command failed").toSring() as string, 
                        headers: { "Error-Code": -1 } 
                    });
                }
            }
        }
    }

    static sync(onComplete?: OnCompleteHandler, onError?: OnErrorHandler) {
        onComplete = onComplete || this.onCompleteDefault;
        onError = onError || this.onErrorDefault;
        if (this.debug) {
            onComplete = (cmd, i) => { console.log(`$eval.sync ${i} complete:`, cmd); if (onComplete) onComplete(cmd, i); }
            onError = (cmd, i, e) => { console.log(`$eval.sync ${i} FAILED:`, cmd, e?.headers?.["Error-Code"], e?.body); if (onError) onError(cmd, i, e); }
        }
        this.commands().sync(onComplete, onError);
    }

    static onCompleteDefault: OnCompleteHandler | undefined;
    static onErrorDefault: OnErrorHandler | undefined;
}

function $eval(cmd: Command | Reference | string, onComplete?: OnCompleteHandler, onError?: OnErrorHandler): void {
    $Eval.cmd(cmd, onComplete, onError);
}

function $sync(onComplete?: OnCompleteHandler, onError?: OnErrorHandler) {
    $Eval.sync(onComplete, onError);
}

class DocumentReference<Type = any> extends Reference<Type> {
    private _typename: string;
    protected _document: $Document;
    private _parent: Reference<any> | undefined;

    app(): $Application { return $app; }
    document(): $Document { return this._document; }
    typename(): string { return this._typename; }
    parent<TParent extends DocumentObject = any>(): Reference<TParent> { if (!this._parent) this._parent = this.dot(o => (o as any).parent); return this._parent; }

    constructor(ref: Reference<Type> | string, typename: string, document: $Document) {
        super(ref instanceof Reference ? ref.address() : ref)
        this._typename = typename;
        this._document = this instanceof $Document ? this : document;
    }

    protected $get<TProperty>(propertySelector: ((o: Type) => any) | string | Reference, value?: Reference | string): Reference<TProperty> {
        var property = propertySelector instanceof Reference ? 
            propertySelector : typeof propertySelector === "string" ? 
            this.dott(propertySelector) : 
            this.dot(propertySelector);

        if (value !== undefined) {
            console.log(value);
            var cmd = $cmd(
                `if (${property} !== undefined) { 
                    ${property} = ${value}; 
                } else { 
                    throw new Error(${$str(property, "does not exist on", this.typename())}); 
                }`
            );
            $eval(cmd);
        }

        return property;
    }

    protected $try(methodSelector: ((o: Type) => any) | string | Reference, ...args: (Reference | string)[]): void {
        var method = methodSelector instanceof Reference ? 
            methodSelector : typeof methodSelector === "string" ? 
            this.dott(methodSelector) : 
            this.dot(methodSelector);

        var cmd = $cmd(
            `if (${method} !== undefined) { 
                ${method.call(...args)}; 
            } else { 
                throw new Error(${$str(method, "does not exist on", this.typename())}); 
            }`
        );

        $eval(cmd);
    }
}

class DocumentObject<TObject = any> extends DocumentReference<TObject> {

        return this.$get("name", newName !== undefined ? $str(newName) : undefined); 
    }

    remove(): void
    remove(...args: Reference[]): void {
        this.$try("remove");
    }
}

class DocumentCollection<TCollection = any, $TElement extends DocumentObject = any, TElement = any> extends DocumentReference<TCollection> {
    private _etypename: string;
    private _length: Reference<string> | undefined;

    app(): $Application { return $app; }
    length(): Reference<string> { if (!this._length) this._length = this.dot(o => (o as any).length); return this._length; }

    constructor(ref: Reference<TCollection> | string, typename: string, document: $Document) {
        super(ref, typename, document)
        this._etypename = typename.slice(0, -1);
    }

    at(index: number): $TElement {
        return $docref<$TElement>(this.of(index), this._etypename, this.document());
    }

    getByName(name: string | Reference): $TElement {
        if (typeof name === "string") name = new StringLiteralReferece(name); 
        return $docref<$TElement>(this.dott<TElement>(`getByName(${name})`), this._etypename, this.document());
    }

    add(): $TElement
    add(...args: Reference[]): $TElement {
        $eval(this.dot(o => (o as any).add).call(...args));
        return this.at(0);
    }

    remove(index: number): void {
        this.$try(this.at(index).dott("remove"));
    }

    removeAll(): void {
        this.$try("removeAll");
    }

    private static nextId = 1;
    private static variableName = "element";
    protected static getNextVariableName() { return `${this.variableName}${this.nextId++}`; }
}
