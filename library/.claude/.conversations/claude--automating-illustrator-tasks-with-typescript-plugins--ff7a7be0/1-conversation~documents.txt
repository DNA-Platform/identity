/// <reference path="./core.ts" />

function getResource(fileName: string): Document {
    return getDocument(combinePaths("resources", fileName), true, false);
}

/**
 * Gets, activates, or creates an Illustrator document based on the provided parameters.
 * Checks both open documents and the documents directory.
 * 
 * @param fileName - The name of the document to find/create
 * @param activate - Whether to make the found/created document active
 * @param create - Whether to create a new document if one with the specified name isn't found
 * 
 * @returns The requested Document object
 * @throws Error if document isn't found and create=false
 */
function getDocument(fileName?: string, activate: boolean = false, create: boolean = false): Document {
    // Case 1: No arguments - return active document
    if (!fileName) {
        if (app.documents.length === 0) {
            throw new Error("No documents are open.");
        }
        return app.activeDocument;
    }

    // Case 2: Try to find in open documents
    for (let i = 0; i < app.documents.length; i++) {
        if (app.documents[i].name === fileName) {
            if (activate) {
                app.documents[i].activate();
            }
            return app.documents[i];
        }
    }

    // Case 3: Try to find in documents directory
    const file = new File(combinePaths(DOCUMENTS_PATH, fileName));
    if (file.exists) {
        const doc = loadDocument(fileName);
        if (activate) {
            doc.activate();
        }
        return doc;
    }

    // Case 4: Document not found anywhere - create or throw error
    if (create) {
        const newDoc = app.documents.add();
        if (activate) {
            newDoc.activate();
        }
        saveDocument(newDoc, fileName);
        return newDoc;
    }

    throw new Error(`Document '${fileName}' not found in open documents or documents directory.`);
}

/**
 * Loads a document from the documents directory
 * @param fileName - The name of the document
 * @returns The loaded Document object
 * @throws Error if document cannot be found or loaded
 */
function loadDocument(fileName: string): Document {
    const file = new File(combinePaths(DOCUMENTS_PATH, fileName));
    
    if (!file.exists) {
        throw new Error(`Document '${fileName}' not found in documents directory.`);
    }

    try {
        return app.open(file);
    } catch (error: any) {
        throw new Error(`Failed to load document '${fileName}': ${error.message}`);
    }
}

/**
 * Saves a document to the documents directory
 * @param doc - The document to save
 * @param fileName - Name to save as
 * @throws Error if save fails
 */
function saveDocument(doc: Document, fileName: string): void {
    const file = new File(combinePaths(DOCUMENTS_PATH, fileName));
    
    try {
        doc.saveAs(file);
    } catch (error: any) {
        throw new Error(`Failed to save document '${fileName}': ${error.message}`);
    }
}

// Gets a layer from a document by index (supports negative indexing) or name
// Usage:
//   const layer = getLayer(doc, 0);        // Get the first layer
//   const layer = getLayer(doc, -1);       // Get the last layer
//   const layer = getLayer(doc, "LayerName"); // Get layer by name
function getLayer(doc: Document, identifier?: number | string): Layer {
    if (doc.layers.length === 0) {
        throw new Error("No layers in the document.");
    }

    if (identifier === undefined) {
        return doc.layers[0];  // Default to the first layer if no identifier is provided
    }

    if (typeof identifier === "number") {
        // Adjust for negative indexing
        const index = identifier < 0 ? doc.layers.length + identifier : identifier;
        return doc.layers[index];
    } else if (typeof identifier === "string") {
        for (let i = 0; i < doc.layers.length; i++) {
            if (doc.layers[i].name === identifier) {
                return doc.layers[i];
            }
        }
        throw new Error(`Layer with the name '${identifier}' not found.`);
    } else {
        throw new Error("Invalid identifier. Must be a number (index) or string (layer name).");
    }
}

// Gets an item in a layer by index (supports negative indexing) or name
// Usage:
//   const item = getItem(layer, 0);         // Get the first item in the layer
//   const item = getItem(layer, -1);        // Get the last item in the layer
//   const item = getItem(layer, "ItemName"); // Get item by name
function getItem(layer: Layer, identifier?: number | string): PageItem {
    if (layer.pageItems.length === 0) {
        throw new Error("No items in the layer.");
    }

    if (identifier === undefined) {
        return layer.pageItems[0];  // Default to the first item if no identifier is provided
    }

    if (typeof identifier === "number") {
        // Adjust for negative indexing
        const index = identifier < 0 ? layer.pageItems.length + identifier : identifier;
        return layer.pageItems[index];
    } else if (typeof identifier === "string") {
        for (let i = 0; i < layer.pageItems.length; i++) {
            if (layer.pageItems[i].name === identifier) {
                return layer.pageItems[i];
            }
        }
        throw new Error(`Item with the name '${identifier}' not found.`);
    } else {
        throw new Error("Invalid identifier. Must be a number (index) or string (item name).");
    }
}

// Groups items in a layer by indices (supports negative indexing) or names; groups all if no arguments are provided
// Usage:
//   groupItems(layer);                    // Group all items in the layer
//   groupItems(layer, 0, "ItemName", -2); // Group specific items by index or name
// Groups items in a layer by indices (supports negative indexing) or names; groups all if no arguments are provided
// Usage:
//   groupItems(layer);                    // Group all items in the layer
//   groupItems(layer, 0, "ItemName", -2); // Group specific items by index or name
function groupItems(layer: Layer, ...identifiers: (number | string)[]): GroupItem {
    if (layer.pageItems.length === 0) {
        throw new Error("No items in the layer to group.");
    }

    const itemsToGroup: PageItem[] = [];

    if (identifiers.length === 0) {
        // No arguments provided, group all items
        for (let i = 0; i < layer.pageItems.length; i++) {
            itemsToGroup.push(layer.pageItems[i]);
        }
    } else {
        // Process each argument to add specific items to the group
        for (let j = 0; j < identifiers.length; j++) {
            const identifier = identifiers[j];
            if (typeof identifier === "number") {
                // Adjust for negative indexing
                const index = identifier < 0 ? layer.pageItems.length + identifier : identifier;
                itemsToGroup.push(layer.pageItems[index]);
            } else if (typeof identifier === "string") {
                // Get item by name
                let found = false;
                for (let i = 0; i < layer.pageItems.length; i++) {
                    if (layer.pageItems[i].name === identifier) {
                        itemsToGroup.push(layer.pageItems[i]);
                        found = true;
                        break;
                    }
                }
                if (!found) {
                    throw new Error(`Item with the name '${identifier}' not found.`);
                }
            } else {
                throw new Error("Invalid identifier. Must be a number (index) or string (item name).");
            }
        }
    }

    // Group the items
    const group = layer.groupItems.add();
    for (let m = 0; m < itemsToGroup.length; m++) {
        itemsToGroup[m].move(group, ElementPlacement.PLACEATEND);
    }

    return group;
}
