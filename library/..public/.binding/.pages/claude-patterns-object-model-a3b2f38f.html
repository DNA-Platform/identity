    return this._a.at(index); 
  }

  // LINQ-style operations (your favorites)
  select<U>(selector: (item: T, i: number) => U): A<U> {
    return new A(this._a.map(selector));
  }

  where(predicate: (item: T, i: number) => boolean): A<T> {
    return new A(this._a.filter(predicate));
  }

  selectMany<U>(selector: (item: T, i: number) => U[]): A<U> {
    return new A(this._a.flatMap(selector));
  }

  distinct(comparer?: (a: T, b: T) => boolean): A<T> {
    if (!comparer) {
      return new A([...new Set(this._a)]);
    }
    // Custom comparer logic
    const result: T[] = [];
    for (const item of this._a) {
      if (!result.some(r => comparer(r, item))) {
        result.push(item);
      }
    }
    return new A(result);
  }

  aggregate<U>(seed: U, func: (acc: U, current: T, i: number) => U): U {
    return this._a.reduce(func, seed);
  }

  // Modern array methods (use native implementations)
  first(predicate?: (item: T, i: number) => boolean): T | undefined {
    return predicate ? this._a.find(predicate) : this._a[0];
  }

  last(predicate?: (item: T, i: number) => boolean): T | undefined {
    if (!predicate) return this._a.at(-1);
    // Find from end
    for (let i = this._a.length - 1; i >= 0; i--) {
      if (predicate(this._a[i], i)) return this._a[i];
    }
    return undefined;
  }

    return predicate ? this._a.some(predicate) : this._a.length > 0;
  }

  all(predicate: (item: T, i: number) => boolean): boolean {
    return this._a.every(predicate);
  }

  // Chaining operations
  skip(count: number): A<T> {
    return new A(this._a.slice(count));
  }

  take(count: number): A<T> {
    return new A(this._a.slice(0, count));
  }

  orderBy(selector?: (item: T) => any): A<T> {
    const copy = [...this._a];
    if (!selector) {
      copy.sort();

        const vb = selector(b);
        return va < vb ? -1 : va > vb ? 1 : 0;

export function assertArrayEquals<T>(
  actual: T[],
  expected: T[],
  message: string = ""
): void {
  if (actual.length !== expected.length) {
    throw new Error(
      `${message}: Array lengths differ. Expected ${expected.length} but got ${actual.length}`
    );
  }
  for (let i = 0; i < actual.length; i++) {
    if (actual[i] !== expected[i]) {
      throw new Error(
        `${message}: Arrays differ at index ${i}. Expected ${expected[i]} but got ${actual[i]}`
      );
    }
  }
}

export function assertPositionEquals(
  actual: [f32, f32],
  expected: [f32, f32],

  message: string = ""
): void {

Why @as-pect for AssemblyScript Testing
@as-pect is the standard testing framework for AssemblyScript because:
AssemblyScript-specific: Regular JavaScript test runners (Jest, Vitest) can't run AssemblyScript code directly - it needs to be compiled to WASM first
Built for WASM: It compiles your tests to WebAssembly and runs them in a WASM runtime
Familiar API: Uses Jest-like syntax (describe, it, expect) that developers know
No alternatives: It's essentially the only mature testing framework for AssemblyScript

# Run tests in watch mode

# Run specific test file

 * Extends native Array for maximum compatibility

export class A<T> extends Array<T> {
  constructor(items?: T[] | ArrayLike<T>) {
    super();
    if (items) {
      if (Array.isArray(items)) {
        // Can't use spread in AssemblyScript, must iterate
        for (let i = 0; i < items.length; i++) {
          this.push(items[i]);
        }

export function evaluateNegativeIndex(index: i32, length: i32): i32 {
  const evaluated = index < 0 ? length + index : index;

export function clone<T>(obj: T): T {
  // In AssemblyScript, we need type-specific cloning
  if (obj instanceof Array) {
    const arr = obj as Array<any>;
    const result = new Array<any>();
    for (let i = 0; i < arr.length; i++) {
      result.push(arr[i]);
    }
    return result as T;
  }
  // For primitives, just return
  return obj;
}

/**
 * Array equality check for AssemblyScript

export function arrayEquals<T>(first: T[], second: T[]): boolean {
  if (!first && !second) return true;
  if (!first || !second) return false;
  if (first.length !== second.length) return false;

  for (let i = 0; i < first.length; i++) {
    // For numbers, use tolerance
    if (typeof first[i] === 'number' && typeof second[i] === 'number') {
      const diff = Math.abs((first[i] as f64) - (second[i] as f64));
      if (diff > 0.0001) return false;
    } else if (first[i] !== second[i]) {
      return false;
    }
  }
  return true;
}

export class TestClass {

    // Handle numbers with tolerance
    if (typeof actual === 'number' && typeof expected === 'number') {
      const diff = Math.abs((actual as f64) - (expected as f64));
      if (diff > this.tolerance) {
        throw new Error(
          `${message}: Expected ${expected} but got ${actual}`
        );
      }
      return;

    // Handle arrays
    if (actual instanceof Array && expected instanceof Array) {
      if (!arrayEquals(actual as Array<any>, expected as Array<any>)) {

          `${message}: Arrays not equal`
        );
      }
      return;

      );
    }
  }
