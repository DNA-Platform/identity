from abc import ABC, abstractmethod
from typing import List, Union, Optional, TypeVar, Generic, Any, Callable

T = TypeVar('T', bound='Element')

class Element(ABC):
    """Base class for all SVG elements. Immutable once created."""
    
    def __init__(
        self,
        attributes: List['Attribute'] = None, 
        content: Union[List['Element'], 'Element', str, None] = None,
        annotation: Optional[str] = None
    ):
        """Initialize an immutable element with attributes, content, and annotation."""
        self._attributes = attributes or []
        self._content = content
        self._annotation = annotation
        self._parent = None
        
        # Internal state for pending changes (not applied until update)
        self._dirty = False
        self._pending_changes = {}
        
    @abstractmethod
    def render(self, indent: int = 0) -> str:
        """Render element as SVG string with appropriate indentation."""
        pass
    
    def update(self: T) -> T:
        """
        Return a new element with all pending changes applied.
        Maintains immutability by creating a new instance.
        """
        pass
    
    def visit(self, visitor: Callable[['Element'], None]) -> None:
        """
        Accept a visitor function that will be called on this element
        and recursively on all attributes and child elements.
        """
        # Call visitor on self
        visitor(self)
        
        # Visit all attributes
        for attr in self._attributes:
            attr.visit(visitor)
        
        # Visit all content if it's a list of elements
        if isinstance(self._content, list):
            for child in self._content:
                if isinstance(child, Element):
                    child.visit(visitor)
        # Visit single element content
        elif isinstance(self._content, Element):
            self._content.visit(visitor)
    
    @property
    def parent(self) -> Optional['Element']:
        """Get the parent element, if any."""
        return self._parent
    
    @parent.setter
    def parent(self, value: 'Element') -> None:
        """Set parent reference (used internally, doesn't violate immutability)."""
        self._parent = value


class ElementCollection:
    """
    An immutable collection of elements that allows deferred modifications.
    Changes only take effect when update() is called.
    """
    
    def __init__(self, elements: List[Element] = None):
        """Initialize with optional list of elements."""
        self._elements = elements or []
        self._pending_adds = []
        self._pending_removes = []
        self._pending_replaces = {}  # {index: new_element}
        
    def add(self, element: Element) -> None:
        """Queue an element to be added."""
        self._pending_adds.append(element)
        
    def remove(self, element: Element) -> None:
        """Queue an element to be removed."""
        self._pending_removes.append(element)
        
    def replace(self, index: int, element: Element) -> None:
        """Queue a replacement of element at index."""
        self._pending_replaces[index] = element
        
    def update(self) -> 'ElementCollection':
        """Return a new collection with all pending changes applied."""
        # Create a new list with all changes applied
        new_elements = [e for e in self._elements if e not in self._pending_removes]
        
        # Apply replacements
        for index, element in self._pending_replaces.items():
            if index < len(new_elements):
                new_elements[index] = element
        
        # Add new elements
        new_elements.extend(self._pending_adds)
        
        return ElementCollection(new_elements)

    Mutable container for a Graphic, manages editing history.
    The only component in the system that maintains state.

    def __init__(self, graphic: Optional[Graphic] = None):
        """Initialize with an optional graphic."""
        self._current_graphic = graphic
        self._history = [graphic] if graphic else []
        self._history_position = 0 if graphic else -1
    
    @property
    def graphic(self) -> Optional[Graphic]:
        """Get the current immutable graphic."""
        return self._current_graphic
    
    def set_graphic(self, graphic: Graphic) -> None:
        """
        Set a new graphic state, adding it to history.
        Clears any redo states.
        """
        # Add to history, removing any future states
        if self._history_position < len(self._history) - 1:
            self._history = self._history[:self._history_position + 1]
        
        self._history.append(graphic)
        self._history_position = len(self._history) - 1
        self._current_graphic = graphic

        """
        Undo the last change, moving back in history.
        Returns the restored Graphic or None if at beginning.
        """
        if self._history_position > 0:
            self._history_position -= 1
            self._current_graphic = self._history[self._history_position]
            return self._current_graphic
        return None

        """
        Redo a previously undone change.
        Returns the restored Graphic or None if at end.
        """
        if self._history_position < len(self._history) - 1:
            self._history_position += 1
            self._current_graphic = self._history[self._history_position]
            return self._current_graphic
        return None
