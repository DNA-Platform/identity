function getAllDescriptors(obj) {
    const descriptors = {};
    let current = obj;
    
    while (current && current !== Object.prototype) {
        Object.assign(descriptors, Object.getOwnPropertyDescriptors(current));
        current = Object.getPrototypeOf(current);
    }

The invariant question - YES with caveats:
javascript
// Flatten an object by binding all methods to 'this'
function flattenBindings(obj) {
    const descriptors = Object.getOwnPropertyDescriptors(obj);
    const bound = {};
    
    for (const [key, desc] of Object.entries(descriptors)) {
        if (typeof desc.value === 'function') {
            // Bind functions
            bound[key] = { value: desc.value.bind(obj) };
        } else {
            // Copy as-is (including getters/setters)
            bound[key] = desc;
        }
    }
    
    Object.defineProperties(obj, bound);
}

✓ Getters/setters with closures stay intact
✓ Field initializers already created own properties
✓ Constructor assignments already on this
This BREAKS invariance IF:
✗ Methods meant to be called with different this (.call(), .apply())
✗ Methods passed as callbacks expecting rebinding

No, TypeScript doesn't support generic arity overloading like C# does. You cannot have:
typescript
// ❌ NOT ALLOWED in TypeScript
type C = {};

typescript
// ✓ Single definition with defaults
type C<T = never, U = never> = {
    // Use conditional types to handle different arities
};

// Usage:
C           // C<never, never>
C<string>   // C<string, never>
C<string, number>  // C<string, number>

typescript
type C<T = never, U = never> =

    U extends never ? { type: 'unary', value: T } :

Research on Function.prototype.toString Specification
1. The ES2018/ES2019 Revolution
The ECMAScript 2018/2019 specification brought a major revision to Function.prototype.toString through a proposal by Michael Ficarra, fundamentally changing how function string representations work. Prior to this revision, engines had discretion over whether to return source code or synthesized representations. The new spec mandates that whenever possible, toString() must return the exact source code as it was declared, including all whitespace and comments. This requirement makes toString() deterministic and reliable for code analysis tools.
2. Source Code Fidelity Requirement
The specification states that for functions defined using ECMAScript code, toString must return the source text slice from the beginning of the first token to the end of the last token matched by the appropriate grammar production. This means the returned string is a precise substring of the original source file, preserving formatting, comments, and even unusual whitespace. However, implementations are not required to retain source text for all functions, particularly in memory-constrained environments.
3. Native Function String Format
When source code is unavailable—such as for built-in functions, bound functions, or host-provided functions—toString() must return a standardized "native function string" following the pattern: "function" BindingIdentifier? "(" FormalParameters ")" "{ [native code] }". This placeholder format is syntactically valid but clearly distinguishes native code from user-defined functions. For well-known intrinsic objects like Array or Math.pow, the function name must appear in the result.
4. Context-Dependent Syntax Validity
An important nuance is that toString() may return code that is only syntactically valid within its original syntactic context. For example, a method extracted from a class will toString as "method() { body }" which is valid as a method definition but not as a standalone function declaration. This reflects the reality that not all function forms can be meaningfully represented outside their lexical context.
5. Arrow Function Representation
Arrow functions have a distinctive toString representation that always includes the "=>" token. Multi-parameter arrow functions return strings like "(a, b) => expression", while single-parameter arrows may omit parentheses: "a => expression". This makes arrow functions uniquely identifiable through the presence of the arrow operator, which never appears in traditional function syntax.
6. Method Shorthand Patterns
ES6 method shorthand syntax produces toString results that start directly with the method name followed by parentheses, without the "function" keyword. For example, { method() {} }.method.toString() returns "method() {}", and computed property methods like { [0]() {} }[0].toString() return "[0]() {}". Generator methods include the asterisk: "*method() {}", and getters/setters include their keywords: "get method() {}" or "set method(x) {}".
7. Class Constructor Representation
Class definitions have their own toString behavior: calling toString() on the class itself returns the entire class declaration including "class ClassName { ... }" with all methods. This differs from calling toString() on the constructor property, which would show just the constructor method. Classes are distinguished by starting with the "class" keyword, making them easily identifiable.
8. Bound Function Behavior
Functions created via Function.prototype.bind() lose their source code representation and instead return a native function string. For example, (function foo() {}).bind(null).toString() returns "function () { [native code] }", effectively hiding the original function's implementation. This makes it impossible to distinguish bound functions from native built-ins through toString() alone.
9. Function Constructor Pattern
Functions created dynamically using the Function constructor (new Function('a', 'b', 'return a + b')) have a special toString format that shows a synthesized function declaration named "anonymous". The format is "function anonymous(params) { body }" where params and body are the strings passed to the constructor. This makes dynamically-created functions distinguishable from statically-defined ones.
10. Async and Generator Modifiers
Async functions and generators preserve their modifiers in toString() output. Async functions return strings beginning with "async function name() {}", async generators with "async function* name() {}", and regular generators with "function* name() {}". These keywords appear before the function keyword and are preserved exactly as written in the source, making it possible to identify these special function types through string analysis.

export function parseFunctionInfo(func: Function): $FunctionInfo {
    const pattern = getFunctionPattern();
    const str = func.toString();
    const name = func.name || '';
    const match = pattern.exec(str);
    
    if (!match?.groups) {
        return { 
            form: 'unknown',
            name,
            async: undefined,
            generator: undefined,
            native: undefined,
            params: { count: undefined, rest: undefined }
        };
    }
    
    const g = match.groups;
    const async = !!g.async;
    const generator = !!g.star;
    const native = !!g.native;
    
    let form: $FunctionInfo['form'];
    if (g.class) form = 'class';
    else if (g.get) form = 'getter';
    else if (g.set) form = 'setter';
    else if (g.arrow || g.arrowParam) form = 'lambda';
    else if (g.function) form = 'function';
    else form = 'method';
    
    const paramStr = g.arrowParams || g.arrowParam || g.params || '';
    const trimmed = paramStr.trim();
    
    if (!trimmed) {
        return { form, name, async, generator, native, params: { count: 0, rest: false } };
    }
    
    const parts = trimmed.split(',').map(p => p.trim());
    const last = parts[parts.length - 1];
    const rest = last.startsWith('...');
    const count = rest ? parts.length - 1 : parts.length;
    
    return { form, name, async, generator, native, params: { count, rest } };
}

let $pattern: RegExp | undefined;
function getFunctionPattern(): RegExp {
    if ($pattern) return $pattern;
    
    const ws = '\\s+';
    const ws0 = '\\s*';
    const id = '[a-zA-Z_$][a-zA-Z0-9_$]*';
    const paramContent = '[^)]*';
    const rest = '.*';
    
    const asyncKw = 'async';
    const classKw = 'class';
    const getKw = 'get';
    const setKw = 'set';
    const funcKw = 'function';
    const starSym = '\\*';
    const arrowSym = '=>';
    const nativeText = '\\[native code\\]';
    
    const asyncMod = `(?<async>${asyncKw}${ws})?`;
    const keywords = 
        `(?:` +
            `(?<class>${classKw})|` +
            `(?<get>${getKw})|` +
            `(?<set>${setKw})|` +
            `(?<function>${funcKw})${ws0}(?<star>${starSym})?` +
        `)?`;
    const postKeywordWs = `${ws}?`;
    const optName = `(?:${id}${ws0})?`;
    const arrowParenParams = `\\((?<arrowParams>${paramContent})\\)${ws0}(?<arrow>${arrowSym})`;
    const arrowSingleParam = `(?<arrowParam>${id})${ws0}${arrowSym}`;
    const arrowSyntax = `(?:${arrowParenParams}|${arrowSingleParam})?`;
    const regularParams = `(?:\\((?<params>${paramContent})\\))?`;
    const nativeMarker = `${rest}(?<native>${nativeText})?`;
    
    $pattern = new RegExp(
        `^${asyncMod}${keywords}${postKeywordWs}${optName}${arrowSyntax}${regularParams}${nativeMarker}`
    );

}
