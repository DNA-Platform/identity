import { Point, PatternParams } from './types';

// Ensures pattern tiles seamlessly by wrapping coordinates
export function wrapCoordinate(coord: f32, max: f32): f32 {
  while (coord < 0) coord += max;
  while (coord >= max) coord -= max;
  return coord;
}

// Apply transformation matrix for rotation and scale
export function transformPoint(
  point: Point, 
  params: PatternParams
): Point {
  const rad = params.rotation * Math.PI / 180;
  const cos = Math.cos(rad) as f32;
  const sin = Math.sin(rad) as f32;
  
  // Apply rotation and scale
  const x = (point.x * cos - point.y * sin) * params.scale;
  const y = (point.x * sin + point.y * cos) * params.scale;
  
  return new Point(
    x + params.offsetX,
    y + params.offsetY
  );
}

// Calculate seamless boundary conditions
export function calculateSeamlessOffset(
  index: i32,
  gridSize: i32,
  tileSize: f32
): f32 {
  const row = index / gridSize;
  const col = index % gridSize;
  return (col as f32) * tileSize;
}

export function generateSeamlessTile(
  motifPath: string,
  scaleValue: f32,
  rotationValue: f32,
  tileWidth: f32,
  tileHeight: f32
): string {
  const params = new PatternParams();
  params.scale = scaleValue;
  params.rotation = rotationValue;
  params.tileWidth = tileWidth;
  params.tileHeight = tileHeight;
  
  // Generate 3x3 grid for seamless preview
  let svg = `<svg width="${tileWidth}" height="${tileHeight}" xmlns="http://www.w3.org/2000/svg">`;
  svg += `<defs><g id="motif">${motifPath}</g></defs>`;
  
  // Center tile and 8 surrounding tiles for seamlessness
  for (let row = -1; row <= 1; row++) {
    for (let col = -1; col <= 1; col++) {
      const x = col * tileWidth / params.scale;
      const y = row * tileHeight / params.scale;
      
      svg += `<use href="#motif" 
              transform="translate(${x}, ${y}) 
                        rotate(${params.rotation}, ${tileWidth/2}, ${tileHeight/2})
                        scale(${params.scale})"
              opacity="${row === 0 && col === 0 ? 1 : 0.3}"/>`;
    }
  }
  
  svg += `</svg>`;
  return svg;
}

export function validateSeamless(
  leftEdge: string,
  rightEdge: string,
  topEdge: string,
  bottomEdge: string

  // Check if edges match for seamless tiling

// Generate unique pattern hash
export function generatePatternHash(
  motif: string,
  timestamp: i64
): string {
  // Simple hash for now - would use crypto in production
  let hash: i64 = timestamp;
  for (let i = 0; i < motif.length; i++) {
    hash = ((hash << 5) - hash) + motif.charCodeAt(i);
    hash = hash & hash; // Convert to 32bit
  }
  return hash.toString();

// Generate a single tile element
export function createTileElement(
  x: f32,
  y: f32,
  size: f32,
  rotation: f32
): string {
  const centerX = x + size / 2;
  const centerY = y + size / 2;
  
  // Simple geometric shape for now
  return `<rect x="${x}" y="${y}" width="${size}" height="${size}" 
          fill="none" stroke="black" stroke-width="2"
          transform="rotate(${rotation} ${centerX} ${centerY})" />`;
}

// Main pattern generation function
export function generatePattern(
  gridSize: i32,
  scale: f32,
  rotation: f32,
  width: i32,
  height: i32
): string {
  const tileSize = (width as f32) / (gridSize as f32) * scale;
  
  let svg = `<svg width="${width}" height="${height}" viewBox="0 0 ${width} ${height}" xmlns="http://www.w3.org/2000/svg">`;
  svg += `<rect width="${width}" height="${height}" fill="white" />`;
  
  // Generate grid of tiles
  for (let row = 0; row < gridSize; row++) {
    for (let col = 0; col < gridSize; col++) {
      const x = (col as f32) * tileSize;
      const y = (row as f32) * tileSize;
      svg += createTileElement(x, y, tileSize, rotation);
    }
  }
  
  svg += `</svg>`;
  return svg;
}

const fs = require('fs');
const path = require('path');

async function bundle() {
  // Read the built index.html
  let html = fs.readFileSync('./build/index.html', 'utf8');
  
  // Find and inline all CSS
  const cssRegex = /<link rel="stylesheet" href="([^"]+)"/g;
  let match;
  while ((match = cssRegex.exec(html)) !== null) {
    const cssPath = path.join('./build', match[1]);
    if (fs.existsSync(cssPath)) {
      const css = fs.readFileSync(cssPath, 'utf8');
      html = html.replace(match[0], `<style>${css}</style>`);
    }
  }
  
  // Find and inline all JS
  const scriptRegex = /<script src="([^"]+)"/g;
  html = html.replace(scriptRegex, (match, src) => {
    const jsPath = path.join('./build', src);
    if (fs.existsSync(jsPath)) {
      const js = fs.readFileSync(jsPath, 'utf8');
      return `<script>${js}`;
    }
    return match;
  });
  
  // Inline WASM as base64
  const wasm = fs.readFileSync('./build/release.wasm');
  const wasmBase64 = wasm.toString('base64');
  html = html.replace('</body>', `
    <script>
      window.WASM_BASE64 = "${wasmBase64}";
    </script>
    </body>
  `);
  
  // Write single file
  fs.writeFileSync('pattern.html', html);
  console.log('Created pattern.html - single file!');

const fs = require('fs');

function bundleToSingleHTML() {

  // Find and inline all CSS files
  const cssFiles = html.match(/<link[^>]+href="([^"]+\.css)"[^>]*>/g) || [];
  cssFiles.forEach(tag => {
    const href = tag.match(/href="([^"]+)"/)[1];
    const cssPath = path.join('build', href.startsWith('/') ? href.slice(1) : href);
